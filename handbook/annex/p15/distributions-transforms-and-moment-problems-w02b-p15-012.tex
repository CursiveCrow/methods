\begin{annexitem}{annex:p15-w02b-p15-012-poisson-identity}{thm:p15-w02b-p15-012-poisson-identity}{The Poisson identity by absolute rearrangement}
\begin{proof}
Let \(|a_j|\leq A\) for all \(j\). Expanding the iterated difference gives
\[
 \Delta^r a_i=\sum_{k=0}^r\binom rk(-1)^{r-k}a_{i+k},
\]
which is the case \(r=0\) trivially and follows for \(r+1\) from the case
\(r\) by \(\Delta^{r+1}a_i=\Delta^ra_{i+1}-\Delta^ra_i\) and Pascal's rule.
Consider the double series
\[
 \sum_{r=0}^\infty\sum_{k=0}^r\frac{x^r}{r!}\binom rk(-1)^{r-k}a_{i+k}.
\]
The sum of the absolute values of its terms is at most
\(\sum_r|x|^r2^rA/r!=Ae^{2|x|}<\infty\). Hence the double series converges
absolutely, its value equals the left side of (5.1), and its terms may be
summed in any order. Put \(r=k+m\) with \(k,m\geq0\). Since
\(\binom{k+m}k/(k+m)!=1/(k!\,m!)\), the sum becomes
\[
 \sum_{k=0}^\infty\frac{x^k}{k!}a_{i+k}\sum_{m=0}^\infty\frac{(-x)^m}{m!}
     =e^{-x}\sum_{k=0}^\infty\frac{x^k}{k!}a_{i+k},
\]
and the last series converges absolutely because \(|a_{i+k}|\leq A\). This is
(5.1).

For (5.2), put \(a_j=u(jh)\), \(i=0\) and \(x=\theta/h\). By the definition of
the normalized difference, \(\Delta_h^ru(0)=h^{-r}\Delta^ra_0\), so the term
\(x^r\Delta^ra_0/r!\) equals \(\theta^r\Delta_h^ru(0)/r!\). The right side of
(5.1) becomes \(e^{-\theta/h}\sum_ju(jh)(\theta/h)^j/j!\), which is
\(\mathbb E\,u(hN_{\theta/h})\). The variable \(hN_{\theta/h}\) has mean
\(h\cdot\theta/h=\theta\) and variance \(h^2\cdot\theta/h=h\theta\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-generalized-taylor}{thm:p15-w02b-p15-012-generalized-taylor}{The generalized Taylor formula and the two examples}
\begin{proof}
Fix \(t\geq0\) and put \(v(x)=u(t+x)\) for \(x\geq0\). Then \(v\) is bounded
and continuous with \(\sup|v|\leq\|u\|\), and \(\Delta_h^rv(0)=\Delta_h^ru(t)\)
for every \(r\), since both are built from the same values
\(u(t+jh)\). By \cref{thm:p15-w02b-p15-012-poisson-identity},
\[
 \sum_{r=0}^\infty\frac{\theta^r}{r!}\Delta_h^ru(t)=\mathbb E\,v(hN_{\theta/h}).
\]
At \(\theta=0\) the variable \(hN_0\) is zero, so the value is \(v(0)=u(t)\).

Let \(b>0\) and \(\epsilon>0\). The function \(v\) is uniformly continuous on
\([0,b+1]\), so there is \(\delta\in(0,1]\) with \(|v(x)-v(y)|<\epsilon\)
whenever \(x,y\in[0,b+1]\) and \(|x-y|<\delta\). Let
\(0\leq\theta\leq b\) and write \(Y=hN_{\theta/h}\), a nonnegative variable.
On the event \(|Y-\theta|<\delta\) we have \(Y\in[0,b+1]\), so
\(|v(Y)-v(\theta)|<\epsilon\). On the complementary event
\(|v(Y)-v(\theta)|\leq2\|u\|\), and by Chebyshev's inequality that event has
probability at most \(\operatorname{Var}(Y)/\delta^2=h\theta/\delta^2\leq
hb/\delta^2\). Therefore
\[
 \bigl|\mathbb E\,v(Y)-v(\theta)\bigr|\leq\epsilon+\frac{2\|u\|hb}{\delta^2},
\]
and the right side is below \(2\epsilon\) for all \(0\leq\theta\leq b\) once
\(h<\epsilon\delta^2/(2\|u\|b+1)\). Since \(v(\theta)=u(t+\theta)\) and
\(\epsilon\) is arbitrary, (5.3) holds uniformly for \(\theta\in[0,b]\).

If \(u\) is uniformly continuous on \([0,\infty)\), choose \(\delta\) so that
\(|u(x)-u(y)|<\epsilon\) whenever \(|x-y|<\delta\). This \(\delta\) serves
for every \(v\), that is, for every \(t\geq0\), and the same bound gives
convergence uniform in \(t\geq0\) and \(0\leq\theta\leq b\).

For the negative case let \(u\) be bounded and continuous on the line,
\(\theta<0\) and \(h<0\). Put \(a_j=u(t+jh)\) and \(x=\theta/h>0\). Then
\(\Delta^ra_0=h^r\Delta_h^ru(t)\) with the quotient taken at step \(h\), and
(5.1) gives
\[
 \sum_{r=0}^\infty\frac{\theta^r}{r!}\Delta_h^ru(t)
   =\mathbb E\,u(t+hN_{\theta/h}).
\]
The variable \(Y=hN_{\theta/h}\) has mean \(\theta\) and variance
\(h^2\theta/h=h\theta>0\), which tends to zero as \(h\to0-\). The preceding
estimate, with uniform continuity of \(u\) on \([t+\theta-1,t+\theta+1]\),
gives \(\mathbb E\,u(t+Y)\to u(t+\theta)\).

For the first example, \(u(x)=e^{-x}\) gives
\(\Delta_hu(x)=e^{-x}(e^{-h}-1)/h=c_hu(x)\), hence \(\Delta_h^ru=c_h^ru\) by
induction, and the series equals
\(e^{-t}\sum_r(\theta c_h)^r/r!=e^{-t}e^{\theta c_h}\). For \(h>0\),
\(e^{-h}>1-h\), so \(c_h>-1\) and \(e^{-t}e^{\theta c_h}>e^{-t-\theta}\) when
\(\theta>0\). As \(h\to0+\), \(c_h\to-1\), the derivative of \(e^{-x}\) at
\(0\), so the value tends to \(e^{-t-\theta}\).
For the second example, \(u(x)=\min(x,1)\) is bounded and continuous on
\([0,\infty)\). The identity above at \(t=0\) gives the expectation
\(\mathbb E\min(hN_{\theta/h},1)\), and the theorem gives the limit
\(\min(\theta,1)\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-exponential-formula}{thm:p15-w02b-p15-012-exponential-formula}{The exponential formula for translations and its operator-norm boundary}
\begin{proof}
For \(u\in C_b\), \(\|T(h)u\|\leq\|u\|\), so \(\|\Delta_hu\|\leq2\|u\|/h\)
and by induction \(\|\Delta_h^ru\|\leq(2/h)^r\|u\|\). The series
\(\sum_r\theta^r\Delta_h^ru/r!\) therefore converges uniformly on
\([0,\infty)\), with sum of norms at most \(e^{2\theta/h}\|u\|\), and each
partial sum is continuous, so the sum lies in \(C_b\). Similarly the
weights of (5.7) sum to \(1\), so \(A_h(\theta)u\) is a uniform limit of
continuous functions, lies in \(C_b\), and has norm at most \(\|u\|\).

Fix \(t\geq0\) and put \(a_j=u(t+jh)\), \(x=\theta/h\). Then
\(\Delta^ra_0=h^r\Delta_h^ru(t)\), and (5.1) reads
\[
 \sum_{r=0}^\infty\frac{\theta^r}{r!}\Delta_h^ru(t)
   =e^{-\theta/h}\sum_{k=0}^\infty\frac1{k!}\Bigl(\frac\theta h\Bigr)^k u(t+kh)
   =A_h(\theta)u(t).
\]
As this holds for every \(u\) and \(t\), the two operators coincide. The
convergence (5.6) at each \(t\) is
\cref{thm:p15-w02b-p15-012-generalized-taylor}, and when \(u\) is uniformly
continuous that theorem gives convergence uniform in \(t\), which is
convergence in the norm of \(C_b\).

For the operator norm, \(\|A_h(\theta)u-T(\theta)u\|\leq2\|u\|\) gives
\(\|A_h(\theta)-T(\theta)\|\leq2\). Let \(\theta>0\) and let \(h>0\) with
\(\theta/h\) not an integer. The grid points \(kh\), \(k\geq0\), then avoid
\(\theta\), and the nearest of them lies at the positive distance
\(d=\min\{\theta-\lfloor\theta/h\rfloor h,\ (\lfloor\theta/h\rfloor+1)h-\theta\}\).
Define \(u(x)=-1+2\max\{0,1-|x-\theta|/d\}\). It is continuous with
\(\|u\|=1\), \(u(\theta)=1\), and \(u(kh)=-1\) for every \(k\geq0\). At
\(t=0\),
\[
 A_h(\theta)u(0)-T(\theta)u(0)=\sum_{k\geq0}e^{-\theta/h}\frac{(\theta/h)^k}{k!}(-1)-1=-2 .
\]
Thus \(\|A_h(\theta)-T(\theta)\|\geq2\), and equality holds. Since such \(h\)
exist in every interval \((0,h_0)\), \(A_h(\theta)\) does not converge to
\(T(\theta)\) in operator norm as \(h\to0+\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-derivatives}{thm:p15-w02b-p15-012-laplace-derivatives}{Differentiation of the Laplace transform and of the Laplace integral}
\begin{proof}
For \(m\geq1\) and \(a>0\), the function \(y\mapsto y^me^{-ay}\) on
\([0,\infty)\) has its maximum at \(y=m/a\), so
\(y^me^{-ay}\leq(m/(ae))^m\).

Fix \(\lambda>0\) and \(\lambda_1\in(0,\lambda)\). Assume (6.3) for some
\(k\geq0\); the case \(k=0\) is the definition. For real \(\eta\neq0\) with
\(|\eta|<\lambda-\lambda_1\), the mean value theorem gives a point \(\xi\)
between \(\lambda\) and \(\lambda+\eta\), depending on \(\theta\), with
\[
 \frac{e^{-(\lambda+\eta)\theta}-e^{-\lambda\theta}}\eta\,\theta^k
   =-\theta^{k+1}e^{-\xi\theta}.
\]
Since \(\xi\geq\lambda_1\), the absolute value is at most
\(\theta^{k+1}e^{-\lambda_1\theta}\leq((k+1)/(\lambda_1e))^{k+1}\), a
constant, which is \(F\)-integrable. As \(\eta\to0\) along any sequence the
quotient tends to \(-\theta^{k+1}e^{-\lambda\theta}\). Dominated convergence
shows that \((-1)^k\varphi^{(k)}\) is differentiable at \(\lambda\) with
derivative \(-\int e^{-\lambda\theta}\theta^{k+1}F\{d\theta\}\), which is
(6.3) for \(k+1\).

For \(w\) the same quotient argument applies with \(F\{d\theta\}\) replaced
by \(u(x)\,dx\) and the dominating function
\(\|u\|x^{k+1}e^{-\lambda_1x}\). This function is Lebesgue integrable on
\([0,\infty)\), since \(x^{k+1}e^{-\lambda_1x/2}\leq(2(k+1)/(\lambda_1e))^{k+1}\)
and \(e^{-\lambda_1x/2}\) is integrable. The base case is
\(w^{(0)}=w\), and the induction gives
\(w^{(k)}(\lambda)=\int_0^\infty(-x)^ke^{-\lambda x}u(x)\,dx\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-inversion}{thm:p15-w02b-p15-012-laplace-inversion}{The Poisson inversion of Laplace transforms}
\begin{proof}
First (6.1). The equality of the two left expressions is the definition of
the Poisson law. If \(x<0\) and \(\theta\geq0\), then \(\theta>x\) and
\(\mathbb P\{N_{\lambda\theta}\leq\lambda x\}=0\). If \(\theta=0<x\), then
\(N_0=0\leq\lambda x\) and the probability is \(1\). Otherwise
\(\theta>0\), and Chebyshev's inequality with mean and variance
\(\lambda\theta\) gives, for \(\theta>x\),
\[
 \mathbb P\{N_{\lambda\theta}\leq\lambda x\}
   \leq\mathbb P\{|N_{\lambda\theta}-\lambda\theta|\geq\lambda(\theta-x)\}
   \leq\frac{\theta}{\lambda(\theta-x)^2}\longrightarrow0,
\]
and for \(\theta<x\) the same bound with \(x-\theta\) controls
\(\mathbb P\{N_{\lambda\theta}>\lambda x\}\), so the probability tends to
\(1\).

Now (6.4). Let \(\lambda>0\). By (6.3) and linearity of the integral over the
finite set \(k\leq\lambda x\),
\[
 \sum_{k\leq\lambda x}\frac{(-1)^k}{k!}\lambda^k\varphi^{(k)}(\lambda)
   =\int_0^\infty e^{-\lambda\theta}\sum_{k\leq\lambda x}\frac{(\lambda\theta)^k}{k!}\,F\{d\theta\}
   =\int_0^\infty\mathbb P\{N_{\lambda\theta}\leq\lambda x\}\,F\{d\theta\}.
\]
If \(x<0\) both sides are zero, as is \(F(x)\). Let \(x\geq0\) be a continuity
point of \(F\), so \(F\{\{x\}\}=0\). The integrand lies in \([0,1]\) and, by
(6.1), tends to \(1\) for \(\theta<x\) and to \(0\) for \(\theta>x\), that
is, for \(F\)-almost every \(\theta\). Dominated convergence gives the limit
\(F\{[0,x)\}=F\{[0,x]\}=F(x)\).

For uniqueness, let \(F\) and \(G\) on \([0,\infty)\) have Laplace transforms
that agree for \(\lambda>\lambda_1\). Their derivatives of every order then
agree there, so the left sides of (6.4) agree for large \(\lambda\), and
\(F(x)=G(x)\) at every common continuity point. A monotone function has at
most countably many discontinuities, so common continuity points are dense.
Both functions are right continuous, so \(F(x)=G(x)\) for every \(x\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-integral-inversion}{thm:p15-w02b-p15-012-laplace-integral-inversion}{Inversion of Laplace integrals by the gamma kernel}
\begin{proof}
By \cref{thm:p15-w02b-p15-012-laplace-derivatives} at \(\lambda=n/\theta>0\),
\[
 \frac{(-1)^{n-1}}{(n-1)!}\Bigl(\frac n\theta\Bigr)^nw^{(n-1)}(n/\theta)
   =\frac{1}{(n-1)!}\Bigl(\frac n\theta\Bigr)^n\int_0^\infty x^{n-1}e^{-nx/\theta}u(x)\,dx,
\]
because \((-1)^{n-1}(-x)^{n-1}=x^{n-1}\). Writing
\((n/\theta)^nx^{n-1}=(nx/\theta)^{n-1}(n/\theta)\) gives the middle expression
of (6.6). It is \(\int u\,d\nu_{n,\theta}\) for the gamma law
\(\nu_{n,\theta}\) of shape \(n\) and rate \(n/\theta\), normalized by
\(\Gamma(n)=(n-1)!\). By \cref{thm:p15-feller-kernel-examples}, with
\(\nu_{n,0}=\delta_0\), this expectation converges to \(u(\theta)\) uniformly
on every compact interval \([0,b]\). Restricting to \(0<\theta\leq b\), where
the left side of (6.6) is defined, gives the assertion.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-moment-inequalities}{thm:p15-w02b-p15-012-laplace-moment-inequalities}{The alternating inequalities and the moment series}
\begin{proof}
For \(m\geq0\) put \(R_m(t)=e^{-t}-\sum_{k=0}^m(-t)^k/k!\), and put
\(R_{-1}(t)=e^{-t}\). Then \(R_m(0)=0\) for \(m\geq0\) and
\[
 R_m'(t)=-e^{-t}-\sum_{k=1}^m\frac{(-1)^kt^{k-1}}{(k-1)!}
       =-e^{-t}+\sum_{j=0}^{m-1}\frac{(-t)^j}{j!}=-R_{m-1}(t).
\]
We show \((-1)^{m+1}R_m(t)>0\) for \(t>0\) and \(m\geq-1\). For \(m=-1\) it
is \(e^{-t}>0\). If it holds for \(m-1\), then for \(t>0\)
\[
 (-1)^{m+1}R_m(t)=\int_0^t(-1)^{m+1}R_m'(s)\,ds
   =\int_0^t(-1)^mR_{m-1}(s)\,ds>0,
\]
because the integrand is continuous and positive on \((0,t]\). With
\(m=2n-1\) this says \(e^{-t}>\sum_{k=0}^{2n-1}(-t)^k/k!\), and with
\(m=2n\) it says \(e^{-t}<\sum_{k=0}^{2n}(-t)^k/k!\). This is (6.8).

For \(\lambda\geq0\) and \(\theta\geq0\), put \(t=\lambda\theta\). At
\(t=0\) all three members of (6.8) equal \(1\), so the non-strict form holds
for every \(t\geq0\). Integrate it against \(F\{d\theta\}\). The polynomial
members are integrable because \(\mu_1,\ldots,\mu_{2n}\) are finite, and
\(\int(\lambda\theta)^kF\{d\theta\}=\mu_k\lambda^k\). This is (6.7).

Suppose \(\sum_k\mu_k\lambda_0^k/k!<\infty\). For \(0\leq\lambda\leq\lambda_0\)
the terms \(\mu_k\lambda^k/k!\) are dominated by \(\mu_k\lambda_0^k/k!\), so
the alternating series in (6.9) converges absolutely, say to \(S\), and its
terms tend to zero. Let \(S_m\) be its partial sums. By (6.7),
\(S_{2n-1}\leq\varphi(\lambda)\leq S_{2n}\) for all \(n\geq1\), and both
bounds tend to \(S\). Hence \(\varphi(\lambda)=S\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-moment-determinacy}{thm:p15-w02b-p15-012-laplace-moment-determinacy}{Real-variable propagation of the Laplace transform}
\begin{proof}
Let \(\varphi\) and \(\psi\) be the transforms of \(F\) and \(G\), and let
\(K=\sum_k\mu_k\lambda_0^k/k!<\infty\). For \(a\geq0\) put
\(\varphi^{[k]}(a)=(-1)^k\int_0^\infty e^{-a\theta}\theta^kF\{d\theta\}\),
and define \(\psi^{[k]}(a)\) from \(G\) in the same way. By
\cref{thm:p15-w02b-p15-012-laplace-derivatives},
\(\varphi^{[k]}(a)=\varphi^{(k)}(a)\) when \(a>0\).

\emph{Local expansion.} Let \(a\geq0\) and \(|\eta|\leq\lambda_0\) with
\(a+\eta\geq0\). Then
\(e^{-(a+\eta)\theta}=\sum_ke^{-a\theta}(-\eta\theta)^k/k!\) for every
\(\theta\geq0\), and
\[
 \sum_k\int_0^\infty e^{-a\theta}\frac{|\eta|^k\theta^k}{k!}\,F\{d\theta\}
    \leq\sum_k\frac{\mu_k\lambda_0^k}{k!}=K<\infty .
\]
By Tonelli's theorem applied to the absolute values, and then dominated
convergence for the partial sums, the series may be integrated term by term:
\[
 \varphi(a+\eta)=\sum_{k=0}^\infty\frac{\eta^k}{k!}\varphi^{[k]}(a),
\]
and likewise for \(\psi\), since \(G\) has the same moments.

\emph{Propagation.} At \(a=0\),
\(\varphi^{[k]}(0)=(-1)^k\mu_k=\psi^{[k]}(0)\), so the local expansion gives
\(\varphi=\psi\) on \([0,\lambda_0]\). Suppose \(\varphi=\psi\) on
\([0,b]\) for some \(b\geq\lambda_0\), and put \(a=b-\lambda_0/2\). Then
\(a\geq\lambda_0/2>0\) is an interior point of \([0,b]\). The derivatives of
every order at \(a\) are determined by the values on a neighborhood of \(a\)
inside \([0,b]\), where the two functions agree, so
\(\varphi^{[k]}(a)=\varphi^{(k)}(a)=\psi^{(k)}(a)=\psi^{[k]}(a)\) for all
\(k\). The local expansion with \(0\leq\eta\leq\lambda_0\) then gives
\(\varphi=\psi\) on \([a,a+\lambda_0]=[b-\lambda_0/2,b+\lambda_0/2]\), hence on
\([0,b+\lambda_0/2]\). By induction \(\varphi=\psi\) on
\([0,\lambda_0+j\lambda_0/2]\) for every \(j\), that is, on \([0,\infty)\).

\emph{Identification.} The two transforms agree for every \(\lambda>0\), so
\cref{thm:p15-w02b-p15-012-laplace-inversion} gives \(F=G\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-examples}{thm:p15-w02b-p15-012-laplace-examples}{The exponential law, an atom and the lognormal moments}
\begin{proof}
(1) For \(\lambda\geq0\),
\(\varphi(\lambda)=\int_0^\infty e^{-\lambda\theta}e^{-\theta}\,d\theta=1/(1+\lambda)\).
Integration by parts gives
\(\int_0^\infty\theta^ke^{-\theta}\,d\theta=k\int_0^\infty\theta^{k-1}e^{-\theta}\,d\theta\),
so \(\mu_k=k!\) by induction. The series (6.9) is
\(\sum_k(-\lambda)^k=1/(1+\lambda)\) for \(0\leq\lambda<1\), and
\(\sum_k\mu_k\lambda_0^k/k!=\sum_k\lambda_0^k<\infty\) for
\(0<\lambda_0<1\), so \cref{thm:p15-w02b-p15-012-laplace-moment-determinacy}
applies. Differentiating gives
\(\varphi^{(k)}(\lambda)=(-1)^kk!(1+\lambda)^{-k-1}\), so
\((-1)^k\lambda^k\varphi^{(k)}(\lambda)/k!=q^k/(1+\lambda)\) with
\(q=\lambda/(1+\lambda)\). For \(x\geq0\) let \(K=\lfloor\lambda x\rfloor\).
Since \(1-q=1/(1+\lambda)\),
\[
 \sum_{k=0}^K\frac{q^k}{1+\lambda}=(1-q)\frac{1-q^{K+1}}{1-q}=1-q^{K+1}.
\]
Now \(q^{K+1}=\exp\{-(K+1)\log(1+1/\lambda)\}\). Here
\(\lambda x<K+1\leq\lambda x+1\), and
\(1/\lambda-1/(2\lambda^2)\leq\log(1+1/\lambda)\leq1/\lambda\) for
\(\lambda>0\), so \((K+1)\log(1+1/\lambda)\to x\) and the left side of (6.4)
tends to \(1-e^{-x}=F(x)\).

(2) For \(F=\delta_{x_0}\), \(\varphi(\lambda)=e^{-\lambda x_0}\) and
\((-1)^k\lambda^k\varphi^{(k)}(\lambda)/k!=e^{-\lambda x_0}(\lambda x_0)^k/k!\),
so the left side of (6.4) at \(x=x_0\) is
\(\mathbb P\{N_{\lambda x_0}\leq\lambda x_0\}\). At \(\lambda_m=m/x_0\) this
is \(\mathbb P\{N_m\leq m\}\). For independent \(N_a\) and \(N_b\),
\[
 \mathbb P\{N_a+N_b=k\}=\sum_{j=0}^ke^{-a}\frac{a^j}{j!}e^{-b}\frac{b^{k-j}}{(k-j)!}
   =e^{-(a+b)}\frac{(a+b)^k}{k!}
\]
by the binomial theorem, so \(N_m\) has the law of a sum \(S_m\) of \(m\)
independent Poisson variables of mean and variance \(1\). By
\cref{thm:p15-clt-iid}, \((S_m-m)/\sqrt m\) converges in distribution to a
standard normal variable \(\chi\). The point \(0\) is a continuity point of
the law of \(\chi\), so
\(\mathbb P\{N_m\leq m\}=\mathbb P\{(S_m-m)/\sqrt m\leq0\}\to
\mathbb P\{\chi\leq0\}=1/2\), whereas \(F(x_0)=1\).

(3) By \cref{thm:p15-feller-lognormal-moments}, the densities \(f_a\),
\(-1\leq a\leq1\), vanish on \((-\infty,0]\), are distinct, and all have the
moments \(\mu_k=e^{k^2/2}\). For \(\lambda>0\), \(k!\leq k^k\) gives
\[
 \frac{\mu_k\lambda^k}{k!}\geq\exp\Bigl\{k\Bigl(\frac k2+\log\lambda-\log k\Bigr)\Bigr\},
\]
and \(k/2-\log k\to\infty\), so the terms are unbounded and the series
diverges.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-exponential-remainder}{thm:p15-w02b-p15-012-exponential-remainder}{Feller's remainder estimate for the exponential}
\begin{proof}
The function \(x\mapsto e^{ix}\) has derivative \(ie^{ix}\), so
\(\rho_1(t)=e^{it}-1=i\int_0^te^{ix}\,dx\). For \(n>1\),
\[
 \rho_n'(t)=ie^{it}-\sum_{k=1}^{n-1}\frac{i(it)^{k-1}}{(k-1)!}
   =i\Bigl(e^{it}-\sum_{j=0}^{n-2}\frac{(it)^j}{j!}\Bigr)=i\rho_{n-1}(t),
\]
and \(\rho_n(0)=0\), so \(\rho_n(t)=i\int_0^t\rho_{n-1}(x)\,dx\).
For \(t>0\), \(|\rho_1(t)|\leq\int_0^t1\,dx=t\). If
\(|\rho_{n-1}(x)|\leq x^{n-1}/(n-1)!\) for \(x>0\), then
\(|\rho_n(t)|\leq\int_0^tx^{n-1}/(n-1)!\,dx=t^n/n!\). This is (4.1) for
\(t>0\). At \(t=0\) both sides vanish. The complex conjugate of
\(\rho_n(t)\) is \(e^{-it}-\sum_{k<n}(-it)^k/k!=\rho_n(-t)\), so
\(|\rho_n(-t)|=|\rho_n(t)|\leq|t|^n/n!\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-xv4-derivatives}{thm:p15-w02b-p15-012-xv4-derivatives}{Feller's derivative formula by dominated convergence}
\begin{proof}
For \(0\leq j\leq n\), \(|x|^j\leq1+|x|^n\), so \(M_j\leq1+M_n<\infty\).
The case \(n=0\) is the definition. Suppose that for some \(k<n\),
\(\varphi^{(k)}(\zeta)=i^k\int e^{i\zeta x}x^kF\{dx\}\) for every \(\zeta\).
For real \(h\neq0\),
\[
 \frac{\varphi^{(k)}(\zeta+h)-\varphi^{(k)}(\zeta)}h
   =i^k\int e^{i\zeta x}x^k\frac{e^{ihx}-1}h\,F\{dx\}.
\]
By \cref{thm:p15-w02b-p15-012-exponential-remainder} with \(n=1\),
\(|e^{ihx}-1|\leq|hx|\), so the integrand is dominated by \(|x|^{k+1}\),
which is integrable since \(M_{k+1}<\infty\). As \(h\to0\) along any sequence,
\((e^{ihx}-1)/h\to ix\). Dominated convergence gives
\(\varphi^{(k+1)}(\zeta)=i^{k+1}\int e^{i\zeta x}x^{k+1}F\{dx\}\). This proves
(4.5) by induction. If \(\zeta_j\to\zeta\), the integrands
\(e^{i\zeta_jx}x^n\) converge pointwise and are dominated by \(|x|^n\), so
\(\varphi^{(n)}\) is continuous. At \(\zeta=0\) the formula gives
\(i^nm_n\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-taylor-bound}{thm:p15-w02b-p15-012-taylor-bound}{The Taylor bound for characteristic functions}
\begin{proof}
Since \(|e^{i\zeta x}|=1\), the left side of (4.13) equals
\(|\rho_n(tx)|\), which is at most \(|tx|^n/n!\) by (4.1).

Let \(M_n<\infty\). By \cref{thm:p15-w02b-p15-012-xv4-derivatives},
\(\varphi^{(k)}(\zeta)=\int e^{i\zeta x}(ix)^kF\{dx\}\) for \(k\leq n-1\), so
\[
 \varphi(\zeta+t)-\sum_{k=0}^{n-1}\frac{t^k}{k!}\varphi^{(k)}(\zeta)
   =\int e^{i\zeta x}\Bigl(e^{itx}-\sum_{k=0}^{n-1}\frac{(itx)^k}{k!}\Bigr)F\{dx\}.
\]
By (4.13) the modulus of the integrand is at most \(|tx|^n/n!\), and
integration gives the bound \(M_n|t|^n/n!\). For \(F=\delta_0\) and
\(n\geq1\) both sides of (4.14) are zero, so the inequality cannot be strict
in general.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-xv4-determinacy}{thm:p15-w02b-p15-012-xv4-determinacy}{Feller's analytic determinacy proof, with the propagation written out}
\begin{proof}
\emph{A factorial bound.} For \(n\geq1\),
\(e^n=\sum_jn^j/j!\geq n^n/n!\), so \(n!\geq(n/e)^n\).

\emph{A uniform moment bound.} Let \(\Lambda>\lambda\) with \(\Lambda>0\).
By (4.15) there is \(n_0\) with \(M_n^{1/n}\leq\Lambda n\) for \(n\geq n_0\).
Put \(C=\max\{1,M_n/(\Lambda n)^n:1\leq n<n_0\}\). Then
\(M_n\leq C(\Lambda n)^n\) for every \(n\geq1\).

\emph{The Taylor series of \(\varphi\).} Let \(\zeta,t\) be real with
\(e\lambda|t|<1\); for \(t=0\) there is nothing to prove. Choose
\(\Lambda\) with \(\max(\lambda,0)<\Lambda<1/(e|t|)\). By (4.14) and the two
bounds,
\[
 \Bigl|\varphi(\zeta+t)-\sum_{k=0}^{n-1}\frac{t^k}{k!}\varphi^{(k)}(\zeta)\Bigr|
   \leq C(\Lambda n)^n\frac{e^n|t|^n}{n^n}=C(e\Lambda|t|)^n\longrightarrow0 .
\]
This is the series assertion. When \(\lambda=0\) every \(t\) is admissible.
If \(|t|<1/(3\lambda)\) then \(e\lambda|t|<3\lambda|t|<1\), which gives
Feller's radius.

\emph{A competing law satisfies the same bound.} Let \(G\) have the same
moments \(m_n\), with absolute moments \(M_n^G\). Even absolute moments are
moments, so \(M_{2k}^G=m_{2k}=M_{2k}\). By the Cauchy--Schwarz inequality
applied to \(|x|^k\cdot|x|^{k+1}\),
\[
 M_{2k+1}^G\leq(M_{2k}M_{2k+2})^{1/2}
   \leq C\bigl(\Lambda(2k+2)\bigr)^{(2k+2k+2)/2}=C\bigl(\Lambda(2k+2)\bigr)^{2k+1},
\]
using \((\Lambda\cdot2k)^{2k}\leq(\Lambda(2k+2))^{2k}\), with the convention
\(0^0=1\) at \(k=0\). Hence \(M_n^G\leq C(\Lambda(n+1))^n\) for every
\(n\geq1\), and
\[
 M_n^G\frac{|t|^n}{n!}\leq C(e\Lambda|t|)^n\Bigl(1+\frac1n\Bigr)^n
   \leq Ce\,(e\Lambda|t|)^n .
\]

\emph{Propagation along the real axis.} Fix \(\Lambda>\max(\lambda,0)\) and
\(t_0>0\) with \(e\Lambda t_0<1\). Let \(\varphi\) and \(\psi\) be the
characteristic functions of \(F\) and \(G\), and \(D=\varphi-\psi\). By
(4.14) for each law and the two bounds, for every real \(\zeta\) and
\(|t|\leq t_0\),
\[
 D(\zeta+t)=\sum_{k=0}^\infty\frac{t^k}{k!}D^{(k)}(\zeta).
\]
Let \(S\) be the set of real \(\zeta\) at which \(D^{(k)}(\zeta)=0\) for every
\(k\geq0\). Each \(D^{(k)}\) is continuous by
\cref{thm:p15-w02b-p15-012-xv4-derivatives}, so \(S\) is closed. By (4.5),
\(\varphi^{(k)}(0)=i^km_k=\psi^{(k)}(0)\), so \(0\in S\). If \(\zeta\in S\),
the expansion gives \(D=0\) on \([\zeta-t_0,\zeta+t_0]\); all derivatives of
\(D\) vanish on the open interval \((\zeta-t_0,\zeta+t_0)\), and by closedness
\([\zeta-t_0,\zeta+t_0]\subseteq S\). Starting from \(0\) and applying this to
the endpoints \(\pm jt_0\), induction gives \([-jt_0,jt_0]\subseteq S\) for all
\(j\). Hence \(D\equiv0\), so \(\varphi=\psi\) on the line, and
\cref{thm:p15-characteristic-inversion-uniqueness} gives \(F=G\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-xv4-bridge}{thm:p15-w02b-p15-012-xv4-bridge}{From the moment power series to (4.15) and to Carleman's condition}
\begin{proof}
Let \(C'=\sum_nm_{2n}s^{2n}/(2n)!\). The term \(n=0\) is \(m_0=1\), so
\(1\leq C'<\infty\), and every term is at most \(C'\):
\[
 M_{2n}=m_{2n}\leq C'(2n)!\,s^{-2n}\leq C'(2n)^{2n}s^{-2n}.
\]
Hence \(M_{2n}^{1/(2n)}/(2n)\leq C'^{1/(2n)}/s\to1/s\). For odd orders,
Cauchy--Schwarz gives
\[
 M_{2n+1}\leq(M_{2n}M_{2n+2})^{1/2}\leq C'(2n+2)^{2n+1}s^{-(2n+1)},
\]
so \(M_{2n+1}^{1/(2n+1)}/(2n+1)\leq C'^{1/(2n+1)}(2n+2)/((2n+1)s)\to1/s\).
Thus the upper limit in (4.15) is at most \(1/s\), and
\cref{thm:p15-w02b-p15-012-xv4-determinacy} gives determinacy.
Putting \(t=s^2\) turns the hypothesis into convergence of
\(\sum_nm_{2n}t^n/(2n)!\), and conversely.
For a law on \([0,\infty)\), the terms \(\mu_{2n}\lambda_0^{2n}/(2n)!\) form a
subseries of the convergent nonnegative series \(\sum_k\mu_k\lambda_0^k/k!\),
so the hypothesis holds with \(s=\lambda_0\).

Assume (4.15) and let \(\Lambda>\max(\lambda,0)\). For \(n\geq n_0\),
\(M_n^{1/n}\leq\Lambda n\), so \(M_n^{-1/n}\geq1/(\Lambda n)\), the value
\(+\infty\) being allowed when \(M_n=0\). The harmonic series diverges, so
\(\sum_nM_n^{-1/n}=\infty\). The even-order subseries
\(\sum_km_{2k}^{-1/(2k)}\geq\sum_{k\geq n_0}1/(2\Lambda k)\) also diverges,
which is the form used in \cref{thm:p15-criteria-carleman}.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-xv4-examples}{thm:p15-w02b-p15-012-xv4-examples}{Normal, exponential and lognormal growth}
\begin{proof}
(1) By \cref{thm:p15-criteria-moment-identities},
\(M_{2k}=(2k-1)!!\), a product of \(k\) odd numbers each at most \(2k\), so
\(M_{2k}\leq(2k)^k\) and \(M_{2k}^{1/(2k)}/(2k)\leq(2k)^{-1/2}\to0\). By
Cauchy--Schwarz,
\(M_{2k+1}\leq((2k)^k(2k+2)^{k+1})^{1/2}\leq(2k+2)^{(2k+1)/2}\), so
\(M_{2k+1}^{1/(2k+1)}/(2k+1)\leq(2k+2)^{1/2}/(2k+1)\to0\). Thus
\(\lambda=0\), and \cref{thm:p15-w02b-p15-012-xv4-determinacy} gives the
expansion for every real increment.

(2) The exponential law is concentrated on \([0,\infty)\), so
\(M_n=\mu_n=n!\) by the computation in
\cref{thm:p15-w02b-p15-012-laplace-examples}. Since \(n!\leq n^n\),
\(M_n^{1/n}/n\leq1\), and the upper limit is at most \(1\).

(3) The lognormal laws are concentrated on \((0,\infty)\), so
\(M_n=e^{n^2/2}\) and \(M_n^{1/n}/n=e^{n/2}/n\to\infty\). Condition (4.15)
fails; by \cref{thm:p15-feller-lognormal-moments} these moments are shared by
distinct laws, so no sufficient criterion can hold for them.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-vague-tools}{thm:p15-w02b-p15-012-vague-tools}{Partitions at non-atoms, uniqueness of limits and selection}
\begin{proof}
(1) For integers \(k,m\geq1\), the atoms of \(U\) in \((-k,k]\) of mass greater
than \(1/m\) number at most \(mU\{(-k,k]\}<\infty\). The atoms are the union
over \(k,m\) of these finite sets, so they are countable, and every
nonempty open interval contains non-atoms. Now assume convergence on all
\((r,s]\) with non-atom endpoints, and let \(I\) be a bounded interval of
continuity with endpoints \(r\leq s\). For a non-atom \(c\) and \(\eta>0\)
choose non-atoms \(c'<c<c''\) with \(U\{(c',c'']\}<\eta\); this is possible
because \(U\{(c',c'']\}\downarrow U\{\{c\}\}=0\) as \(c'\uparrow c\) and
\(c''\downarrow c\) through non-atoms, by continuity from above of the finite
measure \(U\) on \((c-1,c+1]\). Then
\(\limsup_nU_n\{\{c\}\}\leq\lim_nU_n\{(c',c'']\}<\eta\), so
\(U_n\{\{c\}\}\to0\). Every bounded interval with endpoints \(r,s\) differs
from \((r,s]\) by at most the two points \(r,s\), so \(U_n\{I\}\to U\{I\}\).
If \(r=s\), then \(I\) is empty or \(\{r\}\), and the same estimate applies.

(2) Let \(\epsilon>0\) and take \(\delta\) from the uniform equicontinuity.
By (1) choose non-atoms \(a=x_0<x_1<\cdots<x_m=b\) with
\(x_j-x_{j-1}\leq\delta\), and put \(I_j=(x_{j-1},x_j]\). For a measure \(V\)
and \(g\in\mathcal G\) let \(S_V(g)=\sum_jg(x_j)V\{I_j\}\). On \(I_j\),
\(|g(x)-g(x_j)|\leq\epsilon\), so
\(|\int_{(a,b]}g\,dV-S_V(g)|\leq\epsilon V\{(a,b]\}\). Also
\(|S_{U_n}(g)-S_U(g)|\leq B\sum_j|U_n\{I_j\}-U\{I_j\}|\), a bound independent
of \(g\) that tends to \(0\) because each \(I_j\) is an interval of continuity.
Since \(U_n\{(a,b]\}\to U\{(a,b]\}\),
\[
 \limsup_n\sup_{g\in\mathcal G}\Bigl|\int_{(a,b]}g\,dU_n-\int_{(a,b]}g\,dU\Bigr|
   \leq2\epsilon\,U\{(a,b]\},
\]
and \(\epsilon\) is arbitrary.

(3) Given \(\epsilon>0\) choose \(T>0\) with \(|g(x)|\leq\epsilon\) for
\(|x|\geq T\) and, by (1), with \(\pm T\) non-atoms of \(U\). The function
\(g\) is uniformly continuous on \([-T,T]\), so (2) with
\(\mathcal G=\{g\}\) gives \(\int_{(-T,T]}g\,dU_n\to\int_{(-T,T]}g\,dU\).
Outside \((-T,T]\) the integrals against \(U_n\) and against \(U\) are at most
\(\epsilon\) in absolute value, because the masses are at most \(1\). Hence
\(\limsup_n|\int g\,dU_n-\int g\,dU|\leq2\epsilon\).

(4) The set \(D\) of points that are atoms of neither \(U\) nor \(V\) has
countable complement. For \(r<s\) in \(D\),
\(U\{(r,s]\}=\lim_nU_n\{(r,s]\}=V\{(r,s]\}\). For arbitrary \(r<s\) choose
\(r_k\downarrow r\) and \(s_k\downarrow s\) in \(D\) with \(r_k<s\). Then
\(\mathbf 1_{(r_k,s_k]}\to\mathbf 1_{(r,s]}\) pointwise, all dominated by
\(\mathbf 1_{(r,s_1]}\), so dominated convergence for the finite measures
\(U\) and \(V\) on \((r,s_1]\) gives \(U\{(r,s]\}=V\{(r,s]\}\). Fix \(k\) and
let \(U_k',V_k'\) be the restrictions to \((-k,k]\). They have the same total
mass \(c=U\{(-k,k]\}\). If \(c=0\) both vanish. Otherwise
\(U_k'/c\) and \(V_k'/c\) are probability laws with the same CDF, namely
\(0\) on \((-\infty,-k]\), \(U\{(-k,x]\}/c\) on \((-k,k]\) and \(1\) beyond, so
they are equal by \cref{thm:p15-weak-quantile-foundations}. Thus
\(U\{A\cap(-k,k]\}=V\{A\cap(-k,k]\}\) for every Borel \(A\), and letting
\(k\to\infty\) gives \(U=V\).

(5) If \(U_n\to U\) fails, there are a bounded interval of continuity \(I\),
an \(\eta>0\) and a subsequence along which \(|U_n\{I\}-U\{I\}|\geq\eta\). By
hypothesis that subsequence has a further subsequence converging to \(U\),
along which \(U_n\{I\}\to U\{I\}\), a contradiction.

(6) Write \(U_n(x)=U_n\{(-\infty,x]\}\). Suppose \(U_n\to U\) and let \(x\) be
a continuity point of \(U(x)\), that is, a non-atom. Given \(\epsilon>0\)
choose \(T>|x|\) with \(\pm T\) non-atoms and \(U\{(-T,T]\}\geq1-\epsilon\).
Then \(U_n(x)\geq U_n\{(-T,x]\}\to U\{(-T,x]\}\geq U(x)-\epsilon\), and
\(1-U_n(x)\geq U_n\{(x,T]\}\to U\{(x,T]\}\geq1-U(x)-\epsilon\). Hence
\(U_n(x)\to U(x)\), which is \(U_n\Rightarrow U\). Conversely, if
\(U_n\Rightarrow U\) and \(r<s\) are non-atoms, then
\(U_n\{(r,s]\}=U_n(s)-U_n(r)\to U\{(r,s]\}\), and (1) gives \(U_n\to U\).

(7) Let \(F_n\) be probability distributions with CDFs \(F_n(x)\). By
\cref{thm:p15-helly-selection} there are a subsequence \(n_k\), a
right-continuous nondecreasing \(F\) with values in \([0,1]\), and a finite
measure \(\nu\) with \(\nu\{(-\infty,x]\}=F(x)-\beta\) and total mass
\(\gamma-\beta\leq1\), such that \(F_{n_k}\{(r,s]\}\to\nu\{(r,s]\}\) for
continuity points \(r<s\) of \(F\). The continuity points of \(F\) are exactly
the non-atoms of \(\nu\), since the jump of \(F\) at \(x\) is \(\nu\{\{x\}\}\).
By (1), \(F_{n_k}\to\nu\), a possibly defective distribution. If every \(F_n\)
is concentrated on \([0,\infty)\), then for non-atoms \(r<s<0\),
\(\nu\{(r,s]\}=\lim_kF_{n_k}\{(r,s]\}=0\); the negative half-line is a
countable union of such intervals, so \(\nu\) is concentrated on
\([0,\infty)\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-uniqueness}{thm:p15-w02b-p15-012-laplace-uniqueness}{Uniqueness by the Poisson inversion formula and the translation principle}
\begin{proof}
(1) Let \(F\) be a finite measure concentrated on \([0,\infty)\) with mass
\(c\) and transform \(\varphi\), and let \(x\geq0\) be a non-atom. We claim
\[
 \sum_{k\leq\lambda x}\frac{(-1)^k}{k!}\lambda^k\varphi^{(k)}(\lambda)\longrightarrow F\{[0,x]\}
 \qquad(\lambda\to\infty).
\]
If \(c=0\) both sides are \(0\). If \(c>0\), \(F/c\) is a probability
distribution with transform \(\varphi/c\), and the claim is \(c\) times (6.4)
of \cref{thm:p15-w02b-p15-012-laplace-inversion}. The left side uses only the
values of \(\varphi\) on a half-line \(\lambda>\lambda_1\), since derivatives
at \(\lambda\) depend only on values near \(\lambda\). Hence two such measures
\(F,G\) whose transforms agree for \(\lambda>\lambda_1\) satisfy
\(F\{[0,x]\}=G\{[0,x]\}\) for every \(x\geq0\) that is a non-atom of both;
these \(x\) are dense in \([0,\infty)\) by part~(1) of
\cref{thm:p15-w02b-p15-012-vague-tools}. Both sides are right-continuous in
\(x\), so they agree for all \(x\geq0\), and both measures vanish on
\((-\infty,0)\). Then \(F\{(r,s]\}=G\{(r,s]\}\) for all \(r<s\), and the
argument of part~(4) of \cref{thm:p15-w02b-p15-012-vague-tools} gives \(F=G\).

(2) Let \(U\) and \(V\) be measures concentrated on \([0,\infty)\) whose
transforms converge and agree on \(a<\lambda<\infty\). Fix \(a'>a\). The
measures \(U^\#\{dx\}=e^{-a'x}U\{dx\}\) and \(V^\#\{dx\}=e^{-a'x}V\{dx\}\) are
finite, with masses \(\omega(a')\), and their transforms at \(\lambda>0\)
equal \(\int e^{-\lambda x}e^{-a'x}U\{dx\}=\omega(\lambda+a')\) and the same
expression for \(V\). This is the translation principle (1.9). By (1),
\(U^\#=V^\#\). Since \(U\{dx\}=e^{a'x}U^\#\{dx\}\) and likewise for \(V\),
\(U\{A\}=\int_Ae^{a'x}U^\#\{dx\}=V\{A\}\) for every Borel \(A\).

(3) Let \(u,v\geq0\) be continuous on \([0,\infty)\) with the same ordinary
transform on \((a,\infty)\). By (2) the measures \(u(x)\,dx\) and
\(v(x)\,dx\) coincide, so \(\int_I(u-v)\,dx=0\) for every bounded interval
\(I\subseteq[0,\infty)\). If \(u(x_0)\neq v(x_0)\), say \(u(x_0)>v(x_0)\), then
by continuity \(u-v>0\) on an interval \([x_0,x_0+\eta]\) or, when \(x_0>0\),
on \([x_0-\eta,x_0+\eta]\cap[0,\infty)\), with \(\eta>0\), and the integral
over that interval is positive, a contradiction. Thus \(u=v\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-continuity}{thm:p15-w02b-p15-012-laplace-continuity}{The Laplace continuity theorem through selection}
\begin{proof}
(1) For \(\lambda>0\) put \(g_\lambda(x)=e^{-\lambda|x|}\). It is continuous,
tends to \(0\) as \(|x|\to\infty\), and equals \(e^{-\lambda x}\) on
\([0,\infty)\), where all the measures live. Part~(3) of
\cref{thm:p15-w02b-p15-012-vague-tools} gives
\(\varphi_n(\lambda)=\int g_\lambda\,dF_n\to\int g_\lambda\,dF=\varphi(\lambda)\).

(2) Let \(\varphi_n(\lambda)\to\varphi(\lambda)\) for each \(\lambda>0\). By
part~(7) of \cref{thm:p15-w02b-p15-012-vague-tools}, every subsequence of
\((F_n)\) has a further subsequence converging to a possibly defective
distribution \(F'\) concentrated on \([0,\infty)\). By part~(1) of the present
theorem, applied along that further subsequence, the transform of \(F'\) is
\(\varphi\) on \(\lambda>0\). If \(F''\) is the limit along another such
subsequence, \(F'\) and \(F''\) are finite measures on \([0,\infty)\) with the
same transform for \(\lambda>0\), so \(F'=F''\) by part~(1) of
\cref{thm:p15-w02b-p15-012-laplace-uniqueness}. Call this common limit \(F\).
Every subsequence of \((F_n)\) thus has a further subsequence converging to
\(F\), and part~(5) of \cref{thm:p15-w02b-p15-012-vague-tools} gives
\(F_n\to F\). Its transform is \(\varphi\).

(3) As \(\lambda\downarrow0\), \(e^{-\lambda x}\uparrow1\) for every
\(x\geq0\), so by monotone convergence \(\varphi(\lambda)\uparrow F\{[0,\infty)\}\).
Thus \(F\) is proper if and only if \(\varphi(\lambda)\to1\). When \(F\) is
proper, \(F_n\Rightarrow F\) by part~(6) of
\cref{thm:p15-w02b-p15-012-vague-tools}.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-extended-continuity}{thm:p15-w02b-p15-012-laplace-extended-continuity}{The extended continuity theorem, including a vanishing limit}
\begin{proof}
All measures are concentrated on \([0,\infty)\), so for \(x>0\) a non-atom,
integrals over \([0,x]\) equal integrals over \((-1,x]\), and \(-1\) is a
non-atom of every measure considered.

(1) Assume \(\omega_n(\lambda)\to\omega(\lambda)<\infty\) for every
\(\lambda>a\).

\emph{Case A: \(\omega(\lambda_0)=0\) for some \(\lambda_0>a\).} For \(x>0\),
\[
 U_n\{[0,x]\}\leq\int_{[0,x]}e^{\lambda_0(x-y)}\,U_n\{dy\}
   \leq e^{\lambda_0x}\omega_n(\lambda_0)\longrightarrow0 .
\]
For \(\lambda\geq\lambda_0\),
\(0\leq\omega_n(\lambda)\leq\omega_n(\lambda_0)\to0\). For
\(a<\lambda<\lambda_0\) choose \(\lambda_1\in(a,\lambda)\); then for \(x>0\)
\[
 \begin{aligned}
 \omega_n(\lambda)
   &\leq U_n\{[0,x]\}+\int_{(x,\infty)}e^{-(\lambda-\lambda_1)y}e^{-\lambda_1y}\,U_n\{dy\}\\
   &\leq e^{\lambda_0x}\omega_n(\lambda_0)+e^{-(\lambda-\lambda_1)x}\omega_n(\lambda_1),
 \end{aligned}
\]
so \(\omega(\lambda)\leq e^{-(\lambda-\lambda_1)x}\omega(\lambda_1)\) for
every \(x>0\), and \(\omega(\lambda)=0\). Thus \(\omega\) is the transform of
the zero measure \(U=0\), and \(U_n\{I\}\leq U_n\{[0,x]\}\to0\) for every
bounded interval \(I\subseteq(-\infty,x]\), that is, \(U_n\to0\).

\emph{Case B: \(\omega(\lambda)>0\) for every \(\lambda>a\).} Fix
\(\lambda_0>a\). For large \(n\), \(\omega_n(\lambda_0)>0\), and
\(U_n^\#\{dx\}=e^{-\lambda_0x}U_n\{dx\}/\omega_n(\lambda_0)\) is a probability
distribution on \([0,\infty)\) with transform
\(\omega_n(\lambda+\lambda_0)/\omega_n(\lambda_0)\to\omega(\lambda+\lambda_0)/\omega(\lambda_0)\)
for \(\lambda>0\). By part~(2) of
\cref{thm:p15-w02b-p15-012-laplace-continuity}, \(U_n^\#\to U^\#\), a possibly
defective distribution on \([0,\infty)\) with transform
\(\omega(\lambda+\lambda_0)/\omega(\lambda_0)\). Define
\(U\{dx\}=\omega(\lambda_0)e^{\lambda_0x}U^\#\{dx\}\). It is concentrated on
\([0,\infty)\), finite on bounded intervals, and has the same atoms as
\(U^\#\). For \(\lambda>\lambda_0\),
\[
 \int_0^\infty e^{-\lambda x}\,U\{dx\}
   =\omega(\lambda_0)\int_0^\infty e^{-(\lambda-\lambda_0)x}\,U^\#\{dx\}
   =\omega(\lambda).
\]
For non-atoms \(r<s\) of \(U\), part~(2) of
\cref{thm:p15-w02b-p15-012-vague-tools} with \(g(x)=e^{\lambda_0x}\) on
\([r,s]\) gives \(\int_{(r,s]}e^{\lambda_0x}\,U_n^\#\{dx\}\to\int_{(r,s]}e^{\lambda_0x}\,U^\#\{dx\}\);
multiplying by \(\omega_n(\lambda_0)\to\omega(\lambda_0)\) gives
\(U_n\{(r,s]\}\to U\{(r,s]\}\), and part~(1) of that lemma gives
\(U_n\to U\). If \(\lambda_0'>a\) yields the measure \(U'\) in the same way,
then \(U_n\to U\) and \(U_n\to U'\), so \(U=U'\) by part~(4). Hence one measure
\(U\) has transform \(\omega(\lambda)\) for \(\lambda>\lambda_0\), for every
\(\lambda_0>a\), that is, for all \(\lambda>a\).

(2) Assume \(U_n\to U\) and \(\omega_n(a)\leq A\). For non-atoms \(r<s<0\),
\(U\{(r,s]\}=\lim U_n\{(r,s]\}=0\), so \(U\) is concentrated on
\([0,\infty)\). For a non-atom \(t>0\), part~(2) of
\cref{thm:p15-w02b-p15-012-vague-tools} gives
\[
 \int_{[0,t]}e^{-ax}\,U\{dx\}=\lim_n\int_{[0,t]}e^{-ax}\,U_n\{dx\}\leq A ;
\]
letting \(t\to\infty\) through non-atoms, monotone convergence gives
\(\int_0^\infty e^{-ax}\,U\{dx\}\leq A\). Let \(\lambda>a\). For \(x>t\),
\(e^{-\lambda x}\leq e^{-(\lambda-a)t}e^{-ax}\), so the integrals of
\(e^{-\lambda x}\) over \((t,\infty)\) against \(U_n\) and against \(U\) are
at most \(Ae^{-(\lambda-a)t}\). Hence
\[
 \begin{aligned}
 \limsup_n|\omega_n(\lambda)-\omega(\lambda)|
   &\leq\limsup_n\Bigl|\int_{[0,t]}e^{-\lambda x}\,(U_n-U)\{dx\}\Bigr|+2Ae^{-(\lambda-a)t}\\
   &=2Ae^{-(\lambda-a)t},
 \end{aligned}
\]
by part~(2) of the lemma again, and \(t\) may be taken arbitrarily large.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-continuity-examples}{thm:p15-w02b-p15-012-laplace-continuity-examples}{Binomial, escaping and exploding examples}
\begin{proof}
(1) By the binomial theorem,
\(\varphi_n(\lambda)=\sum_k\binom nk p^k(1-p)^{n-k}e^{-\lambda k}=(1-p+pe^{-\lambda})^n\)
with \(p=\mu/n\), which is the stated expression. Put
\(c=\mu(1-e^{-\lambda})\in(0,\mu)\) and \(y=c/n\). For \(n\geq2\mu\),
\(0<y\leq\frac12\), and from \(\log(1-y)=-\sum_{j\geq1}y^j/j\) we get
\(-y/(1-y)\leq\log(1-y)\leq-y\). Hence
\(-c/(1-c/n)\leq n\log(1-y)\leq-c\), so \(n\log(1-y)\to-c\) and
\(\varphi_n(\lambda)\to e^{-c}\). The Poisson law of mean \(\mu\) has transform
\(\sum_ke^{-\mu}\mu^ke^{-\lambda k}/k!=e^{-\mu}e^{\mu e^{-\lambda}}=e^{-c}\).
By part~(2) of \cref{thm:p15-w02b-p15-012-laplace-continuity}, \(F_n\to F\)
with transform \(e^{-c}\); by part~(1) of
\cref{thm:p15-w02b-p15-012-laplace-uniqueness}, \(F\) is the Poisson law.
Since \(e^{-\mu(1-e^{-\lambda})}\to1\) as \(\lambda\to0+\), \(F\) is proper and
\(F_n\Rightarrow F\). The interval \((k-\frac12,k+\frac12]\) is an interval of
continuity of \(F\) containing only the integer \(k\), so the binomial
probability of \(k\) converges to \(e^{-\mu}\mu^k/k!\).

(2) \(\varphi_n(\lambda)=e^{-n\lambda}\to0\). Every bounded interval lies in
\((-\infty,x]\) for some \(x\), and \(\delta_n\) gives it mass \(0\) once
\(n>x\), so \(\delta_n\to0\). Likewise
\(\varphi_n=\frac12+\frac12e^{-n\lambda}\to\frac12\) and
\(\frac12\delta_0+\frac12\delta_n\to\frac12\delta_0\). The limits of
\(\varphi(\lambda)\) as \(\lambda\to0+\) are \(0\) and \(\frac12\), the
masses of the limits.

(3) A bounded interval lies in \((-\infty,x]\) for some \(x\), and
\(U_n\) gives it mass \(0\) once \(n>x\), so \(U_n\to0\). The transform of
\(U_n\) is \(e^{n^2}e^{-\lambda n}=e^{n(n-\lambda)}\), which tends to
\(\infty\) for every real \(\lambda\). In particular
\(\omega_n(a)=e^{n(n-a)}\) is unbounded for every \(a\), and the conclusion of
part~2 of \cref{thm:p15-w02b-p15-012-laplace-extended-continuity} fails, the
transform of the limit being \(0\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-integer-values}{thm:p15-w02b-p15-012-laplace-integer-values}{The change of variable \(y=e^{-x}\) and Hausdorff's uniqueness}
\begin{proof}
Since \(X\) takes values in \([0,\infty)\), \(Y=e^{-X}\) takes values in
\((0,1]\), so \(G\) is a probability distribution concentrated on \((0,1]\).
For \(0<y<1\) and \(x=-\log y>0\), the events \(\{Y\leq y\}\) and
\(\{X\geq x\}\) coincide, so \(G(y)=1-\mathbb P\{X<x\}=1-F(x-)\), which is
\(1-F(x)\) when \(F\) is continuous at \(x\).

For \(\lambda>0\) put \(k_\lambda(y)=y^\lambda\) on \([0,1]\), with
\(k_\lambda(0)=0\). By the integration rule for the law of a random variable
(\cref{thm:p15-expectation-law-integral}), applied to the nonnegative Borel
functions \(k_\lambda\) and \(x\mapsto e^{-\lambda x}\),
\[
 \int_0^1y^\lambda\,G\{dy\}=\mathbb E\,Y^\lambda=\mathbb E\,e^{-\lambda X}
   =\int_0^\infty e^{-\lambda x}\,F\{dx\}=\varphi(\lambda),
\]
which is (1.4). At \(\lambda=k\in\{1,2,\ldots\}\) this says that the \(k\)th
moment of \(G\) is \(\varphi(k)\); the moment of order \(0\) is \(1\).

Let \(F'\) be a second probability distribution concentrated on
\([0,\infty)\) whose transform \(\varphi'\) satisfies \(\varphi'(k)=\varphi(k)\)
for \(k=1,2,\ldots\), and let \(G'\) be the corresponding law of
\(e^{-X'}\). Then \(G\) and \(G'\) are probabilities on \([0,1]\) with the same
moment sequence, and the uniqueness part of
\cref{thm:p15-feller-hausdorff} gives \(G=G'\). The map
\(x\mapsto e^{-x}\) is a homeomorphism of \([0,\infty)\) onto \((0,1]\), so for
every Borel \(A\subseteq[0,\infty)\) the image \(e^{-A}\) is Borel, and
\(F\{A\}=\mathbb P\{Y\in e^{-A}\}=G\{e^{-A}\}=G'\{e^{-A}\}=F'\{A\}\). Since
both vanish on \((-\infty,0)\), \(F=F'\).

Now let \(F\) be possibly defective, with mass \(m=F\{[0,\infty)\}\leq1\).
Define \(G\{B\}=F\{x\geq0:e^{-x}\in B\}+(1-m)\,\delta_0\{B\}\) for Borel
\(B\subseteq[0,1]\); this is a probability on \([0,1]\). If \(m=0\), then
\(F=0\) and \(G=\delta_0\). If \(m>0\), \(F/m\) is a probability distribution,
and the identity above, multiplied by \(m\), gives
\(\int_{(0,1]}y^k\,G\{dy\}=\varphi(k)\) for \(k\geq1\); the atom at \(0\) adds
\(0^k=0\). So in every case the moments of \(G\) are \(1,\varphi(1),\varphi(2),\ldots\),
and \(G\) is determined by them. Finally
\(F\{A\}=G\{e^{-A}\}\) for Borel \(A\subseteq[0,\infty)\), because
\(e^{-A}\subseteq(0,1]\) does not contain the added atom. Hence \(F\) is
determined by \(\varphi(1),\varphi(2),\ldots\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-laplace-xiii1-examples}{thm:p15-w02b-p15-012-laplace-xiii1-examples}{Generating functions, gamma integrals, divergence and differentiation}
\begin{proof}
(a) The law is \(\sum_np_n\delta_n\), so by countable additivity and monotone
convergence \(\varphi(\lambda)=\sum_np_ne^{-n\lambda}\). Since
\(0<e^{-\lambda}<1\) and \(\sum p_n=1\), the series \(P(s)\) converges at
\(s=e^{-\lambda}\), and the two series are the same.

(b) By \cref{def:p15-gamma-density} and \cref{thm:p15-gamma-normalization},
the gamma density of shape \(\alpha\) and rate \(\lambda+1>0\) has integral
one:
\(\int_0^\infty\frac{(\lambda+1)^\alpha}{\Gamma(\alpha)}x^{\alpha-1}e^{-(\lambda+1)x}\,dx=1\).
Dividing by \((\lambda+1)^\alpha\) gives
\(\varphi_\alpha(\lambda)=\int_0^\infty e^{-\lambda x}f_\alpha(x)\,dx
=\frac1{\Gamma(\alpha)}\int_0^\infty e^{-(\lambda+1)x}x^{\alpha-1}\,dx=(\lambda+1)^{-\alpha}\).

(c) Put \(\alpha=a+1>0\). The same normalization with rate \(\lambda>0\) gives
\(\int_0^\infty x^{a}e^{-\lambda x}\,dx=\Gamma(a+1)/\lambda^{a+1}\).

(d) Let \(a\) be real. For \(\lambda>a\) and \(R>0\),
\(\int_0^Re^{-(\lambda-a)x}\,dx=(1-e^{-(\lambda-a)R})/(\lambda-a)\), and
monotone convergence as \(R\to\infty\) gives \(1/(\lambda-a)\). For
\(\lambda\leq a\) the integrand \(e^{(a-\lambda)x}\) is at least \(1\) on
\([0,\infty)\), so the integral over \([0,R]\) is at least \(R\) and (1.6)
diverges.

(e) For real \(\lambda\) and \(x\geq x_0=\max(\lambda,0)\), the exponent
\(x^2-\lambda x=x(x-\lambda)\) is nonnegative, so
\(\int_0^Re^{-\lambda x}e^{x^2}\,dx\geq R-x_0\) for \(R>x_0\), and (1.6)
diverges.

(f) Fix \(\lambda>0\) and let \(0<|\epsilon|<\lambda/2\). Then
\[
 \frac{\varphi(\lambda+\epsilon)-\varphi(\lambda)}\epsilon
  =\int_0^\infty e^{-\lambda x}\,\frac{e^{-\epsilon x}-1}\epsilon\,F\{dx\}.
\]
Since \(e^{-\epsilon x}-1=-\int_0^\epsilon xe^{-sx}\,ds\), we have
\(|e^{-\epsilon x}-1|\leq|\epsilon|xe^{|\epsilon|x}\), so the integrand is
bounded by \(xe^{-\lambda x/2}\leq2/(e\lambda)\), the maximum of
\(xe^{-cx}\) on \([0,\infty)\) being \(1/(ec)\). The integrand tends to
\(-xe^{-\lambda x}\) as \(\epsilon\to0\), and \(F\) is a finite measure, so
dominated convergence gives (1.7). The measure \(U\{dx\}=x\,F\{dx\}\) gives a
bounded interval \(I\subseteq[0,b]\) the mass \(U\{I\}\leq bF\{I\}<\infty\), and its
total mass is \(\int x\,F\{dx\}\). For \(F\{dx\}=(1+x)^{-2}dx\) on
\([0,\infty)\), the total mass of \(F\) is \(\int_0^\infty(1+x)^{-2}dx=1\); for
\(x\geq1\), \(1+x\leq2x\), so \(x(1+x)^{-2}\geq1/(4x)\) and
\(\int_1^Rx(1+x)^{-2}dx\geq\frac14\log R\to\infty\). Yet
\(xe^{-\lambda x}\leq1/(e\lambda)\), so the transform of \(U\) is finite for
every \(\lambda>0\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-parseval}{thm:p15-w02b-p15-012-parseval}{Parseval by Fubini and the normal transform}
\begin{proof}
Identity (3.1) is \(e^{-i\zeta t}e^{i\zeta x}=e^{i\zeta(x-t)}\)
integrated against \(F\). The function \((\zeta,x)\mapsto e^{i\zeta(x-t)}\) is
bounded by \(1\) and jointly continuous, and \(G\times F\) is a finite
measure, so Fubini's theorem gives
\[
 \int e^{-i\zeta t}\varphi(\zeta)\,G\{d\zeta\}
   =\int\!\!\int e^{i\zeta(x-t)}\,G\{d\zeta\}\,F\{dx\}
   =\int\gamma(x-t)\,F\{dx\}.
\]
The density \(a\mathfrak n(a\zeta)\) is that of \(Z/a\) for a standard normal
\(Z\), a normal law of mean \(0\) and variance \(a^{-2}\). By
\cref{thm:p15-characteristic-normal-transform} its characteristic function is
\(\gamma(y)=e^{-y^2/(2a^2)}=\sqrt{2\pi}\,\mathfrak n(y/a)\). Substituting in
(3.2) gives (3.3). Since \(a\mathfrak n(a\zeta)=a(2\pi)^{-1/2}e^{-\frac12a^2\zeta^2}\),
dividing (3.3) by \(a\sqrt{2\pi}\) and using that \(\mathfrak n\) is even gives
(3.4) with the constant \(1/(2\pi)\).

Let \(g\) be bounded and continuous. Since
\(\int\!\int|g(t)|a^{-1}\mathfrak n((t-x)/a)\,dt\,F\{dx\}\leq\|g\|F\{(-\infty,\infty)\}\),
Fubini's theorem and the substitution \(t=x+az\) give
\[
 \int g(t)f_a(t)\,dt=\int\Bigl(\int g(x+az)\mathfrak n(z)\,dz\Bigr)F\{dx\}.
\]
As \(a\to0+\), the inner integral tends to \(g(x)\) by dominated convergence
(the integrand is bounded by \(\|g\|\mathfrak n(z)\) and converges to
\(g(x)\mathfrak n(z)\)), and it is bounded by \(\|g\|\). Dominated convergence
against the finite measure \(F\) gives \(\int g f_a\,dt\to\int g\,dF\).

If \(F\) is proper, \(f_a\geq0\), and the case \(g=1\) of the displayed
identity gives \(\int f_a\,dt=1\), so \(f_a\) is a probability density; call
its law \(\mathfrak N_a\star F\). The convergence just proved for every
bounded continuous \(g\) is \(\mathfrak N_a\star F\Rightarrow F\) by
\cref{thm:p15-weak-bounded-continuous}.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-cf-uniqueness}{thm:p15-w02b-p15-012-cf-uniqueness}{Uniqueness from the smoothed densities}
\begin{proof}
Let \(F\) and \(F'\) be possibly defective distributions satisfying (3.4) with
the same \(\varphi\) on the left. Their smoothed densities \(f_a\) and \(f_a'\)
then coincide for every \(a>0\), so by \cref{thm:p15-w02b-p15-012-parseval},
\(\int g\,dF=\lim_{a\to0+}\int gf_a\,dt=\int g\,dF'\) for every bounded
continuous \(g\). For \(r<s\) and \(k>1/(s-r)\) let \(g_k\) be \(0\) on
\((-\infty,r]\), \(k(x-r)\) on \([r,r+1/k]\), \(1\) on \([r+1/k,s]\),
\(1-k(x-s)\) on \([s,s+1/k]\) and \(0\) beyond. Then \(0\leq g_k\leq1\), each
\(g_k\) is continuous, and \(g_k\to\mathbf 1_{(r,s]}\) pointwise: a point
\(x\leq r\) gives \(0\), a point of \((r,s]\) gives \(1\) once
\(r+1/k\leq x\), and a point \(x>s\) gives \(0\) once \(s+1/k\leq x\). By
dominated convergence \(F\{(r,s]\}=F'\{(r,s]\}\) for all \(r<s\), and the
argument of part~(4) of \cref{thm:p15-w02b-p15-012-vague-tools} gives
\(F=F'\). If \(F\) and \(F'\) are probability distributions with the same
characteristic function, both satisfy (3.4) with that function by
\cref{thm:p15-w02b-p15-012-parseval}, so they are equal.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-cf-lemma}{thm:p15-w02b-p15-012-cf-lemma}{Every subsequential limit satisfies the same identity}
\begin{proof}
The limit \(\varphi\) is measurable as a pointwise limit of continuous
functions, and \(|\varphi|\leq1\). By part~(7) of
\cref{thm:p15-w02b-p15-012-vague-tools}, every subsequence of \((F_n)\) has a
further subsequence \(F_{n_k}\to F'\), with \(F'\) possibly defective. Fix
\(a>0\) and \(t\). In (3.4) for \((\varphi_{n_k},F_{n_k})\), the left side
tends to \(\frac1{2\pi}\int e^{-i\zeta t}\varphi(\zeta)e^{-\frac12a^2\zeta^2}d\zeta\)
by dominated convergence, the integrands being bounded by the integrable
\(e^{-\frac12a^2\zeta^2}\). The right side is \(\int g\,dF_{n_k}\) with
\(g(x)=a^{-1}\mathfrak n((t-x)/a)\), which is continuous and tends to \(0\) as
\(|x|\to\infty\); by part~(3) of \cref{thm:p15-w02b-p15-012-vague-tools} it
tends to \(\int g\,dF'\). Hence \(F'\) satisfies (3.4) with \(\varphi\) on the
left. By \cref{thm:p15-w02b-p15-012-cf-uniqueness}, at most one possibly
defective distribution does so; call it \(F\). Every subsequence of \((F_n)\)
has a further subsequence converging to \(F\), and part~(5) of
\cref{thm:p15-w02b-p15-012-vague-tools} gives \(F_n\to F\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-cf-continuity}{thm:p15-w02b-p15-012-cf-continuity}{Necessity by tightness, sufficiency by smoothing}
\begin{proof}
\emph{Necessity.} Let \(F_n\to F\) with \(F\) proper, and let \(\varphi\) be
the characteristic function of \(F\). By part~(6) of
\cref{thm:p15-w02b-p15-012-vague-tools}, \(F_n\Rightarrow F\), so by
\cref{thm:p15-weak-tightness-selection} the sequence is tight, and in the
uniform formulation there is, for \(\epsilon>0\), an \(R\) with
\(\sup_nF_n\{|x|\geq R\}\leq\epsilon\); enlarging \(R\) we may also assume
\(F\{|x|\geq R\}\leq\epsilon\). Choose \(T\geq R\) with \(\pm T\) non-atoms of
\(F\). Fix \(Z>0\). The functions \(x\mapsto e^{i\zeta x}\), \(|\zeta|\leq Z\),
are bounded by \(1\) on \([-T,T]\) and uniformly equicontinuous there, because
\(e^{iu}-e^{iv}=i\int_v^ue^{is}\,ds\) gives
\(|e^{i\zeta x}-e^{i\zeta y}|\leq Z|x-y|\). For \(|\zeta|\leq Z\),
\[
 \begin{aligned}
 |\varphi_n(\zeta)-\varphi(\zeta)|
   &\leq\Bigl|\int_{(-T,T]}e^{i\zeta x}\,F_n\{dx\}-\int_{(-T,T]}e^{i\zeta x}\,F\{dx\}\Bigr|\\
   &\qquad+F_n\{|x|\geq T\}+F\{|x|\geq T\},
 \end{aligned}
\]
and part~(2) of \cref{thm:p15-w02b-p15-012-vague-tools} makes the first term
tend to \(0\) uniformly in \(|\zeta|\leq Z\). Hence
\(\limsup_n\sup_{|\zeta|\leq Z}|\varphi_n-\varphi|\leq2\epsilon\) for every
\(\epsilon\): the convergence is uniform on \([-Z,Z]\). The function
\(\varphi\) is continuous everywhere by dominated convergence, and in
particular near the origin.

\emph{Sufficiency.} Let \(\varphi_n\to\varphi\) pointwise with \(\varphi\)
continuous at \(0\). By \cref{thm:p15-w02b-p15-012-cf-lemma}, \(F_n\to F\) for a
possibly defective \(F\) that satisfies (3.3) with \(\varphi\) on the left
(that form being (3.4) multiplied by \(a\sqrt{2\pi}\)). At \(t=0\), after the
substitution \(\zeta=s/a\), (3.3) reads
\[
 \int\varphi(s/a)\,\mathfrak n(s)\,ds=\int e^{-x^2/(2a^2)}\,F\{dx\}
   \leq F\{(-\infty,\infty)\}.
\]
As \(a\to\infty\), \(\varphi(s/a)\to\varphi(0)\) for each \(s\) by continuity at
\(0\), and \(|\varphi|\leq1\), so the left side tends to
\(\varphi(0)=\lim_n\varphi_n(0)=1\). Hence the mass of \(F\) is at least
\(1\), and \(F\) is proper. By the necessity part, \(\varphi_n\) converges to
the characteristic function of \(F\), which is therefore \(\varphi\), with the
stated continuity and uniform convergence.

\emph{Corollary.} If continuous \(\varphi\) is the pointwise limit of
characteristic functions \(\varphi_n\) of laws \(F_n\), the sufficiency part
applies, and \(\varphi\) is the characteristic function of the proper limit
\(F\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-convolution-powers}{thm:p15-w02b-p15-012-convolution-powers}{A null set of unit values, and symmetrization}
\begin{proof}
\emph{The set where \(\omega=1\).} Since \(\omega\) is real,
\(\omega(\zeta)=\int\cos\zeta x\,U\{dx\}\), and \(0\leq\omega\leq1\). Let
\(Z=\{\zeta:\omega(\zeta)=1\}\), a closed set because \(\omega\) is
continuous. If \(\zeta\in Z\), then \(\int(1-\cos\zeta x)\,U\{dx\}=0\) with a
nonnegative integrand, so \(U\) is concentrated on
\(\{x:\zeta x\in2\pi\mathbb Z\}\). Let
\(A=\{(\zeta,x):x\neq0,\ \zeta x\in2\pi\mathbb Z\}\), a Borel subset of the
plane, and fix \(T>0\). For fixed \(x\neq0\), the section
\(\{\zeta:\zeta x\in2\pi\mathbb Z\}\) is countable and has Lebesgue measure
\(0\). For fixed \(\zeta\in Z\), the section \(\{x:(\zeta,x)\in A\}\) has
\(U\)-mass \(U\{x\neq0\}\). By Tonelli's theorem for Lebesgue measure on
\([-T,T]\) times \(U\),
\[
 \bigl|Z\cap[-T,T]\bigr|\cdot U\{x\neq0\}
   \leq\int_{-T}^T\!\!\int\mathbf 1_A(\zeta,x)\,U\{dx\}\,d\zeta
   =\int\!\!\int_{-T}^T\mathbf 1_A(\zeta,x)\,d\zeta\,U\{dx\}=0 .
\]
Since \(U\) is not concentrated at the origin, \(U\{x\neq0\}>0\), so
\(Z\cap[-T,T]\) is a null set for every \(T\), and \(Z\) is null.

\emph{The limit of \(U^{n\star}\).} By the product rule in
\cref{thm:p15-characteristic-algebra}, \(U^{n\star}\) has characteristic
function \(\omega^n\), and \(\omega^n\to\mathbf 1_Z\) pointwise. By
\cref{thm:p15-w02b-p15-012-cf-lemma}, \(U^{n\star}\to F\) for a possibly
defective \(F\) satisfying (3.4) with \(\mathbf 1_Z\) on the left. Since \(Z\)
is null, that left side is \(0\) for all \(a\) and \(t\). The zero measure
also satisfies (3.4) with \(\mathbf 1_Z\) on the left, both sides being
\(0\). By the second assertion of
\cref{thm:p15-w02b-p15-012-cf-uniqueness}, \(F=0\). The zero measure has no
atoms, so every bounded interval is an interval of continuity, and
\(U^{n\star}\{I\}\to0\) for every bounded interval \(I\).

\emph{General \(G\).} If \(G=\delta_b\) with \(b\neq0\), then
\(G^{n\star}=\delta_{nb}\), and \(nb\) leaves every bounded interval. Otherwise
\(G\) is not concentrated at any single point. Let
\(X_1,X_1',X_2,X_2',\ldots\) be independent with law \(G\), and let
\({}^0G\) be the law of \(X_1-X_1'\). Its characteristic function is
\(|\gamma|^2\geq0\) by \cref{thm:p15-characteristic-algebra}, where
\(\gamma\) is that of \(G\). It is not \(\delta_0\): by Tonelli,
\(\mathbb P\{X_1=X_1'\}=\int G\{\{x\}\}\,G\{dx\}=\sum_xG\{\{x\}\}^2\), the sum
running over the countably many atoms. Each atom has mass \(p_x<1\), so
\(p_x^2<p_x\); the sum is \(0\) if there are no atoms and otherwise strictly
less than \(\sum_xp_x\leq1\). By the first part, \(({}^0G)^{n\star}\to0\). Put
\(S_n=X_1+\cdots+X_n\) and \(S_n'=X_1'+\cdots+X_n'\); they are independent
with law \(G^{n\star}\), and \(S_n-S_n'=\sum_{i\leq n}(X_i-X_i')\) has law
\(({}^0G)^{n\star}\). If \(I\) is a bounded interval of length \(\ell\), then
\(S_n,S_n'\in I\) forces \(|S_n-S_n'|\leq\ell\), so
\[
 G^{n\star}\{I\}^2=\mathbb P\{S_n\in I,\ S_n'\in I\}
   \leq({}^0G)^{n\star}\{[-\ell,\ell]\}\longrightarrow0 .
\]
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-polya-feller}{thm:p15-w02b-p15-012-polya-feller}{Inscribed polygons, the corollary, and the three failures}
\begin{proof}
\emph{Three-chord inequality.} Write \(\sigma(u,w)=(\omega(w)-\omega(u))/(w-u)\)
for \(0\leq u<w\). If \(u<v<w\), then \(v=\theta u+(1-\theta)w\) with
\(\theta=(w-v)/(w-u)\), and convexity
\(\omega(v)\leq\theta\omega(u)+(1-\theta)\omega(w)\) rearranges to
\(\sigma(u,v)\leq\sigma(u,w)\leq\sigma(v,w)\). Consequently \(\omega\) is
continuous on \((0,\infty)\): for \(0<s<x<t\) and \(y\in(s,t)\), the three-chord
inequality bounds \(\sigma(x,y)\) or \(\sigma(y,x)\) between \(\sigma(s,x)\)
and \(\sigma(x,t)\), so \(|\omega(y)-\omega(x)|\leq K|y-x|\) with
\(K=\max(|\sigma(s,x)|,|\sigma(x,t)|)\). With continuity at \(0\), \(\omega\)
is continuous on \([0,\infty)\), and, being even, on the line.

\emph{Inscribed polygons.} For \(m\geq1\) let \(x_k=k/m\), \(0\leq k\leq m^2\).
Let \(\omega_m\) be even, equal to the linear interpolation of \(\omega\) at the
nodes \(x_k\) on \([0,m]\), and equal to \(\omega(m)\) on \([m,\infty)\). Its
slopes are \(\sigma_k=\sigma(x_{k-1},x_k)\), \(1\leq k\leq m^2\), and \(0\) on
\((m,\infty)\). By the three-chord inequality the slopes are nondecreasing in
\(k\), and since \(\omega\) is nonincreasing all of them are \(\leq0\), the
slope of the last piece. Let \(\ell_k\) be the line through the \(k\)th piece.
For \(\zeta\geq x_k\), \(\omega_m(\zeta)-\omega_m(x_k)\) is the integral of
slopes \(\geq\sigma_k\), so \(\omega_m(\zeta)\geq\ell_k(\zeta)\); for
\(\zeta\leq x_{k-1}\), \(\omega_m(x_{k-1})-\omega_m(\zeta)\) is the integral of
slopes \(\leq\sigma_k\), so again \(\omega_m(\zeta)\geq\ell_k(\zeta)\). Hence
\(\omega_m\) is the maximum of finitely many affine functions on
\([0,\infty)\), and therefore convex there. It is nonincreasing, satisfies
\(0\leq\omega_m\leq1\), \(\omega_m(0)=1\), is continuous at \(0\), and has
finitely many linear pieces, with limit \(c=\omega(m)\in[0,1]\) at infinity.
By \cref{thm:p15-criteria-polya}, including its cases \(c\in[0,1)\) and
\(c=1\), \(\omega_m\) is a characteristic function.

\emph{Passage to the limit.} Fix \(\zeta\geq0\) and take \(m>\zeta\). Then
\(\zeta\) lies in some \([x_{k-1},x_k]\) of length \(1/m\), and since both
\(\omega\) and the interpolant are monotone there,
\(\omega_m(\zeta)\) lies between \(\omega(x_{k-1})\) and \(\omega(x_k)\). Both
tend to \(\omega(\zeta)\) as \(m\to\infty\) by continuity. By evenness,
\(\omega_m\to\omega\) on the line. The limit \(\omega\) is continuous, so the
corollary of \cref{thm:p15-w02b-p15-012-cf-continuity} shows that \(\omega\) is
a characteristic function.

\emph{The failures.} Every characteristic function is bounded by \(1\) and
continuous (\cref{thm:p15-characteristic-algebra}). So \(1+\zeta^2\), which
exceeds \(1\) for \(\zeta\neq0\), is not one, and neither is
\(\mathbf 1_{\{0\}}\). For \(\omega(\zeta)=-\frac12+\frac32e^{-|\zeta|}\),
\[
 \frac1{2T}\int_{-T}^T\omega(\zeta)\,d\zeta=-\frac12+\frac3{2T}(1-e^{-T})
   \longrightarrow-\frac12 ,
\]
while for a characteristic function the same limit equals the mass of the
law at \(0\) by \cref{thm:p15-characteristic-atom-inversion} with \(a=0\),
which is not negative. The convexity claims are direct: \(1+\zeta^2\) has
nondecreasing chord slopes; \(-\frac12+\frac32e^{-\zeta}\) is the sum of a
constant and a convex exponential on \([0,\infty)\); and for
\(\mathbf 1_{\{0\}}\), if \(0\leq u<w\) and \(0<\theta<1\), the point
\(\theta u+(1-\theta)w\) is positive, so the left side of the convexity
inequality is \(0\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-fourier-inversion}{thm:p15-w02b-p15-012-fourier-inversion}{Bounded convergence of the smoothed densities}
\begin{proof}
\emph{Theorem~3.} Let \(f_a\) be the function in (3.4). Since \(F\) is proper,
\cref{thm:p15-w02b-p15-012-parseval} shows that \(f_a\geq0\) is the density of
the probability distribution \(F_a=\mathfrak N_a\star F\) and that
\(F_a\Rightarrow F\) as \(a\to0+\). From the left side of (3.4),
\(|f_a(t)|\leq B=(2\pi)^{-1}\int|\varphi|\,d\zeta\), and as \(a\to0+\) the
integrand \(e^{-i\zeta t}\varphi(\zeta)e^{-\frac12a^2\zeta^2}\) tends to
\(e^{-i\zeta t}\varphi(\zeta)\) under the integrable bound \(|\varphi|\); by
dominated convergence \(f_a(t)\to f(t)\), the right side of (3.5). The same
bound and dominated convergence in \(x\) show that \(f\) is continuous, and
\(|f|\leq B\); as a limit of nonnegative reals, \(f\geq0\).

Take \(a_j\to0+\). For a bounded interval \(I\), bounded convergence on \(I\)
gives \(F_{a_j}\{I\}=\int_If_{a_j}\,dt\to\int_If\,dt\). If \(I=(r,s]\) with
\(r,s\) points of continuity of \(F\), then \(F_{a_j}\{I\}\to F\{I\}\) by the
definition of \(\Rightarrow\) (\cref{def:p15-weak-law-convergence}). Thus
\(F\{(r,s]\}=\int_r^sf\,dt\) for all such \(r<s\). Letting \(r\to-\infty\)
and \(s\to\infty\) through continuity points, monotone convergence gives
\(\int f\,dt=1\). The probability measures \(F\) and \(f\,dt\) agree on every
\((r,s]\) with endpoints in the dense set of continuity points of \(F\), and
by right-continuity of both distribution functions they agree on every
\((r,s]\); the argument of part~(4) of
\cref{thm:p15-w02b-p15-012-vague-tools} gives \(F=f\,dt\).

\emph{Corollary.} If \(\varphi\geq0\) and \(\varphi\in L\), Theorem~3 gives a
bounded density. Conversely let \(F\) have a density \(f\) with \(f\leq M\)
almost everywhere. At \(t=0\), (3.4) reads
\[
 \frac1{2\pi}\int\varphi(\zeta)e^{-\frac12a^2\zeta^2}\,d\zeta
 =\int\frac1a\mathfrak n\Bigl(\frac xa\Bigr)f(x)\,dx
 =\frac1{\sqrt{2\pi}\,a}\int e^{-x^2/(2a^2)}f(x)\,dx
 \leq M\int\frac1a\mathfrak n\Bigl(\frac xa\Bigr)dx=M,
\]
which is (3.7). As \(a\downarrow0\), \(e^{-\frac12a^2\zeta^2}\uparrow1\), and
\(\varphi\geq0\), so by monotone convergence the left side increases to
\((2\pi)^{-1}\int\varphi\,d\zeta\). Hence \(\int\varphi\,d\zeta\leq2\pi M\) and
\(\varphi\in L\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-xv3-inversion-examples}{thm:p15-w02b-p15-012-xv3-inversion-examples}{Plancherel through smoothing, uniform density limits, and interval inversion}
\begin{proof}
\emph{(c) Symmetrization.} Let \(X,X'\) be independent with law \(F\). By
\cref{thm:p15-characteristic-algebra}, \(X-X'\) has characteristic function
\(|\varphi|^2\). The variable \(-X'\) has density \(y\mapsto f(-y)\), so by
\cref{thm:p15-convolution-density} a density of \(X-X'\) is
\(\int f(x-z)f(-z)\,dz\); the substitution \(z=-y\)
(\cref{thm:p15-affine-lebesgue}) turns it into (3.9). Two densities of the
same law agree almost everywhere: if \(g_1,g_2\) are densities and
\(A=\{g_1>g_2\}\cap[-n,n]\), then \(\int_A(g_1-g_2)\,dx=0\) with a positive
integrand on \(A\), so \(A\) is null, and likewise with the roles exchanged.

\emph{A pointwise bound.} By the Cauchy--Schwarz inequality
(\cref{thm:p15-holder-integral}) and translation invariance of Lebesgue
measure,
\[
 {}^0f(x)\leq\Bigl(\int f(y+x)^2\,dy\Bigr)^{1/2}\Bigl(\int f(y)^2\,dy\Bigr)^{1/2}
   =\int f^2\,dy\qquad\text{for every }x,
 \tag{\(\ast\)}
\]
whenever the right side is finite.

\emph{From \(|\varphi|^2\in L\) to \(f^2\in L\).} Let \(|\varphi|^2\in L\). By
\cref{thm:p15-w02b-p15-012-fourier-inversion}, \({}^0F\) has the bounded
continuous density
\(g(x)=(2\pi)^{-1}\int e^{-i\zeta x}|\varphi(\zeta)|^2\,d\zeta\), so
\(g(0)=(2\pi)^{-1}\int|\varphi|^2\,d\zeta\), and \(g={}^0f\) almost everywhere.
For \(\delta>0\) let \(k_\delta(x)=\delta^{-1}\mathfrak n(x/\delta)\) and
\(h_\delta(w)=\int f(z)k_\delta(w-z)\,dz\). Then \(0\leq h_\delta\leq\mathfrak n(0)/\delta\)
and \(\int h_\delta\,dw=1\), so \(h_\delta^2\) is integrable. By Tonelli's
theorem,
\[
 \int h_\delta(w)^2\,dw=\iint f(y)f(z)\Bigl(\int k_\delta(w-y)k_\delta(w-z)\,dw\Bigr)dy\,dz
   =\iint f(y)f(z)\,k_{\sqrt2\delta}(z-y)\,dy\,dz ,
\]
because the inner integral is the density at \(z-y\) of the difference of
two independent normal variables of mean \(0\) and variance \(\delta^2\),
which is normal with variance \(2\delta^2\) (\cref{thm:p15-normal-sums}).
Substituting \(z=y+x\) and using Tonelli again,
\[
 \int h_\delta^2\,dw=\int k_{\sqrt2\delta}(x)\,{}^0f(x)\,dx
   =\int k_{\sqrt2\delta}(x)\,g(x)\,dx=\int\mathfrak n(s)\,g(\sqrt2\delta s)\,ds
   \longrightarrow g(0)\qquad(\delta\to0+),
\]
by bounded convergence, \(g\) being bounded and continuous.

Next, \(h_\delta\to f\) in \(L^1\). Put \(\tau(x)=\int|f(w-x)-f(w)|\,dw\leq2\).
Given \(\epsilon>0\), \cref{thm:p15-criteria-real-integration} supplies a
continuous \(c\) vanishing outside some \([-R,R]\) with
\(\int|f-c|\,dw<\epsilon\). By translation invariance,
\(\tau(x)\leq2\epsilon+\int|c(w-x)-c(w)|\,dw\), and for \(|x|\leq1\) the last
integral is at most \((2R+2)\sup_w|c(w-x)-c(w)|\), which tends to \(0\) as
\(x\to0\) by uniform continuity of \(c\). Hence \(\tau(x)\to0\). Since
\(\int k_\delta=1\),
\[
 \int|h_\delta-f|\,dw\leq\iint|f(w-x)-f(w)|\,k_\delta(x)\,dx\,dw
   =\int\mathfrak n(s)\,\tau(\delta s)\,ds\longrightarrow0
\]
by bounded convergence. Choose \(\delta_j\to0\) with
\(\int|h_{\delta_j}-f|\leq2^{-j}\). Then
\(\int\sum_j|h_{\delta_j}-f|\leq1\), so the series is finite almost
everywhere (\cref{thm:p15-nonnegative-integral-laws}) and
\(h_{\delta_j}\to f\) almost everywhere. Fatou's lemma gives
\[
 \int f^2\,dy\leq\liminf_j\int h_{\delta_j}^2\,dw=g(0)<\infty .
\]

\emph{Equality.} With \(f^2\in L\), \((\ast)\) gives
\({}^0f\leq\int f^2\) everywhere, hence \(g\leq\int f^2\) almost everywhere.
If \(g(0)>\int f^2\), continuity would give \(g>\int f^2\) on an interval
around \(0\), which has positive measure; so \(g(0)\leq\int f^2\). Together,
\(\int f^2=g(0)=(2\pi)^{-1}\int|\varphi|^2\), which is (3.8).

\emph{From \(f^2\in L\) to \(|\varphi|^2\in L\).} By \((\ast)\), \({}^0F\) has
the bounded density \({}^0f\). Its characteristic function \(|\varphi|^2\) is
nonnegative, so the corollary of
\cref{thm:p15-w02b-p15-012-fourier-inversion} gives \(|\varphi|^2\in L\), and
(3.8) follows from the previous step.

\emph{(d)} By (3.5) for \(\varphi_n\) and for \(\varphi\),
\(f_n(x)-f(x)=(2\pi)^{-1}\int e^{-i\zeta x}(\varphi_n(\zeta)-\varphi(\zeta))\,d\zeta\),
and \(|e^{-i\zeta x}|=1\) gives the bound, which does not depend on \(x\) and
tends to \(0\) by (3.10).

\emph{(e)} Let \(V\) be uniform on \([-h,0]\) and independent of \(X\), which
has law \(F\). The characteristic function of \(V\) is
\(\psi_h(\zeta)=h^{-1}\int_{-h}^0e^{i\zeta v}\,dv=(1-e^{-i\zeta h})/(i\zeta h)\) for
\(\zeta\neq0\), and \(\psi_h(0)=1\); thus \(|\psi_h|\leq1\) and
\(|\psi_h(\zeta)|\leq2/(h|\zeta|)\). By \cref{thm:p15-characteristic-algebra},
\(X+V\) has characteristic function \(\varphi\psi_h\). By
\cref{thm:p15-convolution-density}, a density of \(X+V\) is
\(z\mapsto h^{-1}\int\mathbf 1_{[-h,0]}(z-y)\,F\{dy\}=h^{-1}F\{[z,z+h]\}\), which
differs from \(d(z)=(F(z+h)-F(z))/h\) only at the countably many atoms of
\(F\); so \(d\) is also a density, and it is right-continuous. If
\(\varphi\psi_h\in L\), Theorem~3 shows that \(X+V\) has the continuous density
\(c\) given by the right side of (3.11). Then \(c=d\) almost everywhere, and if
\(c(x)\neq d(x)\) for some \(x\), right-continuity of \(d-c\) would make them
differ on an interval \([x,x+\eta)\) of positive measure. Hence \(c=d\)
everywhere, which is (3.11). If \(|\varphi(\zeta)|\leq C/|\zeta|\) for
\(|\zeta|\geq\zeta_0\), then \(|\varphi\psi_h|\leq1\) on
\([-\zeta_0,\zeta_0]\) and \(\leq2C/(h\zeta^2)\) outside, so
\(\varphi\psi_h\in L\).

For (3.12), \(F_a\) has characteristic function
\(\varphi(\zeta)e^{-\frac12a^2\zeta^2}\), and its product with \(\psi_h\) is
integrable, so (3.11) for \(F_a\) is (3.12), for every \(x\). By
\cref{thm:p15-w02b-p15-012-parseval}, \(F_a\Rightarrow F\); if \(x\) and \(x+h\)
are continuity points of \(F\), then \(F_a(x)\to F(x)\) and
\(F_a(x+h)\to F(x+h)\) as \(a\to0+\), and the left side of (3.12) tends to
\((F(x+h)-F(x))/h\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-lattice-inversion}{thm:p15-w02b-p15-012-lattice-inversion}{Termwise integration of the lattice series}
\begin{proof}
Multiplying (3.13) by \(e^{-i(b+rh)\zeta}\) gives the series
\(\sum_kp_ke^{i(k-r)h\zeta}\). Its partial sums are bounded in modulus by
\(\sum_kp_k=1\), so dominated convergence on the bounded interval
\([-\pi/h,\pi/h]\) permits termwise integration. For an integer \(j\neq0\),
\[
 \int_{-\pi/h}^{\pi/h}e^{ijh\zeta}\,d\zeta=\frac{e^{ij\pi}-e^{-ij\pi}}{ijh}
   =\frac{2\sin(j\pi)}{jh}=0,
\]
while for \(j=0\) the integral is \(2\pi/h\). Only the term \(k=r\) survives,
and the integral equals \(2\pi p_r/h\), which is (3.14).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-semigroup-continuity}{thm:p15-w02b-p15-012-semigroup-continuity}{Completeness of the function space and Feller's estimate}
\begin{proof}
If \((u_n)\) is Cauchy in \(\mathscr L\), then for each \(x\) the numbers
\(u_n(x)\) are Cauchy, with a limit \(u(x)\); the usual estimate
\(|u_n(x)-u(x)|\leq\sup_{m\geq n}\|u_n-u_m\|\) shows that \(u\) is bounded and
\(\|u_n-u\|\to0\), so \(u\in\mathscr L\) by property (ii).

For \(0<t'<t''\), the semigroup property gives
\(\mathfrak Q(t'')u-\mathfrak Q(t')u=\mathfrak Q(t')[\mathfrak Q(t''-t')u-u]\),
and \(\|\mathfrak Q(t')\|\leq1\) gives (8.7). For \(t'=0\) (8.7) is an equality,
since \(\mathfrak Q(0)=\mathbf 1\). Given \(\epsilon>0\), continuity at the
origin yields \(\delta>0\) with \(\|\mathfrak Q(s)u-u\|<\epsilon\) for
\(0\leq s<\delta\); then \(\|\mathfrak Q(t'')u-\mathfrak Q(t')u\|<\epsilon\)
whenever \(0\leq t''-t'<\delta\), which is uniform continuity. For \(s,t>0\),
\(\mathfrak Q(s)\mathfrak Q(t)=\mathfrak Q(s+t)=\mathfrak Q(t)\mathfrak Q(s)\),
and \(\mathfrak Q(0)=\mathbf 1\) commutes with everything.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-pseudo-poisson}{thm:p15-w02b-p15-012-pseudo-poisson}{The pseudo-Poisson series}
\begin{proof}
Since \(\|T^n\|\leq\|T\|^n\leq1\), the series (9.1) has
\(\sum_n e^{-\alpha t}(\alpha t)^n\|T\|^n/n!=e^{-\alpha t+\alpha t\|T\|}\leq1\)
as a bound for the sum of the norms of its terms, so it converges in operator
norm; for each \(u\) the partial sums converge in \(\mathscr L\), which is
complete by \cref{thm:p15-w02b-p15-012-semigroup-continuity}. The same bound
gives \(\|\mathfrak Q(t)\|\leq e^{-\alpha t+\alpha t\|T\|}\leq1\). Each \(T^n\)
is positive, so if \(u\geq0\) every partial sum applied to \(u\) is
nonnegative, and so is the uniform limit: \(\mathfrak Q(t)\) is a contraction.
If \(T1=1\), every partial sum maps \(1\) to
\(e^{-\alpha t}\sum_{n\leq N}(\alpha t)^n/n!\), whose limit is \(1\).

Semigroup property: for \(u\in\mathscr L\),
\[
 \mathfrak Q(s)\mathfrak Q(t)u
  =e^{-\alpha(s+t)}\sum_{j}\frac{(\alpha s)^j}{j!}T^j\sum_k\frac{(\alpha t)^k}{k!}T^ku
  =e^{-\alpha(s+t)}\sum_{j,k}\frac{(\alpha s)^j(\alpha t)^k}{j!\,k!}T^{j+k}u,
\]
where the first equality applies the bounded operator \(T^j\) to a norm
convergent series, and the double series converges absolutely in norm, with
sum of norms at most \(e^{\alpha(s+t)}\|u\|\). Grouping the terms with
\(j+k=n\) and using the binomial theorem gives
\(e^{-\alpha(s+t)}\sum_n(\alpha(s+t))^nT^nu/n!=\mathfrak Q(s+t)u\). Finally
\(\mathfrak Q(0)=\mathbf 1\), and
\(\mathfrak Q(h)-\mathbf 1=(e^{-\alpha h}-1)\mathbf 1+e^{-\alpha h}\sum_{n\geq1}(\alpha h)^nT^n/n!\)
has norm at most \((1-e^{-\alpha h})+e^{-\alpha h}(e^{\alpha h}-1)=2(1-e^{-\alpha h})\to0\),
so the semigroup is continuous, even in operator norm.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-semigroup-exponential}{thm:p15-w02b-p15-012-semigroup-exponential}{The exponential formula by Chebyshev's inequality}
\begin{proof}
By induction on the semigroup property, \(\mathfrak Q(nh)=\mathfrak Q(h)^n\) for
\(n\geq1\), and \(\mathfrak Q(0)=\mathbf 1=\mathfrak Q(h)^0\). Hence (9.3) is
(9.1) with the contraction \(T=\mathfrak Q(h)\) and \(\alpha=1/h\), which is
the pseudo-Poisson semigroup generated by \(h^{-1}[\mathfrak Q(h)-\mathbf 1]\);
\cref{thm:p15-w02b-p15-012-pseudo-poisson} shows that it is a continuous
semigroup of contractions.

Write \(\pi_n=e^{-t/h}(t/h)^n/n!\), the Poisson weights of mean and variance
\(t/h\). Since \(\sum\pi_n=1\) and both series converge in norm,
\[
 \mathfrak Q_h(t)u-\mathfrak Q(t)u=\sum_{n=0}^\infty\pi_n[\mathfrak Q(nh)u-\mathfrak Q(t)u],
 \tag{9.5}
\]
and the norm of the left side is at most \(\sum_n\pi_n\|\mathfrak Q(nh)u-\mathfrak Q(t)u\|\).
If \(|nh-t|<\delta\), (8.7) of \cref{thm:p15-w02b-p15-012-semigroup-continuity}
gives \(\|\mathfrak Q(nh)u-\mathfrak Q(t)u\|\leq\|\mathfrak Q(|nh-t|)u-u\|\leq\epsilon\).
For the remaining \(n\) the norm is at most \(2\|u\|\), and by Chebyshev's
inequality for a Poisson variable \(N\) of mean \(t/h\),
\[
 \sum_{|nh-t|\geq\delta}\pi_n=\mathbb P\{|N-t/h|\geq\delta/h\}
   \leq\frac{t/h}{(\delta/h)^2}=\frac{th}{\delta^2}.
\]
Thus \(\|\mathfrak Q_h(t)u-\mathfrak Q(t)u\|\leq\epsilon+2\|u\|th\delta^{-2}\).
For \(0\leq t\leq b\) the bound is at most \(\epsilon+2\|u\|bh\delta^{-2}\),
which is below \(2\epsilon\) for small \(h\), uniformly in \(t\); since
\(\delta\) depends only on \(\epsilon\) and \(u\), this proves (9.4)
uniformly in finite \(t\)-intervals.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-subordinated-variant}{thm:p15-w02b-p15-012-subordinated-variant}{Compound Poisson weights}
\begin{proof}
Let \(p^{*k}\) be the \(k\)-fold convolution of \((p_j)\), with
\(p^{*0}\) the unit mass at \(0\); it is the law of a sum of \(k\) independent
variables with law \((p_j)\), so it has mean \(km_1\) and second moment
\(km_2+k(k-1)m_1^2\), and \(p(\zeta)^k=\sum_np^{*k}_n\zeta^n\) for
\(|\zeta|\leq1\). For \(|\zeta|\leq1\),
\[
 \exp(s\alpha[p(\zeta)-1])=e^{-s\alpha}\sum_k\frac{(s\alpha)^k}{k!}p(\zeta)^k
   =\sum_n\Bigl(e^{-s\alpha}\sum_k\frac{(s\alpha)^k}{k!}p^{*k}_n\Bigr)\zeta^n,
\]
the rearrangement being justified because the double series of absolute
values is at most \(e^{-s\alpha}\sum_k(s\alpha)^k/k!=1\). Power-series
coefficients on \(|\zeta|<1\) are unique, so
\(u_n(s)=e^{-s\alpha}\sum_k(s\alpha)^kp^{*k}_n/k!\). These are nonnegative and
sum to \(1\): \(\{u_n(s)\}\) is the Poisson mixture of the laws \(p^{*k}\)
with Poisson weights of mean \(s\alpha\). Its mean is
\(\sum_k e^{-s\alpha}(s\alpha)^kkm_1/k!=s\alpha m_1=bs\), and its second moment
is \(\sum_ke^{-s\alpha}(s\alpha)^k[km_2+k(k-1)m_1^2]/k!=s\alpha m_2+(s\alpha)^2m_1^2\),
so its variance is \(s\alpha m_2=cs\).

The two expressions in (9.9) agree because \(\mathfrak Q(nh)=\mathfrak Q^n(h)\).
For \(s=t/(bh)\) the weights \(u_n(s)\) have mean \(t/h\) and variance
\(ct/(bh)\), so for a variable \(N\) with this law, \(hN\) has mean \(t\) and
variance \(cth/b\). The proof of
\cref{thm:p15-w02b-p15-012-semigroup-exponential}, with these weights in place
of \(\pi_n\), gives
\(\|\mathfrak Q_h(t)u-\mathfrak Q(t)u\|\leq\epsilon+2\|u\|\,\mathbb P\{|hN-t|\geq\delta\}
\leq\epsilon+2\|u\|cth/(b\delta^2)\), and the conclusion follows as there.

The series (9.10) converges in operator norm, its terms having norms at most
\(p_n\); it is positive with norm at most \(1\), hence a contraction. By
induction on \(k\), \((\mathfrak Q^\#)^k=\sum_np^{*k}_n\mathfrak Q(nh)\): the
case \(k=0\) is \(\mathbf 1\), and multiplying two absolutely convergent
operator series and using \(\mathfrak Q(jh)\mathfrak Q(mh)=\mathfrak Q((j+m)h)\)
gives coefficients \(\sum_jp^{*k}_{n-j}p_j=p^{*(k+1)}_n\). Hence, for
\(u\in\mathscr L\) and \(s=t/(bh)\),
\[
 \mathfrak Q_h(t)u=\sum_n\sum_ke^{-s\alpha}\frac{(s\alpha)^k}{k!}p^{*k}_n\mathfrak Q(nh)u
   =\sum_ke^{-s\alpha}\frac{(s\alpha)^k}{k!}(\mathfrak Q^\#)^ku,
\]
the double series converging absolutely in norm with sum of norms at most
\(\|u\|\). Since \(s\alpha=\alpha t/(bh)\), the right side is (9.1) for the
contraction \(\mathfrak Q^\#\) with rate \(\alpha/(bh)\), evaluated at time
\(t\). This is (9.11), and by \cref{thm:p15-w02b-p15-012-pseudo-poisson} the
operators form a pseudo-Poisson semigroup in \(t\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-semigroup-examples}{thm:p15-w02b-p15-012-semigroup-examples}{The translation semigroup on uniformly continuous functions}
\begin{proof}
(1) Linear combinations of bounded uniformly continuous functions are such
functions. If \(\|u_n-u\|\to0\) with \(u_n\in\mathscr L_u\), then \(u\) is
bounded, and \(|u(x)-u(y)|\leq2\|u-u_n\|+|u_n(x)-u_n(y)|\) shows uniform
continuity. Since \(|\max(s,0)-\max(r,0)|\leq|s-r|\), \(u^+\) and
\(u^-=(-u)^+\) are uniformly continuous. Each \(T(\theta)\) maps
\(\mathscr L_u\) into itself, is linear and positive, satisfies
\(\|T(\theta)u\|\leq\|u\|\) and \(T(\theta)1=1\). The semigroup property is
\(u(x+s+t)=(T(s)T(t)u)(x)\), and \(T(0)=\mathbf 1\). Finally
\(\|T(h)u-u\|=\sup_x|u(x+h)-u(x)|\to0\) as \(h\to0+\) by uniform continuity.

(2) For the translation semigroup, (9.3) at \(x\) is
\(e^{-\theta/h}\sum_k(\theta/h)^ku(x+kh)/k!\), which is \(A_h(\theta)u(x)\) as
defined in (5.7). \Cref{thm:p15-w02b-p15-012-semigroup-exponential} gives the
norm convergence, uniformly for \(\theta\) in finite intervals.

(3) Both operators are contractions on \(\mathscr L_u\), so the norm of the
difference is at most \(2\). The function \(u\) used in the proof of
\cref{thm:p15-w02b-p15-012-exponential-formula},
\(u(x)=-1+2\max\{0,1-|x-\theta|/d\}\), is Lipschitz, hence in
\(\mathscr L_u\), and satisfies \(\|u\|=1\) and
\(A_h(\theta)u(0)-T(\theta)u(0)=-2\). Hence the norm on \(\mathscr L_u\) is
\(2\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-semigroup-integral}{thm:p15-w02b-p15-012-semigroup-integral}{Pointwise integrals as norm limits of Riemann sums}
\begin{proof}
By \cref{thm:p15-w02b-p15-012-semigroup-continuity}, \(s\mapsto\mathfrak Q(s)w\)
is uniformly continuous in norm, so for each \(x\) the function
\(s\mapsto\mathfrak Q(s)w(x)\) is continuous and bounded by \(\|w\|\), and
\(Ew(x)=\int\mathfrak Q(s)w(x)\,G\{ds\}\) is defined. Let \(c=G\{[0,\infty)\}\).

(1) Given \(\epsilon>0\) choose \(\eta>0\) with
\(\|\mathfrak Q(s)w-\mathfrak Q(s')w\|\leq\epsilon\) for \(|s-s'|\leq\eta\).
Put \(I_k=[k\eta,(k+1)\eta)\) and choose \(t_k\in I_k\). The series
\(R=\sum_k\mathfrak Q(t_k)w\,G\{I_k\}\) converges in norm, the norms of its
terms summing to at most \(c\|w\|\), so \(R\in\mathscr L\) by completeness,
and \(R(x)=\sum_k\mathfrak Q(t_k)w(x)G\{I_k\}\) because norm convergence
implies pointwise convergence. For every \(x\),
\[
 |Ew(x)-R(x)|\leq\sum_k\int_{I_k}|\mathfrak Q(s)w(x)-\mathfrak Q(t_k)w(x)|\,G\{ds\}\leq\epsilon c .
\]
Thus \(Ew\) is a uniform limit of members of \(\mathscr L\), and lies in
\(\mathscr L\) by property (ii); the Riemann sums converge to it in norm.

(2) Linearity follows from linearity of each \(\mathfrak Q(s)\) and of the
integral. \(|Ew(x)|\leq\int\|w\|\,G\{ds\}=c\|w\|\). If \(w\geq0\) then every
\(\mathfrak Q(s)w\geq0\), so \(Ew\geq0\). For a probability \(F\), \(c=1\) and
\(E\) is a contraction; if \(\mathfrak Q(s)1=1\) for all \(s\), then
\(E1(x)=\int1\,F\{ds\}=1\).

(3) For each \(x\),
\(E\mathfrak Q(t)w(x)=\int\mathfrak Q(s)\mathfrak Q(t)w(x)\,G\{ds\}=\int\mathfrak Q(t+s)w(x)\,G\{ds\}\),
using the semigroup property (also when \(s=0\) or \(t=0\)). For the other
order, with \(R\) as in (1),
\(\mathfrak Q(t)R=\sum_k\mathfrak Q(t+t_k)w\,G\{I_k\}\), because the bounded
operator \(\mathfrak Q(t)\) commutes with norm-convergent sums. The function
\(s\mapsto\mathfrak Q(t+s)w\) has the same modulus of continuity as
\(s\mapsto\mathfrak Q(s)w\), so the estimate of (1) gives
\(\|\mathfrak Q(t)R-\int\mathfrak Q(t+s)w\,G\{ds\}\|\leq\epsilon c\), while
\(\|\mathfrak Q(t)Ew-\mathfrak Q(t)R\|\leq\|Ew-R\|\leq\epsilon c\). Since
\(\epsilon\) is arbitrary, \(\mathfrak Q(t)Ew=\int\mathfrak Q(t+s)w\,G\{ds\}\).
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-resolvent-limit}{thm:p15-w02b-p15-012-resolvent-limit}{The resolvent near infinity and the recovery of the semigroup}
\begin{proof}
(1) The measure \(\lambda e^{-\lambda s}\,ds\) is a probability distribution,
so \cref{thm:p15-w02b-p15-012-semigroup-integral} shows that
\(\lambda\mathfrak R(\lambda)\) is a contraction, a transition operator when
every \(\mathfrak Q(s)\) is, and commutes with every \(\mathfrak Q(t)\).

(2) For each \(x\),
\(|\lambda\mathfrak R(\lambda)w(x)-w(x)|=|\int_0^\infty[\mathfrak Q(t)w(x)-w(x)]\lambda e^{-\lambda t}\,dt|
\leq\int_0^\infty\|\mathfrak Q(t)w-w\|\lambda e^{-\lambda t}\,dt\); the integrand
is continuous in \(t\) by (8.7), so the integral exists, and this is (9.14).
Substituting \(t=s/\lambda\), the right side is
\(\int_0^\infty\|\mathfrak Q(s/\lambda)w-w\|e^{-s}\,ds\). The integrand is at
most \(2\|w\|e^{-s}\) and tends to \(0\) for each \(s\) as \(\lambda\to\infty\),
by continuity at the origin. Dominated convergence gives (9.13).

(3) Fix \(w\) and \(x\). The function \(v(t)=\mathfrak Q(t)w(x)\) is bounded
and continuous on \([0,\infty)\), and
\(\mathfrak R(\lambda)w(x)=\int_0^\infty e^{-\lambda t}v(t)\,dt\) is its Laplace
integral (6.5). By \cref{thm:p15-w02b-p15-012-laplace-integral-inversion},
\(v(t)\) for \(t>0\) is the limit (6.6) of an expression in the derivatives of
this Laplace integral, which are determined by its values; and
\(v(0)=w(x)\). Thus the values \(\mathfrak R(\lambda)w(x)\), \(\lambda>0\),
determine \(\mathfrak Q(t)w(x)\) for all \(t\geq0\), and therefore all
operators of the semigroup.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-resolvent-derivatives}{thm:p15-w02b-p15-012-resolvent-derivatives}{Norm differentiation under the integral}
\begin{proof}
For \(k\geq0\) and \(\lambda>0\) let
\(R_k(\lambda)=\int_0^\infty e^{-\lambda s}s^k\mathfrak Q(s)\,ds\), the
integral of \cref{thm:p15-w02b-p15-012-semigroup-integral} for the finite
measure \(e^{-\lambda s}s^k\,ds\) of mass \(k!/\lambda^{k+1}\). Thus
\(R_0=\mathfrak R\). We show that for \(w\in\mathscr L\),
\[
 \begin{aligned}
 \Bigl\|\frac{R_k(\lambda+\eta)w-R_k(\lambda)w}{\eta}+R_{k+1}(\lambda)w\Bigr\|
   &\leq\frac{|\eta|}2\|w\|\int_0^\infty s^{k+2}e^{-\lambda s/2}\,ds,\\
   &\qquad0<|\eta|\leq\lambda/2,
 \end{aligned}
\]
so the norm derivative of \(R_k(\cdot)w\) at \(\lambda\) is
\(-R_{k+1}(\lambda)w\), and induction gives
\(\mathfrak R^{(n)}(\lambda)=(-1)^nR_n(\lambda)\), which is (10.25).

At a point \(x\), the expression inside the norm equals
\[
 \int_0^\infty e^{-\lambda s}\,\frac{e^{-\eta s}-1+\eta s}{\eta}\,s^k\,\mathfrak Q(s)w(x)\,ds .
\]
By Taylor's formula with Lagrange remainder,
\(e^{-y}-1+y=\frac12y^2e^{-\xi}\) for some \(\xi\) between \(0\) and \(y\), so
\(|e^{-y}-1+y|\leq\frac12y^2e^{|y|}\). With \(y=\eta s\) and
\(|\eta|\leq\lambda/2\), the integrand is at most
\(e^{-\lambda s}\frac12|\eta|s^2e^{\lambda s/2}s^k\|w\|\) in absolute value.
Integrating gives the displayed bound, uniformly in \(x\); the integral
\(\int_0^\infty s^{k+2}e^{-\lambda s/2}\,ds=(k+2)!(2/\lambda)^{k+3}\) is finite.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-abstract-inversion}{thm:p15-w02b-p15-012-abstract-inversion}{The gamma kernel applied to the semigroup}
\begin{proof}
By \cref{thm:p15-w02b-p15-012-resolvent-derivatives} at \(\lambda=n/t\),
\[
 \frac{(-1)^{n-1}}{(n-1)!}\mathfrak R^{(n-1)}(n/t)(n/t)^n
   =\int_0^\infty\mathfrak Q(s)\,\frac{(n/t)^ns^{n-1}e^{-ns/t}}{(n-1)!}\,ds
   =\int_0^\infty\mathfrak Q(s)\,G_{n,t}\{ds\},
\]
and \((n/t)^ns^{n-1}=(ns/t)^{n-1}n/t\) gives Feller's form of the density.
\(G_{n,t}\) is the gamma law of shape \(n\) and rate \(n/t\), with expectation
\(t\) and variance \(t^2/n\) (\cref{thm:p15-feller-kernel-examples}).
By (8.7), \(s\mapsto\mathfrak Q(s)w\) is uniformly continuous; given
\(\epsilon\) choose \(\delta\) with \(\|\mathfrak Q(s)w-\mathfrak Q(s')w\|\leq\epsilon\)
for \(|s-s'|<\delta\), independent of \(t\). At each \(x\),
\[
 \Bigl|\int\mathfrak Q(s)w(x)\,G_{n,t}\{ds\}-\mathfrak Q(t)w(x)\Bigr|
   \leq\int|\mathfrak Q(s)w(x)-\mathfrak Q(t)w(x)|\,G_{n,t}\{ds\}
   \leq\epsilon+2\|w\|G_{n,t}\{|s-t|\geq\delta\},
\]
and Chebyshev's inequality bounds the last probability by
\(t^2/(n\delta^2)\). For \(0<t\leq b\) the error is at most
\(\epsilon+2\|w\|b^2/(n\delta^2)\), which is below \(2\epsilon\) for large
\(n\), uniformly in \(t\). This proves (10.24) with the stated bound.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-w02b-p15-012-resolvent-examples}{thm:p15-w02b-p15-012-resolvent-examples}{Two explicit resolvents}
\begin{proof}
(1) For \(w\in\mathscr L\) and each \(x\),
\(\mathfrak R(\lambda)w(x)=\int_0^\infty e^{-\lambda s}e^{-\alpha s}\sum_n\frac{(\alpha s)^n}{n!}T^nw(x)\,ds\).
The terms are bounded in absolute value by
\(e^{-(\lambda+\alpha)s}(\alpha s)^n\|w\|/n!\), whose integrals
\(\alpha^n\|w\|/(\lambda+\alpha)^{n+1}\) have a finite sum, so summation and
integration may be interchanged, and
\(\int_0^\infty s^ne^{-(\lambda+\alpha)s}\,ds=n!/(\lambda+\alpha)^{n+1}\) gives
\(\mathfrak R(\lambda)w(x)=\sum_n\alpha^n(\lambda+\alpha)^{-n-1}T^nw(x)\), a
series that converges in operator norm. Then
\[
 \lambda\mathfrak R(\lambda)-\mathbf 1
   =\Bigl(\frac\lambda{\lambda+\alpha}-1\Bigr)\mathbf 1
    +\sum_{n\geq1}\frac{\lambda\alpha^n}{(\lambda+\alpha)^{n+1}}T^n,
\]
whose norm is at most
\(\frac\alpha{\lambda+\alpha}+\frac\lambda{\lambda+\alpha}\sum_{n\geq1}\bigl(\frac\alpha{\lambda+\alpha}\bigr)^n
=\frac{2\alpha}{\lambda+\alpha}\).

(2) The formula for \(\mathfrak R(\lambda)w(x)\) is the definition with
\(\mathfrak Q(s)w(x)=w(x+s)\). The operators \(E_{n,t}\) and \(T(t)\) are
contractions, so \(\|E_{n,t}-T(t)\|\leq2\). For \(0<\eta<t\) let
\(u_\eta(x)=-1+2\max\{0,1-|x-t|/\eta\}\), a Lipschitz function in
\(\mathscr L_u\) with \(\|u_\eta\|=1\), \(u_\eta(t)=1\), and \(u_\eta=-1\)
outside \((t-\eta,t+\eta)\). At \(x=0\),
\[
 E_{n,t}u_\eta(0)-T(t)u_\eta(0)=\int u_\eta(s)\,G_{n,t}\{ds\}-1
   \leq-1+2G_{n,t}\{(t-\eta,t+\eta)\}-1 .
\]
As \(\eta\to0+\), \(G_{n,t}\{(t-\eta,t+\eta)\}\to0\) because \(G_{n,t}\) has a
bounded density, so \(\|E_{n,t}-T(t)\|\geq2\).
\end{proof}
\end{annexitem}
