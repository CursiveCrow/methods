% Reciprocal proofs for the explicitly specified1945 Hempel reconstruction.
\begin{annexitem}{annex:p15-hempel-nicod-spelling}{thm:p15-hempel-nicod-spelling}{All equivalent spellings and specimen cases}
\begin{proof}
At any object, $R\to B$, $\neg B\to\neg R$ and $\neg R\lor B$
have the same Boolean value. Universal quantification preserves this
pointwise equivalence. For a raven which is black, the first displayed
conditional has true antecedent and consequent, whereas its contrapositive
has false antecedent. Nicod therefore classifies the first as confirming
and the second as neutral. A nonblack raven makes both antecedents true
and both consequents false, so both classifications are disconfirming.
A black nonraven makes both antecedents false, giving neutrality twice.
A nonblack nonraven makes only the contrapositive's antecedent true, with
its consequent also true, giving the final row. These exhaust the four
Boolean assignments to raven and black for the named specimen.

The consequent $R\land\neg R$ of the alternative spelling is always
false, so its matrix is equivalent to $\neg(R\land\neg B)$, hence to
$R\to B$. No object satisfies its antecedent and consequent together.
The final spelling has an always-true antecedent and consequent
$\neg R\lor B$, so it too is equivalent to the raven hypothesis.
A report of $\neg R(a)$, or of $B(a)$, entails that consequent and the
tautological antecedent. Unary instance sufficiency consequently confirms
the final spelling. Hypothesis equivalence transfers that classification
to $H_R$. This establishes the conditional result; the stipulated
confirmation principles, not logical equivalence by itself, supplied
the confirmation judgment.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-multiple-instance}{thm:p15-hempel-multiple-instance}{The ordered-pair counterexample}
\begin{proof}
For Boolean values $p,q$, the formula
$\neg(p\land q)\to(p\land\neg q)$ is equivalent to $p$.
If $p$ is false its antecedent is true and consequent false. If $p$ is
true and $q$ false, both are true; if both are true the antecedent is
false and the conditional true. Applying this with $p=R(x,y)$ and
$q=R(y,x)$ proves the universal equivalence.

In the displayed model, $R(a,b)$ is true and $R(b,a)$ false. Therefore
the antecedent and consequent of the $(a,b)$ matrix are true. But the
universal equivalent requires $R(b,a)$ too, and the report explicitly
denies it. The report is consistent, witnessed by the specified model,
and entails $\neg H_*$. It cannot be compatible with $H_*$. A rule
confirming the universal sentence from this isolated positive tuple
would violate individual compatibility, irrespective of probabilities.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-existential-import}{thm:p15-hempel-existential-import}{Two singleton separating interpretations}
\begin{proof}
On $\{u\}$ let both raven and black have extension $\{u\}$. The raven
universal is true and a raven exists. Its contrapositive is also true,
but no nonblack object exists, so the second augmented sentence is false.
On $\{u\}$ instead let both extensions be empty. The universal and its
contrapositive are true. No raven exists, but a nonblack object does;
the first augmented sentence is false and the second true. Both domains
are nonempty. The added existential conjuncts therefore change the
content in two different directions rather than preserving contraposition.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-prediction-gap}{thm:p15-hempel-prediction-gap}{A conservative hypothesis and a weaker unproductive one}
\begin{proof}
Suppose $E\not\models A$. Then the finite quantifier-free sentence
$E\land\neg A$ is satisfiable. In this identity-free relational language
its finitely many ground atoms can be assigned their satisfying Boolean
values on a domain containing distinct representatives for every name
which occurs. Distinct ground tuples with the same predicate then
correspond to distinct atomic inputs, so all those values can be realized
simultaneously. This also follows from any satisfying model's atomic
values, realized with the names separated.

Adjoin infinitely many unnamed objects, including one fresh $d$, and
preserve all predicate values on the old named tuples. Set $R_1(u,d)$
false for every object $u$, including every newly adjoined object. No
recorded tuple contains $d$, so this changes neither $E$ nor $A$.
For each $u$, the antecedent $\forall yR_1(u,y)$ of $J$ is now false.
Thus $J$ is true, while $E$ is true and $A$ false. This is an infinite
countermodel to $E\land J\models A$. Contraposition proves the stated
conservativity, for arbitrary ground Boolean $A$, not just a literal.

For the second assertion, partition $\{P(a),Q(a)\}$ as
$E_1=\{P(a)\}$ and $E_2=\{Q(a)\}$. The implication $H$ supplies $Q(a)$
from $P(a)$, while $P(a)$ alone does not. Let $K=H\lor T(c)$.
For every possible partition with nonempty $E_2$, make $T(c)$ true.
Then $K$ is true independently of $P(a),Q(a)$. If $E_1$ is empty,
either required observation can be false; if $E_1=\{P(a)\}$, make
$P(a)$ true and $Q(a)$ false; if $E_1=\{Q(a)\}$, make $Q(a)$ true
and $P(a)$ false. These cases exhaust the partitions. None yields all
the predictions demanded by the criterion for $K$, although $H\models K$.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-converse-collapse}{thm:p15-hempel-converse-collapse}{Hempel's three-condition argument}
\begin{proof}
Fix arbitrary $E,H$. Since $E\models E$, entailment gives $C(E,E)$.
The stronger $E\land H$ entails $E$, so converse consequence gives
$C(E,E\land H)$. Since $E\land H\models H$, special consequence
then gives $C(E,H)$. No property of $H$ was used. The argument applies
to consistent reports and to an arbitrary extraneous hypothesis as
well as to inconsistent reports. It is a proof about the conjunction
of the three conditions, not an objection to every restricted upward
inference or to all qualitative confirmation relations.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-joint-consistency}{thm:p15-hempel-joint-consistency}{The dependency between the consistency demands}
\begin{proof}
Let $E$ be consistent and let $\Gamma=\{H:C(E,H)\}$.
Suppose $E$ is not jointly compatible with $\Gamma$. No model of
$\Gamma$ satisfies $E$, which is precisely $\Gamma\models\neg E$.
General consequence, applied to the hypotheses in $\Gamma$, would
therefore give $C(E,\neg E)$. Individual compatibility forbids this:
$E$ and $\neg E$ have no common model. The supposition is false,
so $E$ and its entire confirmed class have a common model. This
semantic argument requires no assumption that $\Gamma$ is finite.
If only finite joint consequence is intended, first-order compactness
instead extracts a finite $\Gamma_0\subseteq\Gamma$ such that
$E\land\bigwedge\Gamma_0$ is inconsistent, and the same argument
uses $\Gamma_0\models\neg E$.

For the numerical example, both functions give value zero at inputs
zero and one. At input two, $f_0(2)=0$ and $f_1(2)=2$. There are
therefore two complete interpretations of one function on
$\{0,1,2\}$ agreeing on the stated record but disagreeing on its
remaining value. Each proposed function is separately compatible
with the record. Their conjunction would require that one function
take both distinct values at two, contradicting single-valuedness.
The mathematics illustrates the fit objection; it does not prove
that both laws must be called confirmed by a chosen relation.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-development}{thm:p15-hempel-development}{All constructors of finite development}
\begin{proof}
Prove the stronger assertion for formulas at arbitrary assignments.
An atom retains the same terms and predicate, so the claim is immediate.
Negation, conjunction, disjunction and implication preserve agreement
of their arguments by their Boolean truth clauses. For $\forall xA$,
truth means truth of $A$ at every element of the domain. Each element
is denoted by some $d\in D$ by the covering hypothesis. Thus this is
equivalent to truth of every $A[d/x]$, with capture avoided; induction
equates each instance with its development, giving the finite conjunction.
For $\exists xA$, a witness has a name in $D$ and conversely each named
true instance gives a witness. Induction gives the finite disjunction.
Repeated denotations cause only repeated conjuncts or disjuncts and do
not alter truth. These are all constructors of the stated language.

The nonempty-domain and covering hypotheses matter. On a two-element
domain with $a$ naming only one element, let $P(a)$ hold and $P$ fail
at the unnamed element. The development of $\forall xP(x)$ on $\{a\}$
is true but the universal is false. An empty conjunction for
$\forall x\bot$ would be true, whereas that universal is false in
every nonempty domain; hence empty named support cannot be handled by
silently introducing an empty actual domain.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-essential-repair}{thm:p15-hempel-essential-repair}{The report-equivalence defect and its repair}
\begin{proof}
$Q(b)\lor\neg Q(b)$ is valid, so the two reports are equivalent.
With all syntactically mentioned names, the first development is $P(a)$,
entailed by $E$. The second is $P(a)\land P(b)$, not entailed by $E'$.
To establish failure of general as well as direct confirmation under that
uncorrected convention, take domain $\{a,b\}$ with distinct names,
$P=\{a\}$, and any $Q$. This model satisfies $E'$. Every hypothesis
directly confirmed by $E'$ is true in it: $E'$ entails its development
on $\{a,b\}$ and the preceding lemma equates development with the
hypothesis on this domain. Every consequence of those hypotheses is
therefore true too. But $\forall xP(x)$ is false, so it is not a
consequence of the directly confirmed class.

As Boolean functions the two reports depend on $P(a)$, not $Q(b)$.
Their essential name set is therefore $\{a\}$. More generally, express
two reports over the union of their finitely many ground atoms. If
they are logically equivalent in the unrestricted identity-free
language, they have the same Boolean function: any Boolean assignment
is realized by interpreting all finitely many names distinctly and
setting each ground predicate tuple accordingly. Essential dependence
on an input is determined by the function, not its written formula.
The essential atom sets, and hence their name sets, coincide. Mutual
entailment of the reports then gives the same direct-confirmation
class and the same consequence closure.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-core-standards}{thm:p15-hempel-core-standards}{Entailment, joint closure and a common model}
\begin{proof}
Fix an admissible $E$ and $D=D(E)$. A quantifier-free formula is its
own development, so $E\in\Delta_E$. Thus $E\models H$ implies
$\Delta_E\models H$, proving entailment. If each $K\in\Gamma$ is
a consequence of $\Delta_E$, every model of $\Delta_E$ models all
of $\Gamma$. If $\Gamma\models H$, it models $H$ too. This proves
general consequence for any class $\Gamma$, including singleton,
conjunction and equivalent-hypothesis cases.

For joint consistency first eliminate every inessential atom from $E$
by Boolean equivalences. Call the resulting formula $E^*$. Its names
are in the finite nonempty $D$. Since $E$ is consistent, choose a
satisfying Boolean valuation for $E^*$. Construct a structure with
domain the names in $D$, each naming itself. Interpret each predicate
tuple mentioned in $E^*$ by that valuation, and assign remaining
predicate tuples arbitrarily. Distinct syntactic ground atoms cannot
impose inconsistent requirements on one tuple: the names have been
interpreted distinctly and identity is absent. Interpret every name
outside $D$ by one fixed element of $D$. The structure satisfies
$E^*$ and therefore $E$.

For each $K\in\Delta_E$, it satisfies $\operatorname{dev}_D(K)$
because $E$ entails that development. Its domain is covered by $D$,
so the development lemma shows it satisfies $K$. It consequently
satisfies every member of $\Delta_E$ and every consequence of that
class. It is a common model of $E$ and \emph{all} confirmed hypotheses.
This proves joint consistency without replacing it by separate
compatibility of a few examples. If an infinite domain is required,
replace this finite domain by $D\times\mathbb N$, pull predicates
back along first-coordinate projection, and let each named constant
denote its previous first coordinate paired with zero. Induction on
formulas preserves truth, with surjectivity supplying all existential
witnesses and universal instances. Absence of equality is essential
to this duplication argument. This gives an infinite common model too.

Equivalent admissible reports have identical essential supports and
direct-confirmation classes by the preceding proof; hence they have
the same confirmation classification. For consistent $E$, a hypothesis
and its negation cannot both be confirmed, because the common model
would satisfy a contradiction. Define disconfirmation through the
negation. Either the hypothesis is confirmed, its negation is confirmed,
or neither is; these are mutually exclusive and exhaust the cases.
No claim about empirical reliability or normative adequacy follows
from these formal properties.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-outside-name}{thm:p15-hempel-outside-name}{The genuinely additional consequence step}
\begin{proof}
Here $D(E)=\{a,b\}$. The development of
$\forall x(P(x)\lor Q(x))$ is
$(P(a)\lor Q(a))\land(P(b)\lor Q(b))$.
$P(a)$ supplies the first disjunction and $Q(b)$ the second.
The development of $\exists xP(x)$ is $P(a)\lor P(b)$, supplied
by $P(a)$. Both hypotheses are therefore in $\Delta_E$.
The former entails $P(c)\lor Q(c)$ by universal instantiation, so
the latter sentence is confirmed through consequence closure.
It is its own development, however, and is not entailed by $E$.
For a countermodel take distinct $a,b,c$, with $P=\{a\}$ and
$Q=\{b\}$. $E$ is true while $P(c)\lor Q(c)$ is false.
Thus the final hypothesis is not directly confirmed.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-likes}{thm:p15-hempel-likes}{The positive and negative finite-class calculations}
\begin{proof}
For the positive report the essential names are $a,b$.
$L(a,b)$ and $L(b,a)$ make respectively the two disjunctions of
the displayed development true, so the report entails that development
and directly confirms $H$. On domain $\{a,b,c\}$, let
$L=\{(a,b),(b,a)\}$. The report is true but $c$ likes nobody,
so $H$ is false. The report does not entail $H$.

For the negative report the development of $\neg H$, equivalently
$\exists x\forall y\neg L(x,y)$, is
\[
 (\neg L(a,a)\land\neg L(a,b))\lor
 (\neg L(b,a)\land\neg L(b,b)).
\]
The report entails its first disjunct, hence directly confirms $\neg H$.
On domain $\{a,b,c\}$ instead let
$L=\{(a,c),(b,c),(c,c)\}$. Both negative observations remain true,
yet everybody likes $c$, so $H$ is true. Thus the report does not
entail $\neg H$. Both reports are consistent. Their finite-class
support therefore must not be turned into a conclusive assertion
about all objects in an unspecified larger domain.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-evidence-extension}{thm:p15-hempel-evidence-extension}{Reversal under an enlarged report}
\begin{proof}
For $E=P(a)$, the development of $H=\forall xP(x)$ on its essential
class $\{a\}$ is $P(a)$, so $E$ directly confirms $H$.
For $E^+=P(a)\land\neg P(b)$, distinct names can be interpreted
on a two-element domain with $P=\{a\}$, proving consistency.
The negative observation gives $E^+\models\neg H$, since $b$
is a counterexample to the universal. Entailment of the satisfaction
construction therefore confirms $\neg H$ and disconfirms $H$.
Joint consistency prevents confirmation of $H$ at the same time.
The change is in the report, including its named class, not in the
logical equivalence of two spellings of one hypothesis.
\end{proof}
\end{annexitem}

\begin{annexitem}{annex:p15-hempel-verification-cases}{thm:p15-hempel-verification-cases}{Fresh objects and quantified-prefix boundaries}
\begin{proof}
$P(a)$ entails $\exists xP(x)$, so is a verifying report; $\neg P(a)$
entails $\neg\forall xP(x)$, so is a falsifying report. For any
consistent finite ground report, realize its atomic values with the
finitely many names distinct, and add infinitely many unnamed objects.
This is possible in the specified unrestricted language, with no
identity or background axioms. Preserving all named tuples preserves
the report. Setting $P$ true at one new object makes $\exists xP(x)$
true, so no such report falsifies it. Setting $P$ false at a new
object makes $\forall xP(x)$ false, so no such report verifies it.

For $H=\forall x\exists yL(x,y)$, choose a fresh unnamed $d$ and
set $L(u,d)$ true for every old and new object $u$. No recorded
tuple contains $d$, so the report is preserved and $H$ true.
Alternatively choose a fresh unnamed $e$ and set every $L(e,v)$
false, preserving the named tuples. Then $H$ is false. These two
infinite extensions show that no consistent finite ground report
entails either $H$ or $\neg H$.

For the source's mixed-prefix exceptions, $Q(a)$ makes
$\forall x\exists y(P(x)\lor Q(y))$ true by choosing $y=a$
for every $x$. In contrast, $\neg P(a)$ makes
$\forall x\exists y(P(x)\land Q(y))$ false at $x=a$, regardless
of $y$. Quantifier spelling alone therefore does not decide relative
verifiability or falsifiability. All these arguments are logical
relations to reports; none makes acceptance of a report irrevocable.
\end{proof}
\end{annexitem}
