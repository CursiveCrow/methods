\calculussheet{sheet:n-c-012-pnueli-er}{Pnueli's ER source rules}{Language: state assertions over a transition system \(\langle S,s_0,R\rangle\), with equality in the underlying assertion language; the displayed arrows are metalinguistic inferences. Here \(\leadsto\) is the source's strict-future temporal implication. First-order predicate-calculus theorems are also ER axioms. These are the source rules with the stated parameter typing; registration does not remove the printed A2/strict-future incompatibility discussed in the presentation \cite[p.~50]{Pnueli1977}.}
\hbrule{rule:n-c-012-pnueli-er-a1}{A1}{\forall s,s'\,
       (p(s)\land R(s,s')\Rightarrow q(s'))}{p\leadsto q}{\none}
\hbrule{rule:n-c-012-pnueli-er-a2}{A2}{\forall s\,(p(s)\Rightarrow q(s))}{p\leadsto q}{Literal source rule; unsound under the printed strict-future reading.}
\hbrule{rule:n-c-012-pnueli-er-r1}{R1}{p\leadsto q\quad
       \forall s,s'\,
       (r(s)\land R(s,s')\Rightarrow r(s'))}{(p\land r)\leadsto(q\land r)}{\none}
\hbrule{rule:n-c-012-pnueli-er-r2}{R2}{p\leadsto q\quad q\leadsto r}{p\leadsto r}{\none}
\hbrule{rule:n-c-012-pnueli-er-r3}{R3}{p_1\leadsto q\quad p_2\leadsto q}{(p_1\lor p_2)\leadsto q}{\none}
\hbrule{rule:n-c-012-pnueli-er-r4}{R4}{\widehat p_u\leadsto q\quad(u\in U\text{ arbitrary})}{(\exists_U\widehat p)\leadsto q}{\(U\ne\varnothing\); the derivation is uniform in \(u\), and \(q\) and every open assumption are independent of \(u\).}
