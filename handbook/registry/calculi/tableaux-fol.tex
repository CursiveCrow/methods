% AUTHORED, NOT EXTRACTED: v1.1 contains no classical first-order tableau
% calculus.  This unsigned parameter tableau follows the classical analytic
% presentation of Smullyan1968.  The base calculus T is equality-free; the
% equality extension T^= adds exactly the two rules marked "=" below
% (tableau reflexivity and left-to-right replacement), defined in
% def:p02-w02-p02-014-fol-equality, not an unstated reading of atomic
% identity.  Depth row D-P3-002 (first-order half).
%
% Formula occurrences are retained.  Propositional and delta occurrences are
% expanded once.  A gamma occurrence records the closed terms already used and
% may be expanded again at every new term; this distinction is required for
% fair saturation and first-order countermodel extraction.  Fairness ranges
% over every closed term of the active root language, not merely terms already
% occurring on a branch.
\calculussheet{sheet:tableaux-fol}{Classical First-Order Analytic Tableaux}{Language: for a finite root $\Gamma$ of sentences, equality-free for the base calculus $\mathsf T$ and possibly containing equations for its equality extension $\mathsf T^{=}$, let $\mathcal L_{\Gamma}$ contain exactly its nonlogical symbols and obtain $\mathcal L_{\Gamma}^*$ by adjoining a countable reserve $a_n$ ($n\in\mathbb N$) of new parameter constants; fix an enumeration of every closed $\mathcal L_{\Gamma}^*$-term; shape: a tableau rooted at $\Gamma$ is a finitely branching tree, possibly infinite, whose branches retain formula occurrences and metalinguistic expansion records; a finite closed tableau is a refutation; propositional and delta occurrences are marked after one expansion, whereas each gamma occurrence records the closed terms used and remains available for every unused enumerated term; a branch closes on $\bot$ or a complementary pair, is saturated when no applicable expansion is omitted, and is fair when every gamma occurrence is eventually instantiated with every term in the fixed enumeration, whether or not that term previously appeared on the branch; freshness: a delta expansion uses the parameter of least index not occurring on the branch; the task bookkeeping that makes the expansion records, saturation and fairness exact is \cref{def:p02-w02-p02-014-fol-branch}; negation: primitive, with dual connective and quantifier expansions; equality: none in $\mathsf T$; $\mathsf T^{=}$ adds the two rules marked $=$ (\cref{def:p02-w02-p02-014-fol-equality}), and in $\mathsf T^{=}$ a branch is saturated, hence fair, only when every reflexivity instance at every enumerated closed term and every replacement instance available on it has been used, so no finite branch of $\mathsf T^{=}$ is saturated; adequacy: closure is sound, an open saturated branch yields a quotient Herbrand model of the root, a finite closed tableau exists exactly for unsatisfiable roots, and fair systematic expansion semidecides unsatisfiability, for $\mathsf T$ and $\mathsf T^{=}$ alike (\cref{thm:p02-w02-p02-014-fol-soundness,thm:p02-w02-p02-014-fol-extraction,thm:p02-w02-p02-014-fol-completeness,thm:p02-w02-p02-014-fol-semidecision}); unlike the propositional sheet, exhaustive expansion need not terminate (\cref{thm:p02-w02-p02-014-fol-no-finite-saturation}).}
\hbrule{rule:tableaux-fol-and}{$\land$}{\varphi\land\psi}{\varphi,\;\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-fol-neg-and}{$\neg\land$}{\neg(\varphi\land\psi)}{\neg\varphi\;\mid\;\neg\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-fol-or}{$\lor$}{\varphi\lor\psi}{\varphi\;\mid\;\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-fol-neg-or}{$\neg\lor$}{\neg(\varphi\lor\psi)}{\neg\varphi,\;\neg\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-fol-imp}{$\to$}{\varphi\to\psi}{\neg\varphi\;\mid\;\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-fol-neg-imp}{$\neg\to$}{\neg(\varphi\to\psi)}{\varphi,\;\neg\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-fol-neg-neg}{$\neg\neg$}{\neg\neg\varphi}{\varphi}{linear expansion: retain and mark the principal occurrence, then append the conclusion on every open branch through it}
\hbrule{rule:tableaux-fol-forall}{$\forall$}{\forall x\,\varphi}{\varphi[t/x]}{gamma expansion: $t$ is any closed $\mathcal L_{\Gamma}^*$-term not yet used for this occurrence; retain the occurrence, record $t$, and use the book's total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-neg-exists}{$\neg\exists$}{\neg\exists x\,\varphi}{\neg(\varphi[t/x])}{gamma expansion: $t$ is any closed $\mathcal L_{\Gamma}^*$-term not yet used for this occurrence; retain the occurrence, record $t$, and use total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-exists}{$\exists$}{\exists x\,\varphi}{\varphi[a/x]}{delta expansion: $a$ is the least parameter constant not occurring earlier on the branch; retain and mark the principal occurrence and use total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-neg-forall}{$\neg\forall$}{\neg\forall x\,\varphi}{\neg(\varphi[a/x])}{delta expansion: $a$ is the least parameter constant not occurring earlier on the branch; retain and mark the principal occurrence and use total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-eq-refl}{$=$refl}{}{t=t}{equality extension $\mathsf T^{=}$ only: $t$ is any closed $\mathcal L_{\Gamma}^*$-term not yet used for this rule on the branch; the rule has no premise and, like a gamma expansion, remains available at every unused term of the fixed enumeration, which fairness requires to be used eventually}
\hbrule{rule:tableaux-fol-eq-repl}{$=$repl}{r=u\qquad\varphi[r/x_0]}{\varphi[u/x_0]}{equality extension $\mathsf T^{=}$ only: $r,u$ are closed $\mathcal L_{\Gamma}^*$-terms, $r=u$ and the atomic sentence $\varphi[r/x_0]$ occur on the branch, and $\varphi$ is an atomic formula in which the variable $x_0$ occurs exactly once and no other variable occurs, so the step rewrites one occurrence of $r$ to $u$, left to right; the two occurrences may coincide; each triple of the two occurrences and $\varphi$ is used at most once on a branch, and fairness requires every such triple to be used eventually; symmetry and transitivity are not rules but hold on every saturated branch (\cref{thm:p02-w02-p02-014-congruence})}
\hbrule{rule:tableaux-fol-close}{close}{\varphi\qquad\neg\varphi}{\text{closed branch}}{a branch containing both displayed formulas, or containing $\bot$, is closed and receives no further expansion; a tableau is closed iff every branch is closed}
