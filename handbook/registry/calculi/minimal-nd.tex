% Modern, explicitly bounded defined-negation propositional presentation.
% Not a transcription of Johansson's primitive-negation calculus.
\calculussheet{sheet:minimal-nd}{Minimal Propositional Natural Deduction}{Language: atoms and $\bot,\land,\lor,\to$, with $\neg A\defeq A\to\bot$; judgment: $\Gamma\derives_{\mathsf{MPL}}A$ from a finite rooted derivation tree whose actual open assumption occurrences belong to the supporting set $\Gamma$; assumption labels are metalinguistic natural numbers. Discharge classes may be empty. No falsity elimination, RAA, primitive negation, quantifiers or equality rules. Falsum is not required to be false at every semantic world.}
\noindent\emph{Presentation standing:} the propositional restriction of the empty-discharge NJ rule surface, with falsity elimination omitted. The modern falsum semantics is compared with \cite[Section~3.1]{ColacitoDeJonghVargas2016SubminimalDraft}; the full local proof is Theorem~\ref{thm:p08-mi-minimal-adequacy}.

\hbrule{rule:minimal-nd-or-i2}{$\lor I_2$}{\Gamma\derives_{\mathsf{MPL}}B}{\Gamma\derives_{\mathsf{MPL}}A\lor B}{\none}
