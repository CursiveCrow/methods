\calculussheet{sheet:p02-syllogistic-s}{Direct syllogistic calculus \(\mathsf S\)}{Language: \(\mathcal S\) of Definition~\ref{def:p02-syllogistic-language}; \(p,q\) are unary atoms, \(l\) a unary literal, and \(\varphi,\psi\) formulas of \(\mathcal S\). Derivations are finite trees with undischarged premises. Existential symmetry and universal contraposition are the formula identifications of Definition~\ref{def:p02-syllogistic-identifications}. Substitutions send atoms to atoms, not to arbitrary formulas. No equality, relations, Boolean term formation or reductio rule is added.}
\hbrule{rule:p02-syllogistic-s-d1}{D1}{\forall(q,l)\qquad\exists(p,q)}{\exists(p,l)}{\(p,q\) atoms; \(l\) a literal.}
\hbrule{rule:p02-syllogistic-s-d2}{D2}{\exists(p,l)\qquad\forall(p,q)}{\exists(q,l)}{\(p,q\) atoms; \(l\) a literal.}
\hbrule{rule:p02-syllogistic-s-d3}{D3}{\forall(q,\bar l)\qquad\exists(p,l)}{\exists(p,\bar q)}{\(p,q\) atoms; \(l\) a literal.}
\hbrule{rule:p02-syllogistic-s-b}{B}{\forall(p,q)\qquad\forall(q,l)}{\forall(p,l)}{\(p,q\) atoms; \(l\) a literal.}
\hbrule{rule:p02-syllogistic-s-a}{A}{\forall(p,\bar p)}{\forall(p,l)}{\(p\) an atom; \(l\) a literal.}
\hbrule{rule:p02-syllogistic-s-t}{T}{}{\forall(p,p)}{\(p\) an atom.}
\hbrule{rule:p02-syllogistic-s-i}{I}{\exists(p,l)}{\exists(p,p)}{\(p\) an atom; \(l\) a literal.}
\hbrule{rule:p02-syllogistic-s-x}{X}{\psi\qquad\bar\psi}{\varphi}{\(\varphi,\psi\in\mathcal S\); formula negation is the defined bar operation. No premise is discharged.}
