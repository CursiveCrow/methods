\calculussheet{sheet:n-c-012-pnueli-dn}{Pnueli's derived eventuality rule}{Language: state assertions over a transition system \(\langle S,s_0,R\rangle\), with equality in the underlying assertion language; the displayed arrows are metalinguistic inferences. Judgements are system-specific; \(F,G\) include the present and \(a\) is a designated action of the transition partition \cite[p.~54]{Pnueli1977}.}
\hbrule{rule:n-c-012-pnueli-dn-dn}{DN}{\begin{gathered}
 \forall s,s'\,
 (p_1(s)\land R(s,s')\land\neg a(s,s')\land p_2(s')
    \Rightarrow p_1(s'))\\
 \forall s,s'\,
 (p_1(s)\land a(s,s')\land p_2(s')\Rightarrow q(s'))
 \end{gathered}}{\vdash(p_1\land Gp_2)\Rightarrow Fq}{Only the action-occurrence-restricted executions of the presentation are allowed; \(a\) occurs infinitely often.}
