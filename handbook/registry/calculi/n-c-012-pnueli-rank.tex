\calculussheet{sheet:n-c-012-pnueli-rank}{Pnueli's printed rank principle}{Language: state assertions over a transition system \(\langle S,s_0,R\rangle\), with equality in the underlying assertion language; the displayed arrows are metalinguistic inferences. The predicate \(A\) is defined also at zero; rule instances use \(n>0\), and \(\leadsto\) has the source's strict-future meaning. This records the literal P2, not a sound rule endorsed by the handbook; both counterexamples remain in the adjacent text \cite[p.~49]{Pnueli1977}.}
\hbrule{rule:n-c-012-pnueli-rank-p2}{P2}{\begin{gathered}
   \forall s\bigl(\varphi(s)\Rightarrow
                    \exists n>0\ A(s,n)\bigr)\\
   \forall s,s'\ \forall n>0\,
   \bigl(A(s,n)\land R(s,s')\Rightarrow
         A(s',n-1)\lor\psi(s)\bigr)
  \end{gathered}}{\varphi\leadsto\psi}{The disjunct is the current-state \(\psi(s)\); no zero-rank boundary premise is added.}
