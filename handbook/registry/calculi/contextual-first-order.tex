\calculussheet{sheet:contextual-first-order}{Contextual intuitionistic first-order logic}{Judgments are term equations \(M=N:\sigma\,[\Gamma]\) and sequents \(\Phi\vdash\psi\,[\Gamma]\). Variable contexts \(\Gamma,\Delta\) are finite lists of distinct sorted variables; hypothesis lists \(\Phi,\Phi',\Phi''\) are finite lists of formulas. Every displayed judgment is well formed in its displayed variable context. All substitutions are simultaneous and capture-avoiding, with bound variables alpha-renamed fresh when needed. For a unary formula abstraction \(\varphi=(x)e\), \(\varphi(M)\) means \(e[M/x]\); \(\psi(M,N)\) means simultaneous replacement of the two displayed variables in \(\psi(x,y)\). A theory \(\mathcal T\) supplies arbitrary sets of well-formed equation and sequent axioms. No unused-variable deletion or reflection of formula equality to term equality is a rule. Source: \cite[Figures~1--3, 5, and~7]{Pitts1995Draft}; the explicit existential witness typing and theory-axiom rows are explained in Definition~\ref{def:p10-contextual-first-order}.}
\hbrule{rule:contextual-first-order-eq-axiom}{Eq-Ax}{}{M=N:\sigma\,[\Gamma]}{The displayed equation is an axiom of \(\mathcal T\).}
\hbrule{rule:contextual-first-order-seq-axiom}{Seq-Ax}{}{\Phi\vdash\psi\,[\Gamma]}{The displayed sequent is an axiom of \(\mathcal T\).}
\hbrule{rule:contextual-first-order-eq-refl}{Eq-Refl}{}{M=M:\sigma\,[\Gamma]}{\(M:\sigma\,[\Gamma]\).}
\hbrule{rule:contextual-first-order-eq-sym}{Eq-Sym}{M=N:\sigma\,[\Gamma]}{N=M:\sigma\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-eq-trans}{Eq-Trans}{M=N:\sigma\,[\Gamma]\qquad N=K:\sigma\,[\Gamma]}{M=K:\sigma\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-eq-subst}{Eq-Subst}{M=M':\sigma\,[\Gamma]\qquad N=N':\tau\,[\Gamma,x:\sigma,\Delta]}{N[M/x]=N'[M'/x]:\tau\,[\Gamma,\Delta]}{The declaration lists are disjoint; \(x\) is not declared in \(\Gamma,\Delta\).}
\hbrule{rule:contextual-first-order-id}{Id}{}{\Phi,\psi\vdash\psi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-cut}{Cut}{\Phi\vdash\psi\,[\Gamma]\qquad\Phi',\psi\vdash\theta\,[\Gamma]}{\Phi,\Phi'\vdash\theta\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-contract}{Contract}{\Phi,\psi,\psi\vdash\theta\,[\Gamma]}{\Phi,\psi\vdash\theta\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-exchange}{Exchange}{\Phi,\psi,\chi,\Phi'\vdash\theta\,[\Gamma]}{\Phi,\chi,\psi,\Phi'\vdash\theta\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-context-weaken}{Weaken}{\Phi\vdash\psi\,[\Gamma]}{\Phi\vdash\psi\,[\Gamma,\Delta]}{\(\Gamma,\Delta\) is a valid variable context; freshen formula binders before extension.}
\hbrule{rule:contextual-first-order-subst}{Subst}{M=M':\sigma\,[\Gamma]\qquad\Phi(x)\vdash\psi(x)\,[\Gamma,x:\sigma,\Delta]}{\Phi(M)\vdash\psi(M')\,[\Gamma,\Delta]}{The declaration lists are disjoint; \(x\) is not declared in \(\Gamma,\Delta\).}
\hbrule{rule:contextual-first-order-false-elim}{false-Elim}{\Phi\vdash\mathsf{false}\,[\Gamma]}{\Phi\vdash\psi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-and-intro}{\(\&\)-Intr}{\Phi\vdash\psi\,[\Gamma]\qquad\Phi'\vdash\chi\,[\Gamma]}{\Phi,\Phi'\vdash\psi\mathbin{\&}\chi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-and-elim1}{\(\&\)-Elim1}{\Phi\vdash\psi\mathbin{\&}\chi\,[\Gamma]}{\Phi\vdash\psi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-and-elim2}{\(\&\)-Elim2}{\Phi\vdash\psi\mathbin{\&}\chi\,[\Gamma]}{\Phi\vdash\chi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-or-intro1}{\(\vee\)-Intr1}{\Phi\vdash\psi\,[\Gamma]}{\Phi\vdash\psi\vee\chi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-or-intro2}{\(\vee\)-Intr2}{\Phi\vdash\chi\,[\Gamma]}{\Phi\vdash\psi\vee\chi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-or-elim}{\(\vee\)-Elim}{\Phi,\psi\vdash\theta\,[\Gamma]\qquad\Phi',\chi\vdash\theta\,[\Gamma]\qquad\Phi''\vdash\psi\vee\chi\,[\Gamma]}{\Phi,\Phi',\Phi''\vdash\theta\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-imp-intro}{\(\Rightarrow\)-Intr}{\Phi,\psi\vdash\chi\,[\Gamma]}{\Phi\vdash\psi\Rightarrow\chi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-imp-elim}{\(\Rightarrow\)-Elim}{\Phi\vdash\psi\Rightarrow\chi\,[\Gamma]\qquad\Phi'\vdash\psi\,[\Gamma]}{\Phi,\Phi'\vdash\chi\,[\Gamma]}{\none}
\hbrule{rule:contextual-first-order-forall-intro}{\(\forall\)-Intr}{\Phi\vdash\varphi(x)\,[\Gamma,x:\sigma]}{\Phi\vdash\forall_\sigma(\varphi)\,[\Gamma]}{\(\varphi\) is a unary formula abstraction with body \(\varphi(x)\) in \(\Gamma,x:\sigma\). The variable \(x\) is fresh for \(\Gamma\); every member of \(\Phi\) is a formula in \(\Gamma\), so none has \(x\) free.}
\hbrule{rule:contextual-first-order-forall-elim}{\(\forall\)-Elim}{\Phi\vdash\forall_\sigma(\varphi)\,[\Gamma]\qquad M:\sigma\,[\Gamma]}{\Phi\vdash\varphi(M)\,[\Gamma]}{\(\varphi\) is a unary formula abstraction whose body \(\varphi(x)\) is a formula in \(\Gamma,x:\sigma\), with \(x\) fresh for \(\Gamma\).}
\hbrule{rule:contextual-first-order-exists-intro}{\(\exists\)-Intr}{\Phi\vdash\varphi(M)\,[\Gamma]\qquad M:\sigma\,[\Gamma]}{\Phi\vdash\exists_\sigma(\varphi)\,[\Gamma]}{\(\varphi\) is a unary formula abstraction whose body \(\varphi(x)\) is a formula in \(\Gamma,x:\sigma\), with \(x\) fresh for \(\Gamma\). The typing premise states explicitly the source's implicit well-sorted instantiation condition.}
\hbrule{rule:contextual-first-order-exists-elim}{\(\exists\)-Elim}{\Phi\vdash\exists_\sigma(\varphi)\,[\Gamma]\qquad\Phi',\varphi(x)\vdash\theta\,[\Gamma,x:\sigma]}{\Phi,\Phi'\vdash\theta\,[\Gamma]}{\(\varphi\) is a unary formula abstraction with body \(\varphi(x)\) in \(\Gamma,x:\sigma\). The variable \(x\) is fresh for \(\Gamma\); \(\Phi,\Phi',\theta\) are well formed in \(\Gamma\), so none has \(x\) free.}
\hbrule{rule:contextual-first-order-equality-intro}{\(=\)-Intr}{M:\sigma\,[\Gamma]}{\Phi\vdash(M=_\sigma M)\,[\Gamma]}{\(\Phi\) is any finite hypothesis list in \(\Gamma\).}
\hbrule{rule:contextual-first-order-equality-elim}{\(=\)-Elim}{\Phi\vdash(M=_\sigma N)\,[\Gamma]\qquad\Phi'\vdash\psi(x,x)\,[\Gamma,x:\sigma]}{\Phi,\Phi'\vdash\psi(M,N)\,[\Gamma]}{\(M,N:\sigma\,[\Gamma]\); \(\psi(x,y)\) is well formed in \(\Gamma,x:\sigma,y:\sigma\), with \(x,y\) distinct and fresh for \(\Gamma\). Every member of \(\Phi'\) is well formed in \(\Gamma\).}
