\calculussheet{sheet:p02-syllogistic-sdagger}{Direct syllogistic calculus \(\mathsf S^\dagger\)}{Language: \(\mathcal S^\dagger\); \(l,m,n\) are unary literals and \(\varphi,\psi\) formulas of that fragment. Use the same atom substitutions, formula identifications and finite direct derivations as \(\mathsf S\). The semantic universe is nonempty; predicate extensions may be empty.}
\hbrule{rule:p02-syllogistic-sdagger-d}{D}{\exists(l,n)\qquad\forall(l,m)}{\exists(m,n)}{\(l,m,n\) literals.}
\hbrule{rule:p02-syllogistic-sdagger-b}{B}{\forall(l,m)\qquad\forall(m,n)}{\forall(l,n)}{\(l,m,n\) literals.}
\hbrule{rule:p02-syllogistic-sdagger-a}{A}{\forall(l,\bar l)}{\forall(l,m)}{\(l,m\) literals.}
\hbrule{rule:p02-syllogistic-sdagger-t}{T}{}{\forall(l,l)}{\(l\) a literal.}
\hbrule{rule:p02-syllogistic-sdagger-i}{I}{\exists(l,m)}{\exists(l,l)}{\(l,m\) literals.}
\hbrule{rule:p02-syllogistic-sdagger-x}{X}{\psi\qquad\bar\psi}{\varphi}{\(\varphi,\psi\in\mathcal S^\dagger\); no premise is discharged.}
\hbrule{rule:p02-syllogistic-sdagger-n}{N}{\forall(\bar l,l)}{\exists(l,l)}{\(l\) a literal; soundness requires a nonempty universe.}
