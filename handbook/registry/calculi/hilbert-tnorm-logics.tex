\phantomsection\label{sheet:hilbert-tnorm-logics}
\calculussheet{sheet:hilbert-mtl}{Monoidal T-Norm Logic \(\mathrm{MTL}\)}{Language: formulas over primitive \(\bot,\wedge,\mathbin{\&},\to\); write \(\top\defeq\bot\to\bot\), \(\neg A\defeq A\to\bot\), \(A\leftrightarrow B\defeq(A\to B)\mathbin{\&}(B\to A)\), and \(A\lor B\defeq((A\to B)\to B)\wedge((B\to A)\to A)\).  Theoremhood is \(\derives_{\mathrm{MTL}}A\); uniform substitution is allowed; equality: none.}
\noindent\emph{Controlling source:} \cite{EstevaGodo2001}.
\hbrule{rule:p08-mtl-1}{MTL1}{}{(A\to B)\to((B\to C)\to(A\to C))}{\none}
\hbrule{rule:p08-mtl-2}{MTL2}{}{(A\mathbin{\&}B)\to A}{\none}
\hbrule{rule:p08-mtl-3}{MTL3}{}{(A\mathbin{\&}B)\to(B\mathbin{\&}A)}{\none}
\hbrule{rule:p08-mtl-4a}{MTL4a}{}{(A\wedge B)\to A}{\none}
\hbrule{rule:p08-mtl-4b}{MTL4b}{}{(A\wedge B)\to(B\wedge A)}{\none}
\hbrule{rule:p08-mtl-4c}{MTL4c}{}{(A\mathbin{\&}(A\to B))\to(A\wedge B)}{\none}
\hbrule{rule:p08-mtl-5a}{MTL5a}{}{(A\to(B\to C))\to((A\mathbin{\&}B)\to C)}{\none}
\hbrule{rule:p08-mtl-5b}{MTL5b}{}{((A\mathbin{\&}B)\to C)\to(A\to(B\to C))}{\none}
\hbrule{rule:p08-mtl-6}{MTL6}{}{((A\to B)\to C)\to(((B\to A)\to C)\to C)}{\none}
\hbrule{rule:p08-mtl-7}{MTL7}{}{\bot\to A}{\none}
\hbrule{rule:p08-mtl-mp}{MP}{A\qquad A\to B}{B}{\none}

\calculussheet{sheet:hilbert-bl}{H\'ajek Basic Fuzzy Logic \(\mathrm{BL}\)}{Language and abbreviations: those of \hyperref[sheet:hilbert-mtl]{\(\mathrm{MTL}\)}.  Theoremhood is \(\derives_{\mathrm{BL}}A\).  The calculus contains every \(\mathrm{MTL}\) axiom and modus ponens above, together with the displayed divisibility axiom; equality: none.}
\noindent\emph{Controlling sources:} \cite{Hajek1998,CignoliEstevaGodoTorrens2000}.
\hbrule{rule:p08-bl-div}{Divisibility}{}{(A\wedge B)\to(A\mathbin{\&}(A\to B))}{\none}

\calculussheet{sheet:hilbert-godel-fuzzy}{Propositional G\"odel Fuzzy Logic \(G\)}{Language and abbreviations: those of \hyperref[sheet:hilbert-bl]{\(\mathrm{BL}\)}.  Theoremhood is \(\derives_G A\).  The calculus contains every \(\mathrm{BL}\) axiom and modus ponens above, together with the displayed contraction axiom; equality: none.}
\noindent\emph{Controlling source:} \cite{Hajek1998}.
\hbrule{rule:p08-godel-con}{Contraction}{}{A\to(A\mathbin{\&}A)}{\none}

\calculussheet{sheet:hilbert-product-fuzzy}{Propositional Product Fuzzy Logic \(\Pi\)}{Language and abbreviations: those of \hyperref[sheet:hilbert-bl]{\(\mathrm{BL}\)}.  Theoremhood is \(\derives_{\Pi}A\).  The calculus contains every \(\mathrm{BL}\) axiom and modus ponens above, together with both displayed Product axioms; equality: none.}
\noindent\emph{Controlling source:} \cite{Hajek1998,CintulaHajekNoguera2011}.
\hbrule{rule:p08-product-1}{\(\Pi1\)}{}{\neg\neg C\to(((A\mathbin{\&}C)\to(B\mathbin{\&}C))\to(A\to B))}{\none}
\hbrule{rule:p08-product-2}{\(\Pi2\)}{}{\neg(A\mathbin{\&}A)\to\neg A}{\none}
