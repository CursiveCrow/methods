% Source-matched signed tableau for basic hybrid K with @ and nominals.
\calculussheet{sheet:tableaux-hybrid-k}{Labelled Tableau for Basic Hybrid \(K(@)\)}{Language: propositional \(K\) with a disjoint set of nominals \(i,j,\ldots\) and satisfaction formulas \(@_iA\); a nominal names exactly one world.  Branch items are signed satisfaction formulas \(@_iA,\neg@_iA\) and accessibility atoms \(@_i\Diamond j\).  A satisfiability tableau for \(A\) starts with \(@_iA\), and a validity refutation with \(\neg@_iA\), for fresh \(i\).  A branch closes on \(@_iA,\neg@_iA\) or \(@_i\bot\).  Each occurrence is expanded at most once; equality: \(@_ij\) is propagated by uniform nominal replacement.}
\noindent\emph{Controlling source:} the \(K(@)\) fragment of the signed
hybrid tableaux in \cite{ArecesTenCate2007}.
\hbrule{rule:hybrid-tableau-neg}{Negation}{@_i\neg A}{\neg@_iA}{same branch}
\hbrule{rule:hybrid-tableau-negneg}{Negated negation}{\neg@_i\neg A}{@_iA}{same branch}
\hbrule{rule:hybrid-tableau-and}{Conjunction}{@_i(A\land B)}{@_iA\qquad @_iB}{same branch}
\hbrule{rule:hybrid-tableau-negand}{Negated conjunction}{\neg@_i(A\land B)}{\neg@_iA\mid \neg@_iB}{branch}
\hbrule{rule:hybrid-tableau-or}{Disjunction}{@_i(A\lor B)}{@_iA\mid @_iB}{branch}
\hbrule{rule:hybrid-tableau-negor}{Negated disjunction}{\neg@_i(A\lor B)}{\neg@_iA\qquad \neg@_iB}{same branch}
\hbrule{rule:hybrid-tableau-imp}{Implication}{@_i(A\to B)}{\neg@_iA\mid @_iB}{branch}
\hbrule{rule:hybrid-tableau-negimp}{Negated implication}{\neg@_i(A\to B)}{@_iA\qquad \neg@_iB}{same branch}
\hbrule{rule:hybrid-tableau-at}{Satisfaction}{@_i@_jA}{@_jA}{same branch}
\hbrule{rule:hybrid-tableau-negat}{Negated satisfaction}{\neg@_i@_jA}{\neg@_jA}{same branch}
\hbrule{rule:hybrid-tableau-diamond}{Diamond}{@_i\Diamond A}{@_i\Diamond j\qquad @_jA}{\(j\) is fresh on the branch}
\hbrule{rule:hybrid-tableau-negbox}{Negated box}{\neg@_i\Box A}{@_i\Diamond j\qquad \neg@_jA}{\(j\) is fresh on the branch}
\hbrule{rule:hybrid-tableau-box}{Box}{@_i\Box A\qquad @_i\Diamond j}{@_jA}{same branch}
\hbrule{rule:hybrid-tableau-negdiamond}{Negated diamond}{\neg@_i\Diamond A\qquad @_i\Diamond j}{\neg@_jA}{same branch}
\hbrule{rule:hybrid-tableau-ref}{Nominal reflexivity}{i\text{ occurs on the branch}}{@_ii}{add once for each occurring nominal}
\hbrule{rule:hybrid-tableau-nominal}{Nominal identity}{@_ij}{[i/j]}{replace \(j\) uniformly by \(i\) throughout the branch}
