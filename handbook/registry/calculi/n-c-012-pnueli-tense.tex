\calculussheet{sheet:n-c-012-pnueli-tense}{Pnueli's pure future-tense proof rules}{Language: propositional formulas with Boolean operations and the inclusive operators \(F,G\), with \(F A=\neg G\neg A\); judgement: theoremhood \(\vdash A\), with no open premises; equality is not an added modal primitive. Together with G1--G3 in the presentation these are the pure basis, before system-dependent axioms \cite[pp.~52--53]{Pnueli1977}.}
\hbrule{rule:n-c-012-pnueli-tense-rt}{RT}{A\text{ is a classical tautology}}{\vdash A}{\none}
\hbrule{rule:n-c-012-pnueli-tense-rg}{RG}{\vdash A}{\vdash GA}{Only theoremhood is generalized; the later initial-state axiom P is not part of this pure basis.}
\hbrule{rule:n-c-012-pnueli-tense-mp}{MP}{\vdash A\quad\vdash A\Rightarrow B}{\vdash B}{\none}
