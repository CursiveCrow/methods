\calculussheet{sheet:sol}{Relational Second-Order Calculus (Translation-Defined)}{Language: relational second-order formulas with object equality, object terms, and relation variables of every positive finite arity, but no function variables, relation terms beyond relation variables and predicate constants, or relation equality; sequent shape: finite-multiset two-sided sequents, inheriting every initial sequent and rule of the linked classical first-order equality calculus and adding the four relation-quantifier rules and the two zero-premise schema families below; derivability: the linked translation-defined relation maps antecedent support and succedent disjunction, includes the total sorted-variable assignment guard, and adds exactly the translated Comp and Choice theory; negation: primitive through the base calculus; equality: object equality only, through the base calculus.}
\noindent\emph{Base calculus:}
\hyperref[sheet:lk-eq]{classical first-order LK with equality}.
\emph{Defining derivability:}
Definition~\ref{def:p02-sol-derived-calculus}.
\emph{Standing:} this sheet is the reader-facing projection of that editorial
translation definition; Henkin's typed calculus is a controlling comparison,
not the source of these six displayed additions \cite{Henkin1950,Manzano1996}.
\hbrule{rule:sol-forall-left}{$\forall^{2}L$}{\varphi[U^n/X^n],\Gamma\Rightarrow\Delta}{\forall X^n\varphi,\Gamma\Rightarrow\Delta}{$n\geq1$; $U^n$ is an $n$-ary relation variable or predicate constant free for $X^n$ in $\varphi$; substitution is capture-avoiding}
\hbrule{rule:sol-forall-right}{$\forall^{2}R$}{\Gamma\Rightarrow\Delta,\varphi[Y^n/X^n]}{\Gamma\Rightarrow\Delta,\forall X^n\varphi}{$n\geq1$; $Y^n$ is fresh and does not occur free in any formula of $\Gamma,\Delta,\forall X^n\varphi$; substitution is capture-avoiding}
\hbrule{rule:sol-exists-left}{$\exists^{2}L$}{\varphi[Y^n/X^n],\Gamma\Rightarrow\Delta}{\exists X^n\varphi,\Gamma\Rightarrow\Delta}{$n\geq1$; $Y^n$ is fresh and does not occur free in any formula of $\Gamma,\Delta,\exists X^n\varphi$; substitution is capture-avoiding}
\hbrule{rule:sol-exists-right}{$\exists^{2}R$}{\Gamma\Rightarrow\Delta,\varphi[U^n/X^n]}{\Gamma\Rightarrow\Delta,\exists X^n\varphi}{$n\geq1$; $U^n$ is an $n$-ary relation variable or predicate constant free for $X^n$ in $\varphi$; substitution is capture-avoiding}
\hbrule{rule:sol-comprehension}{Comp}{}{\begin{gathered}\Rightarrow\operatorname{ucl}\!\Bigl(\exists X^n\,\forall x_1\cdots\forall x_n\\[-2pt]\bigl(X^n(x_1,\ldots,x_n)\leftrightarrow\chi(x_1,\ldots,x_n,\bar p,\bar Q)\bigr)\Bigr)\end{gathered}}{$n\geq1$; the $x_i$ are pairwise distinct object variables; $X^n$ is not free in $\chi$; $\bar p,\bar Q$ list all permitted free object and relation parameters; $\chi$ may itself contain relation quantifiers; $\operatorname{ucl}$ universally closes exactly the displayed parameters}
\hbrule{rule:sol-choice}{Choice}{}{\begin{gathered}\Rightarrow\operatorname{ucl}\!\Bigl(\bigl(\forall\bar x\,\exists y\,\chi(\bar x,y,\bar p,\bar Q)\bigr)\to\\[-2pt]\exists F^{n+1}\,\forall\bar x\,\exists y\,\bigl(F^{n+1}(\bar x,y)\land\chi(\bar x,y,\bar p,\bar Q)\\[-2pt]{}\land\forall z(F^{n+1}(\bar x,z)\to z=y)\bigr)\Bigr)\end{gathered}}{$n\geq1$ and $\bar x=(x_1,\ldots,x_n)$ is pairwise distinct; $F^{n+1}$ is fresh and not free in $\chi$; $\bar p,\bar Q$ list all permitted free parameters; $\operatorname{ucl}$ universally closes exactly those parameters; this is single-valued graph choice and introduces no function variable}
