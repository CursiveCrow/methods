\calculussheet{sheet:lk-defneg}{Classical Sequent Calculus \(\mathsf{LK}_{\mathrm d}\), Defined-Negation Presentation}{Language: first-order formulas over the primitives \(\bot,\land,\lor,\to,\forall,\exists\); sequent shape: two-sided sequents \(\Gamma\Rightarrow\Delta\) with finite multisets on both sides (exchange built into the multiset representation; weakening and contraction are not), read as \(\bigwedge\Gamma\to\bigvee\Delta\) with \(\bigwedge\varnothing=\top\) and \(\bigvee\varnothing=\bot\); initial sequents are uncontexted (\(\varphi\Rightarrow\varphi\) and \(\bot,\Gamma\Rightarrow\Delta\)); negation: defined, \(\neg\varphi\defeq\varphi\to\bot\), with no primitive negation rules; equality: none.}
\hbrule{rule:lk-defneg-id}{Id}{}{\varphi\Rightarrow\varphi}{\none}
\hbrule{rule:lk-defneg-bot-l}{\(\bot L\)}{}{\bot,\Gamma\Rightarrow\Delta}{\none}
\hbrule{rule:lk-defneg-wl}{WL}{\Gamma\Rightarrow\Delta}{\varphi,\Gamma\Rightarrow\Delta}{\none}
\hbrule{rule:lk-defneg-wr}{WR}{\Gamma\Rightarrow\Delta}{\Gamma\Rightarrow\Delta,\varphi}{\none}
\hbrule{rule:lk-defneg-cl}{CL}{\varphi,\varphi,\Gamma\Rightarrow\Delta}{\varphi,\Gamma\Rightarrow\Delta}{\none}
\hbrule{rule:lk-defneg-cr}{CR}{\Gamma\Rightarrow\Delta,\varphi,\varphi}{\Gamma\Rightarrow\Delta,\varphi}{\none}
\hbrule{rule:lk-defneg-cut}{Cut}{\Gamma\Rightarrow\Delta,\varphi\qquad\varphi,\Pi\Rightarrow\Sigma}{\Gamma,\Pi\Rightarrow\Delta,\Sigma}{the displayed occurrence of \(\varphi\) is the cut formula}
\hbrule{rule:lk-defneg-mix}{Mix}{\Gamma\Rightarrow\Delta,\varphi^{m}\qquad\varphi^{n},\Pi\Rightarrow\Sigma}{\Gamma,\Pi\Rightarrow\Delta,\Sigma}{\(m,n\geq1\); \(\varphi^{k}\) abbreviates exactly \(k\) comma-separated occurrences of the mix formula \(\varphi\); \(\Delta\) and \(\Sigma\) may contain further occurrences of \(\varphi\)}
\hbrule{rule:lk-defneg-and-l}{\(\land L\)}{\varphi,\psi,\Gamma\Rightarrow\Delta}{\varphi\land\psi,\Gamma\Rightarrow\Delta}{\none}
\hbrule{rule:lk-defneg-and-r}{\(\land R\)}{\Gamma\Rightarrow\Delta,\varphi\qquad\Gamma\Rightarrow\Delta,\psi}{\Gamma\Rightarrow\Delta,\varphi\land\psi}{\none}
\hbrule{rule:lk-defneg-or-l}{\(\lor L\)}{\varphi,\Gamma\Rightarrow\Delta\qquad\psi,\Gamma\Rightarrow\Delta}{\varphi\lor\psi,\Gamma\Rightarrow\Delta}{\none}
\hbrule{rule:lk-defneg-or-r}{\(\lor R\)}{\Gamma\Rightarrow\Delta,\varphi,\psi}{\Gamma\Rightarrow\Delta,\varphi\lor\psi}{\none}
\hbrule{rule:lk-defneg-imp-l}{\(\to L\)}{\Gamma\Rightarrow\Delta,\varphi\qquad\psi,\Pi\Rightarrow\Sigma}{\varphi\to\psi,\Gamma,\Pi\Rightarrow\Delta,\Sigma}{\none}
\hbrule{rule:lk-defneg-imp-r}{\(\to R\)}{\varphi,\Gamma\Rightarrow\Delta,\psi}{\Gamma\Rightarrow\Delta,\varphi\to\psi}{\none}
\hbrule{rule:lk-defneg-forall-l}{\(\forall L\)}{\varphi[t/x],\Gamma\Rightarrow\Delta}{\forall x\,\varphi,\Gamma\Rightarrow\Delta}{substitution: every displayed instance \(\varphi[t/x]\) uses the total capture-avoiding substitution fixed in the first-order syntax chapter (bound variables renamed when required), so there is no additional definedness or substitutability side condition}
\hbrule{rule:lk-defneg-forall-r}{\(\forall R\)}{\Gamma\Rightarrow\Delta,\varphi[y/x]}{\Gamma\Rightarrow\Delta,\forall x\,\varphi}{eigenvariable freshness: \(y\) must not occur free in the conclusion sequent; this is a rule condition, not a convention that may be repaired after the inference; substitution total and capture avoiding}
\hbrule{rule:lk-defneg-exists-l}{\(\exists L\)}{\varphi[y/x],\Gamma\Rightarrow\Delta}{\exists x\,\varphi,\Gamma\Rightarrow\Delta}{eigenvariable freshness: \(y\) must not occur free in the conclusion sequent; this is a rule condition, not a convention that may be repaired after the inference; substitution total and capture avoiding}
\hbrule{rule:lk-defneg-exists-r}{\(\exists R\)}{\Gamma\Rightarrow\Delta,\varphi[t/x]}{\Gamma\Rightarrow\Delta,\exists x\,\varphi}{substitution: total capture-avoiding substitution, as in \(\forall L\)}
