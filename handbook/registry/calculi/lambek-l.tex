% NEW for Phase 4, P8.6.  Exact associative unit-free Lambek calculus L,
% controlled against Lambek (1958) and Buszkowski (2010).  L* (empty
% antecedents) and L_1 (a multiplicative unit) are deliberately excluded.
\calculussheet{sheet:lambek-l}{Associative Lambek Calculus $\mathsf L$}{Language: the least formula set containing the atoms and closed under product $A\mathbin{\cdot}B$, left division $A\backslash B$, and right division $B/A$; sequent shape: $\Gamma\Rightarrow A$, with $\Gamma$ a finite nonempty ordered sequence and exactly one succedent; exchange, weakening, and contraction: absent; empty antecedents and a multiplicative unit: absent; negation and equality: none.  Context metavariables in a rule may be empty only when every displayed sequent still has a nonempty total antecedent.}
\hbrule{rule:lambek-l-id}{Id}{}{A\Rightarrow A}{\none}
\hbrule{rule:lambek-l-cut}{Cut}{\Gamma,A,\Gamma'\Rightarrow B\qquad\Delta\Rightarrow A}{\Gamma,\Delta,\Gamma'\Rightarrow B}{every displayed antecedent is nonempty}
\hbrule{rule:lambek-l-product-l}{$\cdot$L}{\Gamma,A,B,\Delta\Rightarrow C}{\Gamma,A\mathbin{\cdot}B,\Delta\Rightarrow C}{every displayed antecedent is nonempty}
\hbrule{rule:lambek-l-product-r}{$\cdot$R}{\Gamma\Rightarrow A\qquad\Delta\Rightarrow B}{\Gamma,\Delta\Rightarrow A\mathbin{\cdot}B}{$\Gamma$ and $\Delta$ are nonempty}
\hbrule{rule:lambek-l-backslash-l}{$\backslash$L}{\Gamma,B,\Delta\Rightarrow C\qquad\Phi\Rightarrow A}{\Gamma,\Phi,A\backslash B,\Delta\Rightarrow C}{$\Phi$ is nonempty and every displayed antecedent is nonempty}
\hbrule{rule:lambek-l-backslash-r}{$\backslash$R}{A,\Gamma\Rightarrow B}{\Gamma\Rightarrow A\backslash B}{$\Gamma$ is nonempty}
\hbrule{rule:lambek-l-slash-l}{/L}{\Gamma,A,\Delta\Rightarrow C\qquad\Phi\Rightarrow B}{\Gamma,A/B,\Phi,\Delta\Rightarrow C}{$\Phi$ is nonempty and every displayed antecedent is nonempty}
\hbrule{rule:lambek-l-slash-r}{/R}{\Gamma,B\Rightarrow A}{\Gamma\Rightarrow A/B}{$\Gamma$ is nonempty}
