% Editorial SLD/SLDNF operational sheet for Part IX.
\calculussheet{sheet:sld-resolution}{SLD Resolution and Ground SLDNF Steps}{Language: non-equality predicate atoms in the definite and normal clauses and goals of Definitions~\ref{def:p09-sld} and~\ref{def:p09-normal-program}; shape: ordered goals with a selected literal, substitutions applied to the entire resolvent, and branch outcomes success, finite failure, divergence, or floundering; negation: \(\mathsf{not}\) is ground finite failure, not classical negation; equality: the displayed mgu is metalinguistic unification in the finite term algebra with the occurs check, not an object-language equality predicate.}
\noindent\emph{Controlling sources:} \cite{Lloyd1987,Clark1978}.
\hbrule{rule:p09-sld-positive}{Positive SLD}{\leftarrow A_1,\ldots,A_i,\ldots,A_n\qquad H\leftarrow B_1,\ldots,B_k\qquad\theta=\operatorname{mgu}(A_i,H)}{\leftarrow(A_1,\ldots,A_{i-1},B_1,\ldots,B_k,A_{i+1},\ldots,A_n)\theta}{the clause is standardized apart from the current derivation; if no clause head unifies with the selected positive atom, that branch fails}
\hbrule{rule:p09-sld-success}{SLD success}{G_0\xRightarrow{\theta_1}\cdots\xRightarrow{\theta_m}\leftarrow}{(\theta_1\cdots\theta_m)|_{\operatorname{Var}(G_0)}}{the conclusion is the computed answer substitution, restricted to variables of the initial goal}
\hbrule{rule:p09-sldnf-negative-success}{Ground negative success}{\leftarrow L_1,\ldots,\mathsf{not}\ A,\ldots,L_n\qquad\text{the complete subsidiary SLDNF tree for }\leftarrow A\text{ is finite and failed}}{\leftarrow L_1,\ldots,L_n}{\(A\) is ground}
\hbrule{rule:p09-sldnf-negative-failure}{Ground negative failure}{\leftarrow L_1,\ldots,\mathsf{not}\ A,\ldots,L_n\qquad\leftarrow A\text{ has an SLDNF refutation}}{\text{failed branch}}{\(A\) is ground}
\hbrule{rule:p09-sldnf-flounder}{Flounder}{\leftarrow L_1,\ldots,\mathsf{not}\ A,\ldots,L_n}{\text{floundered branch}}{the selected negative atom \(A\) is nonground; neither success nor failure is inferred}

Every program and goal atom on this sheet excludes logical equality.  The mgu
calculation uses the correct occurs check on finite first-order terms; a Prolog
engine that deliberately omits it has a different operational term semantics.
