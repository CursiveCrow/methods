\calculussheet{sheet:p07-combinatory-abstraction}{Auxiliary relations for S--K abstraction}{Terms are the binder-free S--K terms of Definition~\ref{def:p07-combinatory-syntax}. Weak contraction remains exactly Sheet~\ref{sheet:sk-combinatory}. The parallel relation below is auxiliary to its confluence proof. Extensional equality is the least equivalence relation closed under application, containing the two weak contraction equations and the final fresh-variable rule; it is a separate theory.}
\hbrule{rule:p07-combinatory-par-atom}{Parallel atom}{}{a\Rightarrow_{\mathsf{SK}}a}{$a$ is a variable or one of $\mathsf S,\mathsf K$}
\hbrule{rule:p07-combinatory-par-app}{Parallel application}{P\Rightarrow_{\mathsf{SK}}P'\qquad Q\Rightarrow_{\mathsf{SK}}Q'}{PQ\Rightarrow_{\mathsf{SK}}P'Q'}{All four expressions are S--K terms}
\hbrule{rule:p07-combinatory-par-k}{Parallel K}{P\Rightarrow_{\mathsf{SK}}P'\qquad Q\Rightarrow_{\mathsf{SK}}Q'}{\mathsf K P Q\Rightarrow_{\mathsf{SK}}P'}{Application associates to the left; the premise for the discarded argument is allowed}
\hbrule{rule:p07-combinatory-par-s}{Parallel S}{P\Rightarrow_{\mathsf{SK}}P'\qquad Q\Rightarrow_{\mathsf{SK}}Q'\qquad R\Rightarrow_{\mathsf{SK}}R'}{\mathsf S P Q R\Rightarrow_{\mathsf{SK}}P'R'(Q'R')}{The same reduct $R'$ is used in both copies}
\hbrule{rule:p07-combinatory-ext}{Extensionality}{Pz=_{\mathrm e}Qz}{P=_{\mathrm e}Q}{$z\notin\FV(P)\cup\FV(Q)$; this is an equality rule, not a weak reduction rule}
