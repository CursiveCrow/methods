\calculussheet{sheet:hilbert-e}{Non-normal Modal Hilbert Calculus $E$}{Language: modal propositional formulas --- the classical propositional grammar ($\bot,\neg,\land,\lor,\to,\leftrightarrow$) extended by $\Box A$, with $\Diamond A\defeq\neg\Box\neg A$ defined; sequent shape: no sequents --- for a premise set $\Gamma$, $\Gamma\derives_{E}A$ means that a finite sequence ending in $A$ has each line either in $\Gamma$, a Taut instance, an MP consequence of earlier lines, or an RE consequence of an equivalence having a separate derivation from no premises; RE is theorem-only; negation: primitive (classical); equality: none.}
\hbrule{rule:p08-hilbert-e-taut}{Taut}{}{A}{$A$ is a uniform substitution instance of a classically valid propositional formula}
\hbrule{rule:p08-hilbert-e-mp}{MP}{A\to B\qquad A}{B}{\none}
\hbrule{rule:p08-hilbert-e-re}{RE}{\derives_{E}A\leftrightarrow B}{\derives_{E}\Box A\leftrightarrow\Box B}{theorem-only: the displayed equivalence has a derivation from no premises}
