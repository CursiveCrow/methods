\calculussheet{sheet:nd-fol}{Natural Deduction for Classical First-Order Logic ($\mathsf{ND}_{\mathsf{FOL}}$)}{Language: $\mathsf{Fm}(\mathcal{L})$, first-order with primitives $\bot,\neg,\land,\lor,\to,\forall,\exists$ and primitive equality $=$; sequent shape: single-conclusion derivability judgments $\Gamma\vdash\varphi$ with $\Gamma$ a finite set of open assumptions (tree form: finite rooted trees with discharge labels drawn from $\mathbb{N}$); negation: primitive, with its own rules alongside primitive $\bot$, and RAA supplying classical strength; equality: primitive logical identity governed by $=I$ and $=E$.}
\hbrule{rule:nd-fol-assumption}{assumption}{}{\Gamma\cup\{\varphi\}\vdash\varphi}{\none}
\hbrule{rule:nd-fol-and-i}{$\land I$}{\Gamma\vdash\varphi\qquad\Delta\vdash\psi}{\Gamma\cup\Delta\vdash\varphi\land\psi}{\none}
\hbrule{rule:nd-fol-and-e1}{$\land E_1$}{\Gamma\vdash\varphi\land\psi}{\Gamma\vdash\varphi}{\none}
\hbrule{rule:nd-fol-and-e2}{$\land E_2$}{\Gamma\vdash\varphi\land\psi}{\Gamma\vdash\psi}{\none}
\hbrule{rule:nd-fol-or-i1}{$\lor I_1$}{\Gamma\vdash\varphi}{\Gamma\vdash\varphi\lor\psi}{\none}
\hbrule{rule:nd-fol-or-i2}{$\lor I_2$}{\Gamma\vdash\psi}{\Gamma\vdash\varphi\lor\psi}{\none}
\hbrule{rule:nd-fol-or-e}{$\lor E$}{\Gamma\vdash\varphi\lor\psi\qquad\Delta\cup\{\varphi\}\vdash\chi\qquad\Sigma\cup\{\psi\}\vdash\chi}{\Gamma\cup\Delta\cup\Sigma\vdash\chi}{discharge: the application carries two labels $i,j$; label $i$ marks at least one open occurrence of $\varphi$ above the second premise and label $j$ at least one open occurrence of $\psi$ above the third, and the application discharges exactly all such marked occurrences}
\hbrule{rule:nd-fol-imp-i}{$\to I$}{\Gamma\cup\{\varphi\}\vdash\psi}{\Gamma\vdash\varphi\to\psi}{discharge: the application carries a label $i$ marking at least one open occurrence of $\varphi$ above its premise and discharges exactly all such marked occurrences; vacuous discharge is not licensed}
\hbrule{rule:nd-fol-imp-e}{$\to E$}{\Gamma\vdash\varphi\to\psi\qquad\Delta\vdash\varphi}{\Gamma\cup\Delta\vdash\psi}{\none}
\hbrule{rule:nd-fol-neg-i}{$\neg I$}{\Gamma\cup\{\varphi\}\vdash\bot}{\Gamma\vdash\neg\varphi}{discharge: the application carries a label $i$ marking at least one open occurrence of $\varphi$ above its premise and discharges exactly all such marked occurrences; vacuous discharge is not licensed}
\hbrule{rule:nd-fol-neg-e}{$\neg E$}{\Gamma\vdash\varphi\qquad\Delta\vdash\neg\varphi}{\Gamma\cup\Delta\vdash\bot}{\none}
\hbrule{rule:nd-fol-bot-e}{$\bot E$}{\Gamma\vdash\bot}{\Gamma\vdash\varphi}{\none}
\hbrule{rule:nd-fol-raa}{RAA}{\Gamma\cup\{\neg\varphi\}\vdash\bot}{\Gamma\vdash\varphi}{discharge: the application carries a label $i$ marking at least one open occurrence of $\neg\varphi$ above its premise and discharges exactly all such marked occurrences; vacuous discharge is not licensed; this is the explicitly classical rule}
\hbrule{rule:nd-fol-forall-e}{$\forall E$}{\Gamma\vdash\forall x\,\varphi}{\Gamma\vdash\varphi[t/x]}{substitution: $[t/x]$ is the total capture-avoiding substitution fixed in the syntax chapter (bound variables are renamed as required), so no separate substitutability proviso is imposed}
\hbrule{rule:nd-fol-exists-i}{$\exists I$}{\Gamma\vdash\varphi[t/x]}{\Gamma\vdash\exists x\,\varphi}{substitution: $[t/x]$ is the total capture-avoiding substitution, as in $\forall E$}
\hbrule{rule:nd-fol-forall-i}{$\forall I$}{\Gamma\vdash\varphi}{\Gamma\vdash\forall x\,\varphi}{eigenvariable freshness: the generalized variable $x$ satisfies $x\notin\mathrm{FV}(\Gamma)$, i.e.\ $x$ is not free in any open assumption}
\hbrule{rule:nd-fol-exists-e}{$\exists E$}{\Gamma\vdash\exists x\,\varphi\qquad\Delta\cup\{\varphi[y/x]\}\vdash\psi}{\Gamma\cup\Delta\vdash\psi}{eigenvariable freshness: $y$ is absent from $\psi$, from every other open assumption (every formula of $\Gamma\cup\Delta$), and from $\exists x\,\varphi$; discharge: the application carries a label $i$ marking at least one open occurrence of $\varphi[y/x]$ above the minor premise and discharges exactly all such marked occurrences}
\hbrule{rule:nd-fol-alpha}{$\alpha$}{\Gamma\vdash\varphi}{\Gamma\vdash\psi}{$\varphi=_{\alpha}\psi$, the syntactic alpha-equivalence relation of the syntax chapter: the rule changes only bound-variable names, and it is neither identity elimination nor replacement by an arbitrary logically equivalent formula}
\hbrule{rule:nd-fol-eq-i}{$=I$}{}{\vdash t=t}{\none}
\hbrule{rule:nd-fol-eq-e}{$=E$}{\Gamma\vdash r=t\qquad\Delta\vdash\varphi[r/x]}{\Gamma\cup\Delta\vdash\varphi[t/x]}{substitution: both displayed instances use the total capture-avoiding substitution; $\varphi$ may contain zero or more free occurrences of $x$; symmetry and transitivity of $=$ are derivable, not assumed}
