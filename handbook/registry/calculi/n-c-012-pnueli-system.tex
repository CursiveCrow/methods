\calculussheet{sheet:n-c-012-pnueli-system}{Pnueli's system-dependent temporal rules}{Language: state assertions over a transition system \(\langle S,s_0,R\rangle\), with equality in the underlying assertion language; the displayed arrows are metalinguistic inferences. Judgements are system-specific \(\vdash A\), with inclusive \(F,G\). The actions form the finite partition of \(R\) defined in the presentation \cite[p.~53]{Pnueli1977}.}
\hbrule{rule:n-c-012-pnueli-system-i1}{I1}{p(s_0)\quad
       \forall s,s'\,
       (p(s)\land R(s,s')\Rightarrow p(s'))}{\vdash Gp}{\none}
\hbrule{rule:n-c-012-pnueli-system-i2}{I2}{\forall s\,(p(s)\Rightarrow q(s))\quad
       \forall s,s'\,
       (q(s)\land R(s,s')\Rightarrow q(s'))}{\vdash p\Rightarrow Gq}{\none}
\hbrule{rule:n-c-012-pnueli-system-e}{E}{\forall s,s'\,
       (p(s)\land R(s,s')\Rightarrow q(s'))}{\vdash p\Rightarrow Fq}{Executions are infinite transition paths.}
\hbrule{rule:n-c-012-pnueli-system-n}{N}{\begin{gathered}
       \forall s,s'\,
       (p(s)\land R(s,s')\land\neg a(s,s')\Rightarrow p(s'))\\
       \forall s,s'\,
       (p(s)\land a(s,s')\Rightarrow q(s'))
       \end{gathered}}{\vdash p\Rightarrow Fq}{For every allowed execution and every position, a designated \(a\)-transition occurs at some later or current transition position.}
