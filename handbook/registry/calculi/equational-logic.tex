%% Finite equational deductions under universally interpreted identities.
\calculussheet{sheet:equational-logic}{Equational logic}{Language: terms over a fixed finitary signature and variable set \(V\) with nonempty term algebra; judgments \(\Gamma\vdash_{\mathrm{eq}}s\approx t\), where \(\Gamma\) is a set of universally interpreted identities; every derivation is finite. The rules give the reflexive, symmetric, transitive, compatible, substitution-closed consequence relation. Compatibility and the source's contextual replacement rule are interderivable as proved in Annex~\ref{annex:p10.eq.2}; source: \cite[II\S14, Definitions~14.16, 14.18]{BurrisSankappanavar2012}.}
\hbrule{rule:equational-logic-premise}{premise}{}{\Gamma\vdash_{\mathrm{eq}}s\approx t}{\(s\approx t\in\Gamma\).}
\hbrule{rule:equational-logic-reflexivity}{reflexivity}{}{\Gamma\vdash_{\mathrm{eq}}t\approx t}{\(t\in T_\tau(V)\).}
\hbrule{rule:equational-logic-symmetry}{symmetry}{\Gamma\vdash_{\mathrm{eq}}s\approx t}{\Gamma\vdash_{\mathrm{eq}}t\approx s}{\none}
\hbrule{rule:equational-logic-transitivity}{transitivity}{\Gamma\vdash_{\mathrm{eq}}s\approx t\qquad\Gamma\vdash_{\mathrm{eq}}t\approx u}{\Gamma\vdash_{\mathrm{eq}}s\approx u}{\none}
\hbrule{rule:equational-logic-compatibility}{compatibility}{\Gamma\vdash_{\mathrm{eq}}s_i\approx t_i\quad(1\leq i\leq n)}{\Gamma\vdash_{\mathrm{eq}}f(s_1,\ldots,s_n)\approx f(t_1,\ldots,t_n)}{\(f\) is \(n\)-ary and \(n\) is finite; for \(n=0\) the premise list is empty and the conclusion is \(\Gamma\vdash_{\mathrm{eq}}f\approx f\).}
\hbrule{rule:equational-logic-substitution}{substitution}{\Gamma\vdash_{\mathrm{eq}}s\approx t}{\Gamma\vdash_{\mathrm{eq}}s\sigma\approx t\sigma}{\(\sigma:V\to T_\tau(V)\) is any simultaneous substitution.}
