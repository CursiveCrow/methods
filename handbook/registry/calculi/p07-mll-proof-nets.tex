\calculussheet{sheet:p07-mll-sequents}{Unit-free MLL and marked auxiliary extensions}{Paired atoms and tensor/par, no units; ordered occurrence roles with finite-multiset sequents; general-formula dual axiom; connected closed subsystem has neither Mix nor marked hypotheses.}
\hbrule{rule:p07-mll-ax}{Ax}{}{\vdash A,A^\perp}{Two distinct occurrences; \(A\) may be composite.}
\hbrule{rule:p07-mll-tensor}{Tensor}{\vdash\Gamma,A\qquad\vdash\Delta,B}{\vdash\Gamma,\Delta,A\otimes B}{Disjoint premise occurrence sets.}
\hbrule{rule:p07-mll-par}{Par}{\vdash\Gamma,A,B}{\vdash\Gamma,A\bindnasrepma B}{The two displayed occurrences are consumed once.}
\hbrule{rule:p07-mll-cut}{Cut}{\vdash\Gamma,A\qquad\vdash\Delta,A^\perp}{\vdash\Gamma,\Delta}{Disjoint premise occurrence sets; this rule adds a cut vertex and is not part of the cut-free subsystem.}
\hbrule{rule:p07-mll-exchange}{Exchange}{\vdash\Gamma}{\vdash\rho(\Gamma)}{\(\rho\) permutes occurrences, not their identities.}
\hbrule{rule:p07-mll-hyp}{Marked hypothesis}{}{\vdash A}{Auxiliary open leaf only; \(A\) is an assumption, not a theorem.}
\hbrule{rule:p07-mll-mix-zero}{Auxiliary Mix zero}{}{\vdash}{Auxiliary system only; graph is empty.}
\hbrule{rule:p07-mll-mix-two}{Auxiliary binary Mix}{\vdash\Gamma\qquad\vdash\Delta}{\vdash\Gamma,\Delta}{Disjoint union; absent from the connected closed subsystem.}

\calculussheet{sheet:p07-mll-trips}{Girard 1987 two-port routing}{Every occurrence \(F\) has exit states \(F^+,F^-\), not polarities or linear negations; settings \(L,R\) are fixed for each tensor AND par during a whole trip; successor clauses follow printed pp.31--32.}
\hbrule{rule:p07-mll-trip-ax}{Axiom routing}{\mathsf{ax}(A,B),\ B=A^\perp}{A^+\mapsto B^-\ ;\ B^+\mapsto A^-}{Distinct occurrence ports; axiom has two outputs.}
\hbrule{rule:p07-mll-trip-terminal}{Terminal routing}{F\text{ has no consumer}}{F^-\mapsto F^+}{The producing link still governs \(F^+\).}
\hbrule{rule:p07-mll-trip-tensor-left}{Tensor L}{C=A\otimes B,\ \sigma(C)=L}{C^+\mapsto B^+\ ;\ B^-\mapsto A^+\ ;\ A^-\mapsto C^-}{The occurrence \(C\) is produced by this tensor link.}
\hbrule{rule:p07-mll-trip-tensor-right}{Tensor R}{C=A\otimes B,\ \sigma(C)=R}{C^+\mapsto A^+\ ;\ A^-\mapsto B^+\ ;\ B^-\mapsto C^-}{The same three occurrence identities are retained.}
\hbrule{rule:p07-mll-trip-par-left}{Par L}{C=A\bindnasrepma B,\ \sigma(C)=L}{C^+\mapsto A^+\ ;\ A^-\mapsto C^-\ ;\ B^-\mapsto B^+}{The switch remains fixed for the whole orbit.}
\hbrule{rule:p07-mll-trip-par-right}{Par R}{C=A\bindnasrepma B,\ \sigma(C)=R}{C^+\mapsto B^+\ ;\ B^-\mapsto C^-\ ;\ A^-\mapsto A^+}{This is original trip routing, not deletion of a tensor premise.}

\calculussheet{sheet:p07-mll-cut}{Closed unit-free net contractions}{\(N[\ldots]\) denotes the specified incidence subgraph and its retained outside ports, not a textual substitution context; all axiom/tensor/par outputs have at most one consumer; cut is a symmetric two-premise vertex.}
\hbrule{rule:p07-mll-cut-principal}{Tensor/par cut}{N[\mathsf{cut}(\mathsf{ten}(A,B),\mathsf{par}(A^\perp,B^\perp))]}{N[\mathsf{cut}(A,A^\perp),\mathsf{cut}(B,B^\perp)]}{Delete the two connective vertices, their outgoing arcs and the old cut; attach each new cut to the corresponding original premise occurrences.}
\hbrule{rule:p07-mll-cut-axiom}{Axiom cut}{N[\mathsf{ax}(A,X),\mathsf{cut}(A,Y)],\ X,Y:A^\perp}{N[X=Y]}{Delete the axiom and cut and their cut-facing arc; identify the companion arc with the opposite premise arc, retaining the latter's producer and the former's consumer/boundary. Equality here means arc gluing, not formula equality; the same-axiom loop is excluded by correctness.}
