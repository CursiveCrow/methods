\section{The constructible inner model and the positive consistency argument}
\label{sec:p06-constructible-inner-model-completion}

Work in \(\mathsf{ZF}\) with the formula-defined hierarchy of
\cref{thm:p06-hierarchy-recursion-existence}.
Class notation abbreviates formulas. No truth predicate for the
universe or for the proper class \(L\) is introduced.
The reflection statements below are schemas for fixed standard formulas.

\begin{hbtheorem}{thm:p06-collapse-complete}{proved=annex:p06-collapse-complete}{Well-founded extensional collapse}
For a well-founded relation \(R\) on a set \(X\), there is exactly one
function \(\pi\) on \(X\) satisfying
\[
 \pi(x)=\{\pi(y):yRx\}.
\]
If \(R\) is extensional, this function is injective and is an
isomorphism onto a transitive set. The transitive range and
the isomorphism are unique.
\end{hbtheorem}

\begin{hblemma}{thm:p06-finite-assignment-def}{proved=annex:p06-finite-assignment-def}{Definability from finite assignments}
The set-sized instruction evaluation of
\cref{thm:p06-set-sized-definability}
has an equivalent finite-assignment form.
For a code \(c\), choose \(m\geq1\) larger than every variable
index occurring in \(c\). Evaluation uses tuples in \(A^m\)
and their single-coordinate updates.
Its truth depends only on those coordinates.
All definable subsets of a nonempty \(A\) are obtained by
varying \(c,m\), and the finite parameter tuple, and evaluating
after replacing coordinate zero by the candidate element.
The complete finite-assignment satisfaction relation and the
complete family of these subsets exist uniformly as sets.
\end{hblemma}

\begin{hbtheorem}{thm:p06-hierarchy-finite-reflection}{proved=annex:p06-hierarchy-finite-reflection}{Finite reflection for the two hierarchies}
Let \(H_\alpha\) denote either \(V_\alpha\) or \(L_\alpha\),
and let \(H\) denote the corresponding formula-defined class.
For any fixed finite family \(\Phi\) of standard membership formulas
and any ordinal \(\beta\), there is a limit ordinal \(\delta>\beta\)
such that
\[
 \varphi^H(\bar a)
 \quad\Longleftrightarrow\quad
 (H_\delta,\in)\models\varphi(\bar a)
\]
for every \(\varphi\in\Phi\) and tuple \(\bar a\) from \(H_\delta\).
For \(H=V\), the left side is simply the ambient formula.
For \(H=L\), it is its syntactic relativization to the defining
formula of \(L\).
This is one theorem for each fixed finite family, not a uniform
class satisfaction relation.
\end{hbtheorem}

\begin{hbtheorem}{thm:p06-inner-l-zf}{proved=annex:p06-inner-l-zf}{Every ZF axiom holds in the constructible class}
For each axiom \(\sigma\) of \(\mathsf{ZF}\), including each
Separation and Replacement instance,
\(\mathsf{ZF}\) proves \(\sigma^L\).
The power set supplied internally is
\(\{y\in L:y\subseteq x\}\), not the ambient power set of \(x\).
\end{hbtheorem}

\begin{hbtheorem}{thm:p06-l-self-construction}{proved=annex:p06-l-self-construction}{The hierarchy computed within the inner model}
The constructible hierarchy computed inside \(L\) has the same
levels \(L_\alpha\) as the ambient construction.
Consequently \(\mathsf{ZF}\) proves \((V=L)^L\).
There is a parameter-free definable well-order of the
constructible class whose restriction to any constructible set
is a set well-order. In particular
\(\mathsf{ZF}\) proves every axiom of
\(\mathsf{ZFC}+(V=L)\) relativized to \(L\).
\end{hbtheorem}

\begin{hbdefinition}{def:p06-condensation-package}{A fixed finite construction package}
Let \(\Theta\) consist of the following fixed first-order sentences,
using the formula abbreviations already defined:
\begin{enumerate}
 \item Pairing and Union, and existence of the least inductive set
       as an ordinal;
 \item the uniform finite-assignment satisfaction construction of
       \cref{thm:p06-finite-assignment-def}, with its Tarskian clauses;
 \item existence of the complete \(\operatorname{Def}(A)\) family
       for every \(A\), including an output subset for every
       finite code and parameter tuple in the preceding lemma;
 \item existence, for every ordinal, of the constructible hierarchy
       sequence through that ordinal, with its stated recursion clauses;
 \item \(V=L\).
\end{enumerate}
Each construction clause is a single first-order assertion about
set codes and set functions, not a scheme asserting satisfaction
for the universe. Apart from \(V=L\), these are the uniform
\(\mathsf{ZF}\) constructions proved above and in
\cref{thm:p06-hierarchy-recursion-existence}.
Thus \(\Theta\) is a fixed finite family of sentences.
\end{hbdefinition}

\begin{hblemma}{thm:p06-adequate-condensation}{proved=annex:p06-adequate-condensation}{Condensation at reflecting stages}
Suppose \(\alpha>\omega\) is a limit ordinal,
\(L_\alpha\models\Theta\), and
\(X\preccurlyeq(L_\alpha,\in)\).
Then its Mostowski collapse is \(L_\gamma\) for a limit
ordinal \(\gamma\leq\alpha\).
Such stages \(\alpha\) occur arbitrarily high in the
constructible class.
This is the explicit finite-package form used in the following
cardinal argument.
\end{hblemma}

\begin{hbtheorem}{thm:p06-gch-inner-l}{proved=annex:p06-gch-inner-l}{GCH in the constructible class}
The theory \(\mathsf{ZF}+(V=L)\) proves
\(2^\kappa=\kappa^+\) for every infinite cardinal \(\kappa\).
Consequently \(\mathsf{ZF}\) proves every axiom of
\(\mathsf{ZFC}+\mathsf{GCH}+(V=L)\) relativized to \(L\).
\end{hbtheorem}

The axiom-by-axiom conclusion supplies the relativization scheme
used in \cref{thm:ch-relative-consistency}.
It proves the ordinary positive relative-consistency implication
through that theorem's finite derivation argument.
The separate arithmetization of a chosen proof calculus requires
its own effective proof-translation verification; it is not inferred
merely from the existence of an inner model.
