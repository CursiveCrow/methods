% NEW for Phase 4.  Exact theoremhood system from Emerson--Halpern 1985,
% Section 6; its transition frames are total, so every path is infinite.
\calculussheet{sheet:hilbert-ctl}{Hilbert Calculus for Computation Tree Logic}{Language: CTL state formulas over $\bot,\neg,\land,\lor,\to$ with $\mathsf{EX}A$, $\mathsf{AX}A$, $\mathsf{EF}A$, $\mathsf{AF}A$, $\mathsf E(A\mathbin{\mathsf U}B)$, and $\mathsf A(A\mathbin{\mathsf U}B)$; sequent shape: no sequents --- theoremhood $\derives_{\mathsf{CTL}}A$ by finite no-premise derivations; negation: primitive (classical); equality: none.}
\hbrule{rule:p08-hilbert-ctl-taut}{Taut}{}{A}{$A$ is a uniform substitution instance of a classically valid propositional formula}
\hbrule{rule:p08-hilbert-ctl-ef}{EF}{}{\mathsf{EF}A\leftrightarrow\mathsf E(\topv\mathbin{\mathsf U}A)}{\none}
\hbrule{rule:p08-hilbert-ctl-af}{AF}{}{\mathsf{AF}A\leftrightarrow\mathsf A(\topv\mathbin{\mathsf U}A)}{\none}
\hbrule{rule:p08-hilbert-ctl-ex-or}{EX$\lor$}{}{\mathsf{EX}(A\lor B)\leftrightarrow(\mathsf{EX}A\lor\mathsf{EX}B)}{\none}
\hbrule{rule:p08-hilbert-ctl-ax-dual}{AX-Dual}{}{\mathsf{AX}A\leftrightarrow\neg\mathsf{EX}\neg A}{\none}
\hbrule{rule:p08-hilbert-ctl-eu-unfold}{E$\mathsf U$}{}{\mathsf E(A\mathbin{\mathsf U}B)\leftrightarrow\bigl(B\lor(A\land\mathsf{EX}\mathsf E(A\mathbin{\mathsf U}B))\bigr)}{\none}
\hbrule{rule:p08-hilbert-ctl-au-unfold}{A$\mathsf U$}{}{\mathsf A(A\mathbin{\mathsf U}B)\leftrightarrow\bigl(B\lor(A\land\mathsf{AX}\mathsf A(A\mathbin{\mathsf U}B))\bigr)}{\none}
\hbrule{rule:p08-hilbert-ctl-total}{Total}{}{\mathsf{EX}\topv\land\mathsf{AX}\topv}{\none}
\hbrule{rule:p08-hilbert-ctl-ex-mon}{EX-Mon}{\derives_{\mathsf{CTL}}(A\to B)}{\derives_{\mathsf{CTL}}(\mathsf{EX}A\to\mathsf{EX}B)}{theorem-only: the displayed premise has a derivation from no premises}
\hbrule{rule:p08-hilbert-ctl-au-ind}{A$\mathsf U$-Ind}{\derives_{\mathsf{CTL}}\bigl(C\to(\neg B\land\mathsf{EX}C)\bigr)}{\derives_{\mathsf{CTL}}\bigl(C\to\neg\mathsf A(A\mathbin{\mathsf U}B)\bigr)}{theorem-only: $A,B,C$ are CTL state formulas and the displayed premise has a derivation from no premises}
\hbrule{rule:p08-hilbert-ctl-eu-ind}{E$\mathsf U$-Ind}{\derives_{\mathsf{CTL}}\bigl(C\to[\neg B\land\mathsf{AX}(C\lor\neg\mathsf E(A\mathbin{\mathsf U}B))]\bigr)}{\derives_{\mathsf{CTL}}\bigl(C\to\neg\mathsf E(A\mathbin{\mathsf U}B)\bigr)}{theorem-only: $A,B,C$ are CTL state formulas and the displayed premise has a derivation from no premises}
\hbrule{rule:p08-hilbert-ctl-mp}{MP}{A\qquad A\to B}{B}{\none}
