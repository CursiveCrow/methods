% Exact Tait-style PA-infinity surface used by the cited cut-elimination
% theorem.  Finite sets are interpreted disjunctively, so exchange,
% contraction, and weakening are built into the sequent representation.
\calculussheet{sheet:omega-arithmetic}{The Semiformal Calculus $\mathsf{PA}_{\infty}$}{Language: function symbols name all primitive-recursive numerical functions and relation symbols name all primitive-recursive numerical relations; number terms are generated from number variables, $0$, $S$, and the function symbols, while relation symbols occur in literals; formulas are semi-closed arithmetic formulas in negation-normal form, possibly with free set parameters but with no free number variables; sequents are finite sets $\Gamma$ interpreted disjunctively; derivations are well-founded trees and the universal-number rule has one premise for every closed number term.}
\noindent\emph{Controlling source:} J\"ager and Probst, Definition~1 and Theorem~2 \cite{JagerProbst2004}; compare Sch\"utte's full treatment \cite{Schuette1977}.
\hbrule{rule:omega-true-literal}{True literal}{}{\Gamma,A}{$A$ is a true semi-closed literal}
\hbrule{rule:omega-numerical-equivalence}{Numerical equivalence}{}{\Gamma,\neg B,C}{$B$ and $C$ are numerically equivalent semi-closed literals}
\hbrule{rule:omega-disjunction-left}{Disjunction 1}{\Gamma,A}{\Gamma,A\lor B}{\none}
\hbrule{rule:omega-disjunction-right}{Disjunction 2}{\Gamma,B}{\Gamma,A\lor B}{\none}
\hbrule{rule:omega-conjunction}{Conjunction}{\Gamma,A\qquad\Gamma,B}{\Gamma,A\land B}{\none}
\hbrule{rule:omega-existential}{Existential}{\Gamma,A(s)}{\Gamma,\exists u\,A(u)}{$s$ is a closed number term}
\hbrule{rule:omega-arithmetic-right}{$\omega$-rule}{\{\Gamma,A(t):t\text{ a closed number term}\}}{\Gamma,\forall u\,A(u)}{one premise for every closed number term}
\hbrule{rule:omega-cut}{Cut}{\Gamma,A\qquad\Gamma,\neg A}{\Gamma}{\none}
