\calculussheet{sheet:lk-eq}{Classical Sequent Calculus $\mathsf{LK}_{=}$ with Equality}{Language: $\mathsf{Fm}(\Lang)$, first-order with primitives $\bot,\neg,\land,\lor,\to,\forall,\exists$ and primitive equality $=$; sequent shape: two-sided sequents $\Gamma\Rightarrow\Delta$ with finite multisets on both sides (exchange built into the multiset convention); initial sequents are contexted ($\Gamma,\varphi\Rightarrow\varphi,\Delta$ and $\Gamma,\bot\Rightarrow\Delta$) plus the equality initial sequent $\Rightarrow t=t$; negation: primitive, with rules $\neg L$/$\neg R$; equality: primitive logical identity, governed by the initial sequent $\Rightarrow t=t$ and the replacement rule $=L$; the symmetric replacement instance is derivable using Cut, with no cut-free derivation claimed here.}
\hbrule{rule:lk-eq-id}{id}{}{\Gamma,\varphi\Rightarrow\varphi,\Delta}{\none}
\hbrule{rule:lk-eq-bot-l}{$\bot L$}{}{\Gamma,\bot\Rightarrow\Delta}{\none}
\hbrule{rule:lk-eq-wl}{$W_L$}{\Gamma\Rightarrow\Delta}{\Gamma,\varphi\Rightarrow\Delta}{\none}
\hbrule{rule:lk-eq-wr}{$W_R$}{\Gamma\Rightarrow\Delta}{\Gamma\Rightarrow\Delta,\varphi}{\none}
\hbrule{rule:lk-eq-cl}{$C_L$}{\Gamma,\varphi,\varphi\Rightarrow\Delta}{\Gamma,\varphi\Rightarrow\Delta}{\none}
\hbrule{rule:lk-eq-cr}{$C_R$}{\Gamma\Rightarrow\Delta,\varphi,\varphi}{\Gamma\Rightarrow\Delta,\varphi}{\none}
\hbrule{rule:lk-eq-neg-l}{$\neg L$}{\Gamma\Rightarrow\Delta,\varphi}{\Gamma,\neg\varphi\Rightarrow\Delta}{\none}
\hbrule{rule:lk-eq-neg-r}{$\neg R$}{\Gamma,\varphi\Rightarrow\Delta}{\Gamma\Rightarrow\Delta,\neg\varphi}{\none}
\hbrule{rule:lk-eq-and-l}{$\land L$}{\Gamma,\varphi,\psi\Rightarrow\Delta}{\Gamma,\varphi\land\psi\Rightarrow\Delta}{\none}
\hbrule{rule:lk-eq-and-r}{$\land R$}{\Gamma\Rightarrow\Delta,\varphi\qquad\Gamma\Rightarrow\Delta,\psi}{\Gamma\Rightarrow\Delta,\varphi\land\psi}{\none}
\hbrule{rule:lk-eq-or-l}{$\lor L$}{\Gamma,\varphi\Rightarrow\Delta\qquad\Gamma,\psi\Rightarrow\Delta}{\Gamma,\varphi\lor\psi\Rightarrow\Delta}{\none}
\hbrule{rule:lk-eq-or-r}{$\lor R$}{\Gamma\Rightarrow\Delta,\varphi,\psi}{\Gamma\Rightarrow\Delta,\varphi\lor\psi}{\none}
\hbrule{rule:lk-eq-imp-l}{$\to L$}{\Gamma\Rightarrow\Delta,\varphi\qquad\Gamma,\psi\Rightarrow\Delta}{\Gamma,\varphi\to\psi\Rightarrow\Delta}{\none}
\hbrule{rule:lk-eq-imp-r}{$\to R$}{\Gamma,\varphi\Rightarrow\Delta,\psi}{\Gamma\Rightarrow\Delta,\varphi\to\psi}{\none}
\hbrule{rule:lk-eq-cut}{Cut}{\Gamma\Rightarrow\Delta,\varphi\qquad\Sigma,\varphi\Rightarrow\Pi}{\Gamma,\Sigma\Rightarrow\Delta,\Pi}{\none}
\hbrule{rule:lk-eq-forall-l}{$\forall L$}{\Gamma,\varphi[t/x]\Rightarrow\Delta}{\Gamma,\forall x\,\varphi\Rightarrow\Delta}{substitution: $[t/x]$ is the total capture-avoiding substitution fixed in the syntax chapter, so no separate substitutability proviso is imposed}
\hbrule{rule:lk-eq-forall-r}{$\forall R$}{\Gamma\Rightarrow\Delta,\varphi[y/x]}{\Gamma\Rightarrow\Delta,\forall x\,\varphi}{eigenvariable freshness: $y$ occurs free in no formula of $\Gamma,\Delta,\forall x\,\varphi$; substitution total and capture-avoiding}
\hbrule{rule:lk-eq-exists-l}{$\exists L$}{\Gamma,\varphi[y/x]\Rightarrow\Delta}{\Gamma,\exists x\,\varphi\Rightarrow\Delta}{eigenvariable freshness: $y$ occurs free in no formula of $\Gamma,\Delta,\exists x\,\varphi$; substitution total and capture-avoiding}
\hbrule{rule:lk-eq-exists-r}{$\exists R$}{\Gamma\Rightarrow\Delta,\varphi[t/x]}{\Gamma\Rightarrow\Delta,\exists x\,\varphi}{substitution: $[t/x]$ is the total capture-avoiding substitution, as in $\forall L$}
\hbrule{rule:lk-eq-mix}{Mix}{\Gamma\Rightarrow\Delta,\varphi^{m}\qquad\varphi^{n},\Sigma\Rightarrow\Pi}{\Gamma,\Sigma\Rightarrow\Delta,\Pi}{$m,n\geq 1$; $\varphi^{k}$ abbreviates $k$ comma-separated occurrences of the mix formula $\varphi$; $\Delta$ and $\Sigma$ may contain further occurrences of $\varphi$; equipollence with primitive Cut follows by contraction, but neither it nor the Part-7 theorem establishes Cut admissibility for this equality extension}
\hbrule{rule:lk-eq-refl}{$=I$}{}{\Rightarrow t=t}{\none}
\hbrule{rule:lk-eq-repl-l}{$=L$}{\Gamma,\varphi[r/x]\Rightarrow\Delta}{\Gamma,r=t,\varphi[t/x]\Rightarrow\Delta}{substitution: both displayed instances use the total capture-avoiding substitution; the symmetric replacement instance (from $\Gamma,\varphi[t/x]\Rightarrow\Delta$ infer $\Gamma,r=t,\varphi[r/x]\Rightarrow\Delta$) is derivable using Cut, not primitive, and no cut-free derivation is asserted}
