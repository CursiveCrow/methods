% Editorial Hoare sheet for the deterministic while language of Part IX.
\calculussheet{sheet:hoare-while}{Hoare Calculus for the Deterministic While Language}{Language: commands, integer states, assertions, and big-step semantics of Definition~\ref{def:p09-hoare-triple}; judgement: \(\derives_{\kappa}\{P\}C\{Q\}\), where \(\kappa\in\{\mathsf{par},\mathsf{tot}\}\) records partial or total correctness; assertion implication in side premises is semantic validity over the fixed integer state structure, uniformly in every rigid logical-variable value; equality: integer equality.}
\noindent\emph{Controlling sources:} \cite{Floyd1967,Hoare1969,Apt1981,Cook1978}.
\hbrule{rule:p09-hoare-skip}{Skip}{}{\derives_{\kappa}\{P\}\mathbf{skip}\{P\}}{\(\kappa\in\{\mathsf{par},\mathsf{tot}\}\)}
\hbrule{rule:p09-hoare-assignment}{Assignment}{}{\derives_{\kappa}\{Q[e/x]\}\ x:=e\ \{Q\}}{capture-avoiding assertion substitution; \(\kappa\in\{\mathsf{par},\mathsf{tot}\}\)}
\hbrule{rule:p09-hoare-consequence}{Consequence}{P\Rightarrow P'\qquad\derives_{\kappa}\{P'\}C\{Q'\}\qquad Q'\Rightarrow Q}{\derives_{\kappa}\{P\}C\{Q\}}{both displayed implications are valid in every state; the tag \(\kappa\) is unchanged}
\hbrule{rule:p09-hoare-sequence}{Sequence}{\derives_{\kappa}\{P\}C\{R\}\qquad\derives_{\kappa}\{R\}D\{Q\}}{\derives_{\kappa}\{P\}C;D\{Q\}}{the tag \(\kappa\) is the same in both premises and the conclusion}
\hbrule{rule:p09-hoare-if}{Conditional}{\derives_{\kappa}\{P\land b\}C\{Q\}\qquad\derives_{\kappa}\{P\land\neg b\}D\{Q\}}{\derives_{\kappa}\{P\}\mathbf{if}\ b\ \mathbf{then}\ C\ \mathbf{else}\ D\{Q\}}{the tag \(\kappa\) is the same in both premises and the conclusion}
\hbrule{rule:p09-hoare-while-partial}{While, partial}{\derives_{\mathsf{par}}\{I\land b\}C\{I\}}{\derives_{\mathsf{par}}\{I\}\mathbf{while}\ b\ \mathbf{do}\ C\{I\land\neg b\}}{\(I\) is the loop invariant}
\hbrule{rule:p09-hoare-while-total}{While, total}{I\land b\Rightarrow t\geq0\qquad\derives_{\mathsf{tot}}\{I\land b\land t=z\}C\{I\land t<z\}}{\derives_{\mathsf{tot}}\{I\}\mathbf{while}\ b\ \mathbf{do}\ C\{I\land\neg b\}}{\(t\) is integer-valued; \(z\) is a fresh rigid logical variable in the exact sense of Definition~\ref{def:p09-hoare-triple}, so the premise compares the new value of \(t\) with its old nonnegative value}

The total-while row is the handbook's editorial decreasing-integer rule and is
proved directly in Annex~\ref{annex:p9.hoare.1}.  Apt surveys total correctness
and prints a distinct Harel-attributed indexed-assertion rule.  Cook's result
controls relative completeness for partial correctness only; neither source is
misidentified as the source of this editorial rule.
