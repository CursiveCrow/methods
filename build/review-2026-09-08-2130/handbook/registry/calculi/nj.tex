\calculussheet{sheet:nj}{Intuitionistic Natural Deduction $\mathsf{NJ}$}{Language: first-order formulas over the primitives $\bot,\land,\lor,\to,\forall,\exists$, with $\neg\varphi\defeq\varphi\to\bot$; the locally declared judgment $\Gamma\derives_{\mathsf{NJ}}\varphi$ has $\Gamma$ as the finite set supporting the open assumption occurrences of a finite rooted derivation tree, and discharge labels are metalinguistic natural numbers; every discharged label marks at least one open occurrence (vacuous discharge is not licensed); total capture-avoiding substitution and the displayed eigenvariable conditions are binding; RAA, primitive-negation rules, equality rules, and arithmetic induction are absent.}
\hbrule{rule:nj-assumption}{assumption}{}{\Gamma\cup\{\varphi\}\derives_{\mathsf{NJ}}\varphi}{\none}
\hbrule{rule:nj-and-i}{$\land I$}{\Gamma\derives_{\mathsf{NJ}}\varphi\qquad\Delta\derives_{\mathsf{NJ}}\psi}{\Gamma\cup\Delta\derives_{\mathsf{NJ}}\varphi\land\psi}{\none}
\hbrule{rule:nj-and-e1}{$\land E_1$}{\Gamma\derives_{\mathsf{NJ}}\varphi\land\psi}{\Gamma\derives_{\mathsf{NJ}}\varphi}{\none}
\hbrule{rule:nj-and-e2}{$\land E_2$}{\Gamma\derives_{\mathsf{NJ}}\varphi\land\psi}{\Gamma\derives_{\mathsf{NJ}}\psi}{\none}
\hbrule{rule:nj-or-i1}{$\lor I_1$}{\Gamma\derives_{\mathsf{NJ}}\varphi}{\Gamma\derives_{\mathsf{NJ}}\varphi\lor\psi}{\none}
\hbrule{rule:nj-or-i2}{$\lor I_2$}{\Gamma\derives_{\mathsf{NJ}}\psi}{\Gamma\derives_{\mathsf{NJ}}\varphi\lor\psi}{\none}
\hbrule{rule:nj-or-e}{$\lor E$}{\Gamma\derives_{\mathsf{NJ}}\varphi\lor\psi\qquad\Delta\cup\{\varphi\}\derives_{\mathsf{NJ}}\chi\qquad\Sigma\cup\{\psi\}\derives_{\mathsf{NJ}}\chi}{\Gamma\cup\Delta\cup\Sigma\derives_{\mathsf{NJ}}\chi}{discharge: the application carries two labels $i,j$; $i$ marks at least one open occurrence of $\varphi$ above the second premise and $j$ at least one open occurrence of $\psi$ above the third, and the application discharges exactly those marked occurrences}
\hbrule{rule:nj-imp-i}{$\to I$}{\Gamma\cup\{\varphi\}\derives_{\mathsf{NJ}}\psi}{\Gamma\derives_{\mathsf{NJ}}\varphi\to\psi}{discharge: the application carries a label $i$ marking at least one open occurrence of $\varphi$ above the premise and discharges exactly those marked occurrences}
\hbrule{rule:nj-imp-e}{$\to E$}{\Gamma\derives_{\mathsf{NJ}}\varphi\to\psi\qquad\Delta\derives_{\mathsf{NJ}}\varphi}{\Gamma\cup\Delta\derives_{\mathsf{NJ}}\psi}{\none}
\hbrule{rule:nj-bot-e}{$\bot E$}{\Gamma\derives_{\mathsf{NJ}}\bot}{\Gamma\derives_{\mathsf{NJ}}\varphi}{\none}
\hbrule{rule:nj-forall-e}{$\forall E$}{\Gamma\derives_{\mathsf{NJ}}\forall x\,\varphi}{\Gamma\derives_{\mathsf{NJ}}\varphi[t/x]}{substitution: $[t/x]$ is the total capture-avoiding substitution fixed in the first-order syntax chapter}
\hbrule{rule:nj-exists-i}{$\exists I$}{\Gamma\derives_{\mathsf{NJ}}\varphi[t/x]}{\Gamma\derives_{\mathsf{NJ}}\exists x\,\varphi}{substitution: total capture-avoiding substitution, as in $\forall E$}
\hbrule{rule:nj-forall-i}{$\forall I$}{\Gamma\derives_{\mathsf{NJ}}\varphi}{\Gamma\derives_{\mathsf{NJ}}\forall x\,\varphi}{eigenvariable freshness: $x\notin\mathrm{FV}(\Gamma)$, so $x$ is free in no open assumption}
\hbrule{rule:nj-exists-e}{$\exists E$}{\Gamma\derives_{\mathsf{NJ}}\exists x\,\varphi\qquad\Delta\cup\{\varphi[y/x]\}\derives_{\mathsf{NJ}}\psi}{\Gamma\cup\Delta\derives_{\mathsf{NJ}}\psi}{eigenvariable freshness: $y$ is absent from $\psi$, every formula of $\Gamma\cup\Delta$, and $\exists x\,\varphi$; discharge: a label $i$ marks at least one open occurrence of $\varphi[y/x]$ above the second premise and the application discharges exactly those marked occurrences}
