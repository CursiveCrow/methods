% Deduction and Sequents" (ND_CPL rule displays), presented in the contexted
% form of the "Rule Sheets" chapter (sec:rule-sheets-01), which the manuscript
% states is literally the active calculus with contexts made explicit.
% R-013 is binding here: every discharge is nonvacuous.  Consistent injective
% renaming of an entire derivation's labels preserves its identity; copies are
% renamed to label sets globally fresh from the receiving derivation and from
% one another.  There is deliberately no repetition rule.
\calculussheet{sheet:nd-cpl}{Classical Propositional Natural Deduction ($\mathsf{ND}_{\mathsf{CPL}}$)}{Language: $\mathsf{Fm}_{\mathsf{CPL}}$ with primitives $\bot,\neg,\land,\lor,\to$; sequent shape: single-conclusion derivability judgments $\Gamma\vdash\varphi$ with $\Gamma$ the finite set supporting the open assumption occurrences of a finite rooted derivation tree; discharge labels are metalinguistic natural numbers, and every label carried by a rule marks at least one open occurrence discharged by that rule (vacuous discharge is not licensed; there is no repetition rule); negation: primitive, with its own rules alongside primitive $\bot$, and RAA supplying classical strength; equality: none (propositional language).}
\hbrule{rule:nd-cpl-assumption}{assumption}{}{\Gamma\cup\{\varphi\}\vdash\varphi}{\none}
\hbrule{rule:nd-cpl-and-i}{$\land I$}{\Gamma\vdash\varphi\qquad\Delta\vdash\psi}{\Gamma\cup\Delta\vdash\varphi\land\psi}{\none}
\hbrule{rule:nd-cpl-and-e1}{$\land E_1$}{\Gamma\vdash\varphi\land\psi}{\Gamma\vdash\varphi}{\none}
\hbrule{rule:nd-cpl-and-e2}{$\land E_2$}{\Gamma\vdash\varphi\land\psi}{\Gamma\vdash\psi}{\none}
\hbrule{rule:nd-cpl-or-i1}{$\lor I_1$}{\Gamma\vdash\varphi}{\Gamma\vdash\varphi\lor\psi}{\none}
\hbrule{rule:nd-cpl-or-i2}{$\lor I_2$}{\Gamma\vdash\psi}{\Gamma\vdash\varphi\lor\psi}{\none}
\hbrule{rule:nd-cpl-or-e}{$\lor E$}{\Gamma\vdash\varphi\lor\psi\qquad\Delta\cup\{\varphi\}\vdash\chi\qquad\Sigma\cup\{\psi\}\vdash\chi}{\Gamma\cup\Delta\cup\Sigma\vdash\chi}{discharge: an application carries two labels $i,j$; label $i$ marks at least one open occurrence of $\varphi$ above the second premise and label $j$ at least one open occurrence of $\psi$ above the third, and the application discharges exactly all such marked occurrences}
\hbrule{rule:nd-cpl-imp-i}{$\to I$}{\Gamma\cup\{\varphi\}\vdash\psi}{\Gamma\vdash\varphi\to\psi}{discharge: the application carries a label $i$ marking at least one open occurrence of $\varphi$ above its premise and discharges exactly all such marked occurrences}
\hbrule{rule:nd-cpl-imp-e}{$\to E$}{\Gamma\vdash\varphi\to\psi\qquad\Delta\vdash\varphi}{\Gamma\cup\Delta\vdash\psi}{\none}
\hbrule{rule:nd-cpl-neg-i}{$\neg I$}{\Gamma\cup\{\varphi\}\vdash\bot}{\Gamma\vdash\neg\varphi}{discharge: the application carries a label $i$ marking at least one open occurrence of $\varphi$ above its premise and discharges exactly all such marked occurrences}
\hbrule{rule:nd-cpl-neg-e}{$\neg E$}{\Gamma\vdash\varphi\qquad\Delta\vdash\neg\varphi}{\Gamma\cup\Delta\vdash\bot}{\none}
\hbrule{rule:nd-cpl-bot-e}{$\bot E$}{\Gamma\vdash\bot}{\Gamma\vdash\varphi}{\none}
\hbrule{rule:nd-cpl-raa}{RAA}{\Gamma\cup\{\neg\varphi\}\vdash\bot}{\Gamma\vdash\varphi}{discharge: the application carries a label $i$ marking at least one open occurrence of $\neg\varphi$ above its premise and discharges exactly all such marked occurrences; this is the explicitly classical rule, and omitting it gives the primitive-negation presentation of the intuitionistic fragment}
