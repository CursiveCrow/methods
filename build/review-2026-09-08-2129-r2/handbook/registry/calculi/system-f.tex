% NEW CONTENT for D-P9-004 / N-c-029.  This is the explicitly typed
% (Church-style) presentation used by Part 7.  It is deliberately distinct
% from Curry-style System F, whose erased terms have a different decision
% problem.  Type metavariables are Greek; a,b are bound type variables;
% M,N are term metavariables; and Theta_ty is the term-variable context.
\calculussheet{sheet:system-f}{Church-style System F}{Fix disjoint countably infinite sets of type and term variables.  Types form the least set containing every type variable and closed under \(\alpha\mathbin{\to_{\mathrm{ty}}}\beta\) and \(\forall a.\alpha\); terms form the least set containing every term variable and closed under \(\lambda x{:}\alpha.M\), \(MN\), \(\Lambda a.M\), and \(M[\alpha]\); type-formation judgments have the form \(\Delta_{\mathrm{ty}}\derives_{\mathsf F}\alpha\ \mathsf{type}\), and term-typing judgments the form \(\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}M{:}\alpha\), where \(\Delta_{\mathrm{ty}}\) lists distinct type variables and \(\Theta_{\mathrm{ty}}\) assigns distinct term variables types well formed over \(\Delta_{\mathrm{ty}}\); \(\to_{\mathrm{ty}}\) is the registered metalinguistic type former; negation and equality are not primitive.}
\hbrule{rule:system-f-type-var}{Type variable}{a\in\Delta_{\mathrm{ty}}}{\Delta_{\mathrm{ty}}\derives_{\mathsf F}a\ \mathsf{type}}{\none}
\hbrule{rule:system-f-type-arrow}{Arrow formation}{\Delta_{\mathrm{ty}}\derives_{\mathsf F}\alpha\ \mathsf{type}\qquad\Delta_{\mathrm{ty}}\derives_{\mathsf F}\beta\ \mathsf{type}}{\Delta_{\mathrm{ty}}\derives_{\mathsf F}\alpha\mathbin{\to_{\mathrm{ty}}}\beta\ \mathsf{type}}{\none}
\hbrule{rule:system-f-type-all}{Universal formation}{\Delta_{\mathrm{ty}},a\derives_{\mathsf F}\alpha\ \mathsf{type}}{\Delta_{\mathrm{ty}}\derives_{\mathsf F}\forall a.\alpha\ \mathsf{type}}{$a\notin\Delta_{\mathrm{ty}}$; bound type variables may be renamed consistently}
\hbrule{rule:system-f-var}{Var}{x{:}\alpha\in\Theta_{\mathrm{ty}}}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}x{:}\alpha}{$\Theta_{\mathrm{ty}}$ is well formed over $\Delta_{\mathrm{ty}}$}
\hbrule{rule:system-f-abs}{$\to_{\mathrm{ty}} I$}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}},x{:}\alpha\derives_{\mathsf F}M{:}\beta}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}\lambda x{:}\alpha.M{:}\alpha\mathbin{\to_{\mathrm{ty}}}\beta}{$x$ is not declared in $\Theta_{\mathrm{ty}}$; bound term variables may be renamed consistently}
\hbrule{rule:system-f-app}{$\to_{\mathrm{ty}} E$}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}M{:}\alpha\mathbin{\to_{\mathrm{ty}}}\beta\qquad\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}N{:}\alpha}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}MN{:}\beta}{\none}
\hbrule{rule:system-f-type-abs}{$\forall I$}{\Delta_{\mathrm{ty}},a;\Theta_{\mathrm{ty}}\derives_{\mathsf F}M{:}\alpha}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}\Lambda a.M{:}\forall a.\alpha}{$a\notin\Delta_{\mathrm{ty}}$ and $a\notin\FV(\beta)$ for every $x{:}\beta\in\Theta_{\mathrm{ty}}$}
\hbrule{rule:system-f-type-app}{$\forall E$}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}M{:}\forall a.\alpha\qquad\Delta_{\mathrm{ty}}\derives_{\mathsf F}\beta\ \mathsf{type}}{\Delta_{\mathrm{ty}};\Theta_{\mathrm{ty}}\derives_{\mathsf F}M[\beta]{:}\alpha[\beta/a]}{$\alpha[\beta/a]$ is capture-avoiding type substitution}
\hbrule{rule:system-f-term-beta}{Term $\beta$}{}{(\lambda x{:}\alpha.M)N\longrightarrow_{\beta}M[N/x]}{$M[N/x]$ is capture-avoiding term substitution; reduction is closed under term contexts}
\hbrule{rule:system-f-type-beta}{Type $\beta$}{}{(\Lambda a.M)[\alpha]\longrightarrow_{\beta}M[\alpha/a]}{$M[\alpha/a]$ is capture-avoiding type substitution; reduction is closed under term contexts}
