% Editorial sheet for the first-order Martelli--Montanari transformation system.
\calculussheet{sheet:unification-first-order}{First-Order Syntactic Unification}{Language: finite first-order terms over a ranked signature and variables; state: a finite multiset $E$ of equations, transformed until failure or solved form; substitution application is simultaneous and capture-free; equality: syntactic identity after substitution; no equational theory such as associativity or commutativity is built in.}
\noindent\emph{Controlling sources:} \cite{MartelliMontanari1982,BaaderSnyder2001}.
\hbrule{rule:unification-delete}{Delete}{E\uplus\{t\doteq t\}}{E}{\none}
\hbrule{rule:unification-decompose}{Decompose}{E\uplus\{f(s_1,\ldots,s_n)\doteq f(t_1,\ldots,t_n)\}}{E\uplus\{s_1\doteq t_1,\ldots,s_n\doteq t_n\}}{same function symbol and arity}
\hbrule{rule:unification-clash}{Clash}{E\uplus\{f(\bar s)\doteq g(\bar t)\}}{\mathsf{fail}}{different function symbols or arities}
\hbrule{rule:unification-orient}{Orient}{E\uplus\{t\doteq x\}}{E\uplus\{x\doteq t\}}{$x$ is a variable and $t$ is not a variable}
\hbrule{rule:unification-occurs}{Occurs}{E\uplus\{x\doteq t\}}{\mathsf{fail}}{$x\ne t$ and $x\in\operatorname{Var}(t)$}
\hbrule{rule:unification-eliminate}{Eliminate}{E\uplus\{x\doteq t\}}{E[x:=t]\uplus\{x\doteq t\}}{$x\notin\operatorname{Var}(t)$ and $x$ occurs in $E$}
