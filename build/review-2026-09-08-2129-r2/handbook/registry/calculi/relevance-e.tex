% Ackermann's gamma is admissible metatheory, not a primitive rule here.
\calculussheet{sheet:relevance-e}{Relevant Entailment Calculus $\mathbf E$}{Language: the least constant-free propositional formula set generated by $\neg,\land,\lor,\to$; shape: no sequents --- theoremhood $\derives_{\mathbf E} A$ is membership in the least set containing every instance of the displayed schemas and closed under modus ponens and adjunction; $\Box A$ is only the eliminable abbreviation $(A\to A)\to A$; negation: primitive relevant negation; equality: none.}
\hbrule{rule:relevance-e-suffix}{Suffixing}{}{(A\to B)\to((B\to C)\to(A\to C))}{\none}
\hbrule{rule:relevance-e-contract}{Contraction}{}{(A\to(A\to B))\to(A\to B)}{\none}
\hbrule{rule:relevance-e-specassert}{Specialized Assertion}{}{((A\to A)\to B)\to B}{\none}
\hbrule{rule:relevance-e-and-elim-l}{$\land E_1$}{}{A\land B\to A}{\none}
\hbrule{rule:relevance-e-and-elim-r}{$\land E_2$}{}{A\land B\to B}{\none}
\hbrule{rule:relevance-e-and-intro}{$\land I$}{}{((A\to B)\land(A\to C))\to(A\to(B\land C))}{\none}
\hbrule{rule:relevance-e-or-intro-l}{$\lor I_1$}{}{A\to A\lor B}{\none}
\hbrule{rule:relevance-e-or-intro-r}{$\lor I_2$}{}{B\to A\lor B}{\none}
\hbrule{rule:relevance-e-or-elim}{$\lor E$}{}{((A\to C)\land(B\to C))\to((A\lor B)\to C)}{\none}
\hbrule{rule:relevance-e-distrib}{Distribution}{}{A\land(B\lor C)\to(A\land B)\lor C}{\none}
\hbrule{rule:relevance-e-modal-and}{Modal Conjunction}{}{\Box A\land\Box B\to\Box(A\land B)}{$\Box X$ abbreviates $(X\to X)\to X$; it is not a new primitive connective}
\hbrule{rule:relevance-e-reductio}{Reductio}{}{(A\to\neg A)\to\neg A}{\none}
\hbrule{rule:relevance-e-contraposition}{Contraposition}{}{(A\to\neg B)\to(B\to\neg A)}{\none}
\hbrule{rule:relevance-e-dne}{Double Negation}{}{\neg\neg A\to A}{\none}
\hbrule{rule:relevance-e-mp}{MP}{A\qquad A\to B}{B}{\none}
\hbrule{rule:relevance-e-adj}{Adjunction}{A\qquad B}{A\land B}{\none}
