% AUTHORED, NOT EXTRACTED: v1.1 contains no classical first-order tableau
% calculus.  This unsigned parameter tableau follows the classical analytic
% presentation of Smullyan1968.  It is deliberately equality-free: adding
% equality requires a separate equality calculus, not an unstated reading of
% atomic identity.  Depth row D-P3-002 (first-order half).
%
% Formula occurrences are retained.  Propositional and delta occurrences are
% expanded once.  A gamma occurrence records the closed terms already used and
% may be expanded again at every new term; this distinction is required for
% fair saturation and first-order countermodel extraction.  Fairness ranges
% over every closed term of the active root language, not merely terms already
% occurring on a branch.
\calculussheet{sheet:tableaux-fol}{Classical First-Order Analytic Tableaux}{Language: for a finite equality-free root $\Gamma$, let $\mathcal L_{\Gamma}$ contain exactly its nonlogical symbols and obtain $\mathcal L_{\Gamma}^*$ by adjoining a countable reserve $a_n$ ($n\in\mathbb N$) of new parameter constants; fix an enumeration of every closed $\mathcal L_{\Gamma}^*$-term; shape: a tableau rooted at $\Gamma$ is a finitely branching tree, possibly infinite, whose branches retain formula occurrences and metalinguistic expansion records; a finite closed tableau is a refutation; propositional and delta occurrences are marked after one expansion, whereas each gamma occurrence records the closed terms used and remains available for every unused enumerated term; a branch closes on $\bot$ or a complementary pair, is saturated when no applicable expansion is omitted, and is fair when every gamma occurrence is eventually instantiated with every term in the fixed enumeration, whether or not that term previously appeared on the branch; negation: primitive, with dual connective and quantifier expansions; equality: none.}
\hbrule{rule:tableaux-fol-and}{$\land$}{\varphi\land\psi}{\varphi,\;\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-fol-neg-and}{$\neg\land$}{\neg(\varphi\land\psi)}{\neg\varphi\;\mid\;\neg\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-fol-or}{$\lor$}{\varphi\lor\psi}{\varphi\;\mid\;\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-fol-neg-or}{$\neg\lor$}{\neg(\varphi\lor\psi)}{\neg\varphi,\;\neg\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-fol-imp}{$\to$}{\varphi\to\psi}{\neg\varphi\;\mid\;\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-fol-neg-imp}{$\neg\to$}{\neg(\varphi\to\psi)}{\varphi,\;\neg\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-fol-neg-neg}{$\neg\neg$}{\neg\neg\varphi}{\varphi}{linear expansion: retain and mark the principal occurrence, then append the conclusion on every open branch through it}
\hbrule{rule:tableaux-fol-forall}{$\forall$}{\forall x\,\varphi}{\varphi[t/x]}{gamma expansion: $t$ is any closed $\mathcal L_{\Gamma}^*$-term not yet used for this occurrence; retain the occurrence, record $t$, and use the book's total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-neg-exists}{$\neg\exists$}{\neg\exists x\,\varphi}{\neg(\varphi[t/x])}{gamma expansion: $t$ is any closed $\mathcal L_{\Gamma}^*$-term not yet used for this occurrence; retain the occurrence, record $t$, and use total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-exists}{$\exists$}{\exists x\,\varphi}{\varphi[a/x]}{delta expansion: $a$ is the least parameter constant not occurring earlier on the branch; retain and mark the principal occurrence and use total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-neg-forall}{$\neg\forall$}{\neg\forall x\,\varphi}{\neg(\varphi[a/x])}{delta expansion: $a$ is the least parameter constant not occurring earlier on the branch; retain and mark the principal occurrence and use total capture-avoiding substitution}
\hbrule{rule:tableaux-fol-close}{close}{\varphi\qquad\neg\varphi}{\text{closed branch}}{a branch containing both displayed formulas, or containing $\bot$, is closed and receives no further expansion; a tableau is closed iff every branch is closed}
