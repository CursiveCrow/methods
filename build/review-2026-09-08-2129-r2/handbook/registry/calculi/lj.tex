\calculussheet{sheet:lj}{Intuitionistic Sequent Calculus \(\mathsf{LJ}\)}{Language: first-order formulas over the primitives \(\bot,\land,\lor,\to,\forall,\exists\); sequent shape: single-succedent sequents \(\Gamma\Rightarrow\chi\) with a finite-multiset antecedent and exactly one succedent formula (the restriction is part of the calculus, not a typographic abbreviation; exchange built into the multiset representation; no right weakening or right contraction); initial sequents are uncontexted (\(\varphi\Rightarrow\varphi\) and \(\bot,\Gamma\Rightarrow\chi\)); negation: defined, \(\neg\varphi\defeq\varphi\to\bot\), with no primitive negation rules; equality: none.}
\hbrule{rule:lj-id}{Id}{}{\varphi\Rightarrow\varphi}{\none}
\hbrule{rule:lj-bot-l}{\(\bot L\)}{}{\bot,\Gamma\Rightarrow\chi}{\none}
\hbrule{rule:lj-wl}{WL}{\Gamma\Rightarrow\chi}{\varphi,\Gamma\Rightarrow\chi}{\none}
\hbrule{rule:lj-cl}{CL}{\varphi,\varphi,\Gamma\Rightarrow\chi}{\varphi,\Gamma\Rightarrow\chi}{\none}
\hbrule{rule:lj-and-l}{\(\land L\)}{\varphi,\psi,\Gamma\Rightarrow\chi}{\varphi\land\psi,\Gamma\Rightarrow\chi}{\none}
\hbrule{rule:lj-and-r}{\(\land R\)}{\Gamma\Rightarrow\varphi\qquad\Gamma\Rightarrow\psi}{\Gamma\Rightarrow\varphi\land\psi}{\none}
\hbrule{rule:lj-or-l}{\(\lor L\)}{\varphi,\Gamma\Rightarrow\chi\qquad\psi,\Gamma\Rightarrow\chi}{\varphi\lor\psi,\Gamma\Rightarrow\chi}{\none}
\hbrule{rule:lj-or-r1}{\(\lor R_1\)}{\Gamma\Rightarrow\varphi}{\Gamma\Rightarrow\varphi\lor\psi}{\none}
\hbrule{rule:lj-or-r2}{\(\lor R_2\)}{\Gamma\Rightarrow\psi}{\Gamma\Rightarrow\varphi\lor\psi}{\none}
\hbrule{rule:lj-imp-l}{\(\to L\)}{\Gamma\Rightarrow\varphi\qquad\psi,\Pi\Rightarrow\chi}{\varphi\to\psi,\Gamma,\Pi\Rightarrow\chi}{\none}
\hbrule{rule:lj-imp-r}{\(\to R\)}{\varphi,\Gamma\Rightarrow\psi}{\Gamma\Rightarrow\varphi\to\psi}{\none}
\hbrule{rule:lj-forall-l}{\(\forall L\)}{\varphi[t/x],\Gamma\Rightarrow\chi}{\forall x\,\varphi,\Gamma\Rightarrow\chi}{substitution: total capture-avoiding substitution of the first-order syntax chapter, so no substitutability side condition is imposed}
\hbrule{rule:lj-forall-r}{\(\forall R\)}{\Gamma\Rightarrow\varphi[y/x]}{\Gamma\Rightarrow\forall x\,\varphi}{eigenvariable freshness: \(y\) must not occur free in the conclusion sequent; substitution total and capture avoiding}
\hbrule{rule:lj-exists-l}{\(\exists L\)}{\varphi[y/x],\Gamma\Rightarrow\chi}{\exists x\,\varphi,\Gamma\Rightarrow\chi}{eigenvariable freshness: \(y\) must not occur free in the conclusion sequent; substitution total and capture avoiding}
\hbrule{rule:lj-exists-r}{\(\exists R\)}{\Gamma\Rightarrow\varphi[t/x]}{\Gamma\Rightarrow\exists x\,\varphi}{substitution: total capture-avoiding substitution, as in \(\forall L\)}
\hbrule{rule:lj-cut}{Cut}{\Gamma\Rightarrow\varphi\qquad\varphi,\Pi\Rightarrow\chi}{\Gamma,\Pi\Rightarrow\chi}{the displayed occurrence of \(\varphi\) is the cut formula}
\hbrule{rule:lj-mix}{Mix}{\Gamma\Rightarrow\varphi\qquad\varphi^{n},\Pi\Rightarrow\chi}{\Gamma,\Pi\Rightarrow\chi}{\(n\geq1\); \(\varphi^{n}\) abbreviates exactly \(n\) comma-separated antecedent occurrences of the mix formula \(\varphi\); the left premise has exactly the one succedent occurrence required by the single-succedent calculus}
