% NEW for Phase 4, P8.6.  One-sided propositional CLL presentation fixed by
% Girard (1995), sec. 1.2.2.  This sheet is distinct from the inherited IMLL
% core and from its source-bounded bang-only extension.
\calculussheet{sheet:cll-one-sided}{One-Sided Classical Linear Logic $\mathsf{CLL}$}{Language: paired atoms $p,p^{\perp}$, multiplicative units $\mathbf 1,\bot$, additive units $\top,0$, and $\otimes,\bindnasrepma,\mathbin{\&},\oplus,!,?$ with involutive De Morgan linear negation; sequent shape: $\derives_{\mathsf{CLL}}\Gamma$, where $\Gamma$ is a finite multiset; exchange: built into multiset equality; unrestricted weakening and contraction: absent, with the quest-prefixed exceptions below; equality: none.  There is no rule introducing the additive unit $0$.}
\hbrule{rule:cll-id}{Id}{}{\derives_{\mathsf{CLL}}A,A^{\perp}}{\none}
\hbrule{rule:cll-cut}{Cut}{\derives_{\mathsf{CLL}}\Gamma,A\qquad\derives_{\mathsf{CLL}}A^{\perp},\Delta}{\derives_{\mathsf{CLL}}\Gamma,\Delta}{\none}
\hbrule{rule:cll-tensor}{$\otimes$}{\derives_{\mathsf{CLL}}\Gamma,A\qquad\derives_{\mathsf{CLL}}B,\Delta}{\derives_{\mathsf{CLL}}\Gamma,A\otimes B,\Delta}{\none}
\hbrule{rule:cll-par}{$\bindnasrepma$}{\derives_{\mathsf{CLL}}\Gamma,A,B}{\derives_{\mathsf{CLL}}\Gamma,A\bindnasrepma B}{\none}
\hbrule{rule:cll-one}{$\mathbf 1$}{}{\derives_{\mathsf{CLL}}\mathbf 1}{\none}
\hbrule{rule:cll-bottom}{$\bot$}{\derives_{\mathsf{CLL}}\Gamma}{\derives_{\mathsf{CLL}}\Gamma,\bot}{\none}
\hbrule{rule:cll-with}{$\&$}{\derives_{\mathsf{CLL}}\Gamma,A\qquad\derives_{\mathsf{CLL}}\Gamma,B}{\derives_{\mathsf{CLL}}\Gamma,A\mathbin{\&}B}{\none}
\hbrule{rule:cll-plus-l}{$\oplus_1$}{\derives_{\mathsf{CLL}}\Gamma,A}{\derives_{\mathsf{CLL}}\Gamma,A\oplus B}{\none}
\hbrule{rule:cll-plus-r}{$\oplus_2$}{\derives_{\mathsf{CLL}}\Gamma,B}{\derives_{\mathsf{CLL}}\Gamma,A\oplus B}{\none}
\hbrule{rule:cll-top}{$\top$}{}{\derives_{\mathsf{CLL}}\Gamma,\top}{$\Gamma$ is any finite multiset}
\hbrule{rule:cll-dereliction}{Dereliction}{\derives_{\mathsf{CLL}}\Gamma,A}{\derives_{\mathsf{CLL}}\Gamma,?A}{\none}
\hbrule{rule:cll-weakening}{Exponential Weakening}{\derives_{\mathsf{CLL}}\Gamma}{\derives_{\mathsf{CLL}}\Gamma,?A}{\none}
\hbrule{rule:cll-contraction}{Exponential Contraction}{\derives_{\mathsf{CLL}}\Gamma,?A,?A}{\derives_{\mathsf{CLL}}\Gamma,?A}{\none}
\hbrule{rule:cll-promotion}{Promotion}{\derives_{\mathsf{CLL}}?\Gamma,A}{\derives_{\mathsf{CLL}}?\Gamma,!A}{every member of $?\Gamma$ is quest-prefixed}
