\calculussheet{sheet:gs5-avron}{Avron Hypersequent Calculus GS5 for $\mathsf{S5}$}{Language: classical modal formulas with box primitive and diamond defined; hypersequents are finite multisets of finite-multiset classical sequents separated by $\mid$, read disjunctively; all classical LK logical rules apply inside one component and carry an arbitrary external context $G$; internal/external weakening and contraction are explicit below; equality: none.}
\noindent\emph{Controlling source:} \cite{Avron1996}.

\hbrule{rule:gs5-box-right}{Box-R}{G\mid\Box\Gamma\Rightarrow A}{G\mid\Box\Gamma\Rightarrow\Box A}{every antecedent member is box-prefixed}
\hbrule{rule:gs5-ms}{Modalized Splitting}{G\mid\Box\Gamma_1,\Gamma_2\Rightarrow\Box\Delta_1,\Delta_2}{G\mid\Box\Gamma_1\Rightarrow\Box\Delta_1\mid\Gamma_2\Rightarrow\Delta_2}{\none}
\hbrule{rule:gs5-iw}{Internal Weakening}{G\mid\Gamma\Rightarrow\Delta}{G\mid A,\Gamma\Rightarrow\Delta,B}{either side may be weakened independently}
\hbrule{rule:gs5-ew}{External Weakening}{G}{G\mid\Gamma\Rightarrow\Delta}{\none}
\hbrule{rule:gs5-ec}{External Contraction}{G\mid\Gamma\Rightarrow\Delta\mid\Gamma\Rightarrow\Delta}{G\mid\Gamma\Rightarrow\Delta}{\none}
