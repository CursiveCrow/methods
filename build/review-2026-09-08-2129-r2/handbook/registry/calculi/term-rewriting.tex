% Editorial calculus sheet for the exact first-order rewrite relation and the
% completion transformations used in Part XII. It is not a source-restoration
% artifact and carries no claim about a different higher-order or AC system.
\calculussheet{sheet:term-rewriting}{First-Order Term Rewriting and Completion}{Language: finite first-order terms over a ranked signature and variables; rewrite rules $l\to r$ satisfy $l$ is not a variable and $\operatorname{Var}(r)\subseteq\operatorname{Var}(l)$; a reduction replaces one occurrence of $l\sigma$ by $r\sigma$ in a context; completion state: a pair $(E,R)$ of unoriented equations and oriented rules; equality: the congruence generated by $E\cup R$.}
\noindent\emph{Controlling sources:} \cite{KnuthBendix1970,Huet1980,BaaderNipkow1998}.
\hbrule{rule:term-rewriting-step}{Rewrite}{l\to r\in R}{C[l\sigma]\to_R C[r\sigma]}{$C[-]$ is a one-hole context and $\sigma$ a first-order substitution}
\hbrule{rule:term-rewriting-orient}{Orient}{E\cup\{s\approx t\},R}{E,R\cup\{s\to t\}}{$s\succ t$ for the fixed reduction ordering $\succ$}
\hbrule{rule:term-rewriting-simplify}{Simplify}{E\cup\{s\approx t\},R}{E\cup\{s'\approx t\},R}{$s\to_R s'$; symmetric in the two equation sides}
\hbrule{rule:term-rewriting-compose}{Compose}{E,R\cup\{s\to t\}}{E,R\cup\{s\to t'\}}{$t\to_R t'$}
\hbrule{rule:term-rewriting-collapse}{Collapse}{E,R\cup\{s\to t\}}{E\cup\{s'\approx t\},R}{$s\to_{R\setminus\{s\to t\}}s'$ at a nonroot or strictly smaller redex}
\hbrule{rule:term-rewriting-deduce}{Deduce}{E,R}{E\cup\{s\approx t\},R}{$s\leftarrow_R u\to_R t$ is a critical overlap not yet joinable}
