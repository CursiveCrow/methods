\calculussheet{sheet:hilbert-fol}{Hilbert Calculus for Classical First-Order Logic}{Language: the $\neg/\to/\forall/\exists$ fragment of $\mathsf{Fm}(\mathcal{L})$ (metavariables $\varphi,\psi,\chi$ range over formulas of that fragment; the book's remaining primitives $\bot,\land,\lor$ are reached by the standard classical definitions when this calculus is used for the full language); sequent shape: no sequents --- Hilbert-style derivability $\Gamma\vdash\varphi$ via finite dependency-annotated sequences of formulas, each a member of $\Gamma$, an instance of an axiom schema, or obtained from earlier lines by modus ponens, generalization, or the displayed existential-elimination rule; negation: primitive; equality: none (no equality axiom schemas --- the pure quantificational calculus).}
\hbrule{rule:hilbert-fol-a1}{A1}{}{\varphi\to(\psi\to\varphi)}{\none}
\hbrule{rule:hilbert-fol-a2}{A2}{}{(\varphi\to(\psi\to\chi))\to((\varphi\to\psi)\to(\varphi\to\chi))}{\none}
\hbrule{rule:hilbert-fol-a3}{A3}{}{(\neg\psi\to\neg\varphi)\to((\neg\psi\to\varphi)\to\psi)}{\none}
\hbrule{rule:hilbert-fol-mp}{MP}{\varphi\to\psi\qquad\varphi}{\psi}{\none}
\hbrule{rule:hilbert-fol-forall-inst}{$\forall 1$}{}{\forall x\,\varphi\to\varphi[t/x]}{substitutability: $[t/x]$ is the book's total capture-avoiding substitution (bound variables renamed as required), so the classical proviso that $t$ be substitutable for $x$ in $\varphi$ is automatically satisfied}
\hbrule{rule:hilbert-fol-forall-dist}{$\forall 2$}{}{\forall x\,(\varphi\to\psi)\to(\varphi\to\forall x\,\psi)}{$x\notin\mathrm{FV}(\varphi)$, i.e.\ $x$ does not occur free in the antecedent formula $\varphi$}
\hbrule{rule:hilbert-fol-exists-intro}{$\exists 1$}{}{\varphi[t/x]\to\exists x\,\varphi}{substitutability: $[t/x]$ is the total capture-avoiding substitution, so the classical substitutability proviso is automatically satisfied}
\hbrule{rule:hilbert-fol-exists-elim}{$\exists E$}{\varphi\to\psi}{\exists x\,\varphi\to\psi}{$x\notin\mathrm{FV}(\psi)$ and $x$ is not free in any premise on which the displayed premise line depends}
\hbrule{rule:hilbert-fol-gen}{Gen}{\varphi}{\forall x\,\varphi}{eigenvariable restriction: $x$ is not free in any premise of $\Gamma$ on which the derivation of $\varphi$ depends; unrestricted when $\varphi$ is derived from no premises}
