% NEW for Phase 4.  Exact standard-fullpath CTL* calculus from Reynolds 2001:
% the bundled-tree base plus LC and AA.  Uniform substitution is deliberately
% not an inference rule because C14 is restricted to atomic propositions.
\calculussheet{sheet:hilbert-ctl-star}{Reynolds Calculus for Full Computation Tree Logic $\mathsf{CTL}^{*}$}{Language: state formulas are generated from atoms and Boolean connectives and $\mathsf A\alpha$; path formulas contain state formulas and are closed under Boolean connectives, $\mathsf X\alpha$, and $\alpha\mathbin{\mathsf U}\beta$, with $\mathsf E\alpha\defeq\neg\mathsf A\neg\alpha$, $\mathsf F\alpha\defeq\topv\mathbin{\mathsf U}\alpha$, and $\mathsf G\alpha\defeq\neg\mathsf F\neg\alpha$; sequent shape: no sequents --- theoremhood $\derives_{\mathsf{CTL}^{*}}A$ for standard fullpath validity; negation: primitive (classical); equality: none.}
\hbrule{rule:p08-hilbert-ctl-star-taut}{C0}{}{\alpha}{$\alpha$ is a substitution instance of a classically valid propositional formula; this is an axiom schema, not a uniform-substitution inference rule}
\hbrule{rule:p08-hilbert-ctl-star-c1}{C1}{}{\mathsf F\neg\neg\alpha\leftrightarrow\mathsf F\alpha}{\none}

\hbrule{rule:p08-hilbert-ctl-star-c3}{C3}{}{\mathsf G\alpha\to\bigl(\alpha\land\mathsf X\alpha\land\mathsf X\mathsf G\alpha\bigr)}{\none}
\hbrule{rule:p08-hilbert-ctl-star-c4}{C4}{}{\mathsf X\neg\alpha\leftrightarrow\neg\mathsf X\alpha}{\none}
\hbrule{rule:p08-hilbert-ctl-star-c5}{C5}{}{\mathsf X(\alpha\to\beta)\to(\mathsf X\alpha\to\mathsf X\beta)}{\none}
\hbrule{rule:p08-hilbert-ctl-star-c6}{C6}{}{\mathsf G(\alpha\to\mathsf X\alpha)\to(\alpha\to\mathsf G\alpha)}{\none}

\hbrule{rule:p08-hilbert-ctl-star-c8}{C8}{}{\alpha\mathbin{\mathsf U}\beta\to\mathsf F\beta}{\none}
\hbrule{rule:p08-hilbert-ctl-star-c9}{C9}{}{\mathsf A(\alpha\to\beta)\to(\mathsf A\alpha\to\mathsf A\beta)}{\none}

\hbrule{rule:p08-hilbert-ctl-star-aa}{AA}{\derives_{\mathsf{CTL}^{*}}(\theta\to\psi)}{\derives_{\mathsf{CTL}^{*}}\psi}{theorem-only; there are disjoint atom sets $L,Q$ such that $\psi$ uses only atoms from $L$ and $\theta$ is functionally $L+Q$-expandable, meaning: $Q$ is finite, $b_0\in Q$, $\{\alpha_1,\ldots,\alpha_n\}$ is a finite propositional partition by state formulas over $L$ (each is a Boolean combination of atoms and formulas $\mathsf E\beta$, distinct members are classically inconsistent, and $\bigvee_{i=1}^{n}\alpha_i$ is a tautology), there is $\rho:Q\times\{1,\ldots,n\}\to Q$, and $\theta=b_0\land\mathsf A\mathsf G\!\bigwedge_{b\in Q,1\le i\le n}((b\land\alpha_i)\to\mathsf A\mathsf X\rho(b,i))\land\mathsf A\mathsf G\!\bigwedge_{b\ne b'\in Q}\neg(b\land b')$; ``functionally $L+Q$-expandable'' is this eliminable metalinguistic side condition, needed here to distinguish the displayed finite functional expansion from mere semantic expandability, and AA is its only use}
