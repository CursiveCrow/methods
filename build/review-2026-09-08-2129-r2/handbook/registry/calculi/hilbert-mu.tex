% NEW for Phase 4: the equational system of Kozen 1983, sec. 5, whose
% completeness was proved by Walukiewicz.  This is not a Hilbert-K grammar
% with an arbitrarily chosen fixed-point extension.
\calculussheet{sheet:hilbert-mu}{Kozen's Equational Calculus for $L_{\mu}$}{Language: exactly Kozen's finitary $L_{\mu}$ language --- propositional constants $p,q,\ldots$ (including $\bottom,\topv$), fixed-point variables $Z,Y,\ldots$, and program constants $a,b,\ldots$, with formulas generated by $Z$, $p$, $A\lor B$, $\neg A$, $\langle a\rangle A$, and $\mu Z.A$; in the last form every free occurrence of $Z$ in $A$ lies under an even number of negations.  Conjunction, implication, $[a]A$, and $\nu Z.A$ are the defined operators stated in Definition~\ref{def:p08-modal-mu-semantics}.  Judgment shape: equations $A=B$ and inequalities $A\leq B$, where $A\leq B$ abbreviates $A\lor B=B$; theoremhood is $A=\topv$.  The logical base is Boolean equational logic, including replacement of equals by equals and capture-avoiding uniform substitution whenever the formation restriction is preserved.  There are no sequents; negation is primitive; formula equality and inequality are metalinguistic judgments.}
In addition to the Boolean-algebra axioms and rules of equational logic, the nonlogical axioms and rule are exactly:
\hbrule{rule:p08-hilbert-mu-join}{Modal join}{}{\derives_{L_{\mu}}\bigl(\langle a\rangle A\lor\langle a\rangle B=\langle a\rangle(A\lor B)\bigr)}{\none}
\hbrule{rule:p08-hilbert-mu-mixed}{Modal meet inequality}{}{\derives_{L_{\mu}}\bigl(\langle a\rangle A\land[a]B\leq\langle a\rangle(A\land B)\bigr)}{\none}
\hbrule{rule:p08-hilbert-mu-zero}{Modal zero}{}{\derives_{L_{\mu}}\bigl(\langle a\rangle\bottom=\bottom\bigr)}{\none}
\hbrule{rule:p08-hilbert-mu-unfold}{$\mu$-prefix}{}{\derives_{L_{\mu}}\bigl(A[\mu Z.A/Z]\leq\mu Z.A\bigr)}{$\mu Z.A$ is free for $Z$ in $A$}
\hbrule{rule:p08-hilbert-mu-park}{Park}{\derives_{L_{\mu}}\bigl(A[Y/Z]\leq Y\bigr)}{\derives_{L_{\mu}}\bigl(\mu Z.A\leq Y\bigr)}{$Y$ does not occur in $A$; substitutions are capture-avoiding and preserve positivity}
