% Repair of the phase4-salvage draft: expansion never consumes a formula.
% The principal occurrence remains on the branch and is marked expanded; that
% occurrence may not be principal again on a descendant branch.  At a split,
% the retained branch (including expansion marks) is copied to both children.
\calculussheet{sheet:tableaux-cpl}{Classical Propositional Analytic Tableaux}{Language: full $\mathsf{Fm}_{\mathsf{CPL}}$ with primitives $\bot,\neg,\land,\lor,\to$; shape: a tableau rooted at a finite set $\Gamma$ is a finite tree of branches, each branch being a retained list of formula occurrences together with metalinguistic expansion marks; applying a rule appends its conclusion formula or formulas but never deletes its principal occurrence, which is marked expanded and may not be principal again below that point; a branch closes when it contains $\bot$ or both $\varphi$ and $\neg\varphi$, is open otherwise, and is saturated when every expandable occurrence is marked; negation: primitive, with one dual expansion for each connective; equality: none (propositional language).}
\hbrule{rule:tableaux-cpl-and}{$\land$}{\varphi\land\psi}{\varphi,\;\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-cpl-neg-and}{$\neg\land$}{\neg(\varphi\land\psi)}{\neg\varphi\;\mid\;\neg\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-cpl-or}{$\lor$}{\varphi\lor\psi}{\varphi\;\mid\;\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-cpl-neg-or}{$\neg\lor$}{\neg(\varphi\lor\psi)}{\neg\varphi,\;\neg\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-cpl-imp}{$\to$}{\varphi\to\psi}{\neg\varphi\;\mid\;\psi}{branching expansion: retain and mark the principal occurrence, copy the marked branch, and append one displayed alternative to each child}
\hbrule{rule:tableaux-cpl-neg-imp}{$\neg\to$}{\neg(\varphi\to\psi)}{\varphi,\;\neg\psi}{linear expansion: retain and mark the principal occurrence, then append both conclusions on every open branch through it}
\hbrule{rule:tableaux-cpl-neg-neg}{$\neg\neg$}{\neg\neg\varphi}{\varphi}{linear expansion: retain and mark the principal occurrence, then append the conclusion on every open branch through it}
\hbrule{rule:tableaux-cpl-close}{close}{\varphi\qquad\neg\varphi}{\text{closed branch}}{a branch containing both displayed formulas, or containing $\bot$, is closed and receives no further expansion; a tableau is closed iff every branch is closed}
