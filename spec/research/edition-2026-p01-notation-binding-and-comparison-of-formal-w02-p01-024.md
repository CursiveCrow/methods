```python
"""Extract Greek, Fraktur, blackboard and Parts 1-2 rows of handbook/registry/symbols.csv
as longtable bodies.

Run from the repository root.  Writes greek-family.tex, greek-domains.tex, fraktur.tex,
blackboard.tex and parts12.tex next to this script.  Cells reuse tools/symbols2tex.py
rendering; scopes are printed literally as recorded (only invisible break points added).
"""
import re
import sys
from pathlib import Path

sys.path.insert(0, 'tools')
import symbol_registry as reg
from symbols2tex import glyph_cell, text_cell

OUT = Path(__file__).parent
rows, errors = reg.read_rows('handbook/registry/symbols.csv')
assert not errors, errors[:5]

GREEK = ('alpha|beta|gamma|delta|epsilon|zeta|eta|theta|iota|kappa|lambda|mu|nu|xi|pi|rho|sigma|tau|'
         'upsilon|phi|chi|psi|omega|Gamma|Delta|Theta|Lambda|Xi|Pi|Sigma|Upsilon|Phi|Psi|Omega')
GREEK_RE = re.compile(r'\\(?:var)?(?:' + GREEK + r')(?![a-zA-Z])|[Ͱ-Ͽ]')
FAMILY_RE = re.compile(r'metavariable|context|substitution', re.I)
FRAK_RE = re.compile(r'\\mathfrak(?![a-zA-Z])')
# \N \Z \Q \R are handbook.sty shorthands for blackboard letters
BB_RE = re.compile(r'\\mathbb(?![a-zA-Z])|\\[NZQR](?![a-zA-Z])')
ESC = {'\\': r'\textbackslash{}', '{': r'\{', '}': r'\}', '_': r'\_', '%': r'\%', '&': r'\&',
       '#': r'\#', '$': r'\$', '^': r'\textasciicircum{}', '~': r'\textasciitilde{}', '–': '--'}


IDENT_RE = re.compile('[a-z]{2,}[0-9]')


def ident_tt(text, escape):
    # source tags such as src063 or har2001 are identifiers, not prose: typewriter
    out = []
    for tok in text.split(' '):
        plain = not any(c in tok for c in chr(92) + '{}$')
        if plain and IDENT_RE.search(tok):
            out.append('TTOPEN' + escape(tok) + '}')
        else:
            out.append(escape(tok) if escape else tok)
    return ' '.join(out).replace('TTOPEN', chr(92) + 'texttt{')


def scope(r):
    out = ident_tt(r['scope'], lambda s: ''.join(ESC.get(c, c) for c in s))
    # break points after label separators; the printed characters are unchanged
    return re.sub(r'([.:/]|(?<!-)-)(?=[A-Za-z0-9])', lambda m: m.group(1) + r'\allowbreak{}', out)


def breakable(cell):
    """Allow line breaks after top-level commas and semicolons inside inline math."""
    out, depth, math, i = [], 0, False, 0
    while i < len(cell):
        two = cell[i:i + 2]
        if two in ('\\(', '\\)'):
            math = two == '\\('
            out.append(two)
            i += 2
            continue
        if cell[i] == '\\' and i + 1 < len(cell):
            out.append(two)
            i += 2
            continue
        c = cell[i]
        depth += c == '{'
        depth -= c == '}'
        out.append(c)
        if math and depth == 0 and c in ',;':
            out.append(r'\allowbreak ')
        i += 1
    return ''.join(out)


def part(r):
    return r['_scope'].part


def body(selected, cols):
    lines, last = [], object()
    order = sorted(selected, key=lambda r: (0 if part(r) == 'global' else 1,
                                            0 if part(r) == 'global' else part(r), r['_line']))
    for r in order:
        p = part(r)
        if p != last:
            title = 'global scope' if p == 'global' else f'scopes recorded under P{p}'
            lines.append(rf'\multicolumn{{{cols}}}{{@{{}}l}}{{\emph{{{title}}}}}\\')
            last = p
        cells = [breakable(glyph_cell(r['glyph'])), ident_tt(text_cell(r['role']), lambda s: s), scope(r)]
        if cols == 4:
            cells.append(text_cell(r['precedence']))
        lines.append(' & '.join(cells) + r' \\')
    return '\n'.join(lines) + '\n'


tables = {
    'greek-domains': [r for r in rows if GREEK_RE.search(r['glyph'])],
    'fraktur': [r for r in rows if FRAK_RE.search(r['glyph'])],
    'blackboard': [r for r in rows if BB_RE.search(r['glyph'])],
}
tables['greek-family'] = [r for r in tables['greek-domains'] if FAMILY_RE.search(r['role'])]
for name, sel in tables.items():
    (OUT / f'{name}.tex').write_text(body(sel, 3), encoding='utf-8', newline='\n')
parts12 = [r for r in rows if part(r) in ('global', 1, 2)]
(OUT / 'parts12.tex').write_text(body(parts12, 4), encoding='utf-8', newline='\n')
print({k: len(v) for k, v in tables.items()}, 'parts12', len(parts12))
```
